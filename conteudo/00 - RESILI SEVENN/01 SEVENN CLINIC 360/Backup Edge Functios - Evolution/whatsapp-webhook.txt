import { serve } from "https://deno.land/std@0.168.0/http/server.ts"
import { createClient } from 'https://esm.sh/@supabase/supabase-js@2.39.3'

const corsHeaders = {
    'Access-Control-Allow-Origin': '*',
    'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
}

serve(async (req) => {
    if (req.method === 'OPTIONS') return new Response('ok', { headers: corsHeaders })

    try {
        const body = await req.json()
        console.log("[PAYLOAD] Recebido:", JSON.stringify(body));

        // Tentar buscar o nome da instância em diversos locais possíveis do payload (Evolution v1 e v2)
        const instanceName = body.instance ||
            body.instanceName ||
            body.data?.instance ||
            body.data?.instanceName ||
            (body.data?.key?.remoteJid ? body.instance : null);

        // Normalização de evento: messages.upsert -> MESSAGES_UPSERT
        const event = body.event ? body.event.toUpperCase().replace(/\./g, '_') : null;
        const data = body.data;

        console.log(`[EVENTO] ${body.event} -> ${event} recebido para instância: ${instanceName}`);
        if (!instanceName) {
            console.error("[ERRO] Nome da instância não identificado no payload.");
            // Não paramos aqui para tentar processar se houver dados, mas o processamento de mensagens vai falhar sem instância
        }

        // 1. Conectar com Service Role
        const supabaseAdmin = createClient(
            Deno.env.get('SUPABASE_URL') ?? '',
            Deno.env.get('SUPABASE_SERVICE_ROLE_KEY') ?? ''
        )

        // --- FLUXO 1: ATUALIZAÇÃO DE CONEXÃO ---
        if (event === 'CONNECTION_UPDATE' || event === 'LOGOUT_INSTANCE') {
            const state = data.state || (event === 'LOGOUT_INSTANCE' ? 'logout' : 'disconnected');
            const statusMap: Record<string, string> = {
                'open': 'connected',
                'connecting': 'connecting',
                'close': 'disconnected',
                'logout': 'disconnected'
            };

            const newStatus = statusMap[state] || 'disconnected';

            await supabaseAdmin
                .from('whatsapp_connections')
                .update({ status: newStatus })
                .eq('instance_name', instanceName);

            console.log(`[STATUS] Instância ${instanceName} -> ${newStatus} (${state})`);
            return new Response(JSON.stringify({ success: true, type: 'status_update' }), {
                headers: { ...corsHeaders, 'Content-Type': 'application/json' }
            });
        }

        // --- FLUXO 2: MENSAGEM APAGADA ---
        if (event === 'MESSAGES_DELETE') {
            const externalId = data.key?.id;
            if (externalId) {
                const { error } = await supabaseAdmin
                    .from('messages')
                    .update({
                        content: "🚫 _Mensagem apagada_",
                        metadata: { deleted: true, deleted_at: new Date().toISOString() }
                    })
                    .eq('external_id', externalId);

                if (error) console.error("Erro ao marcar mensagem como deletada:", error);
            }
            return new Response(JSON.stringify({ success: true, type: 'message_delete' }), {
                headers: { ...corsHeaders, 'Content-Type': 'application/json' }
            });
        }

        // --- FLUXO 3: DIGITANDO (PRESENCE_UPDATE) ---
        if (event === 'PRESENCE_UPDATE') {
            const customerNumber = data.key?.remoteJid?.split('@')[0];
            const isComposing = data.presence?.lastKnownPresence === 'composing';

            if (customerNumber) {
                // Buscamos a conexão para saber a clínica
                const { data: connection } = await supabaseAdmin
                    .from('whatsapp_connections')
                    .select('clinic_id')
                    .eq('instance_name', instanceName)
                    .single();

                if (connection) {
                    // Atualizamos o status de digitação no chat
                    // Isso disparará o Realtime para quem estiver ouvindo a tabela 'chats'
                    await supabaseAdmin
                        .from('chats')
                        .update({ is_typing: isComposing })
                        .eq('clinic_id', connection.clinic_id)
                        .eq('customer_number', customerNumber);
                }
            }
            return new Response(JSON.stringify({ success: true, type: 'presence_update' }), {
                headers: { ...corsHeaders, 'Content-Type': 'application/json' }
            });
        }

        // --- FLUXO 4: ATUALIZAÇÃO DE STATUS DE MENSAGEM (Ticks) ---
        if (event === 'MESSAGES_UPDATE') {
            const updates = Array.isArray(data) ? data : [data];
            console.log(`[STATUS_MSG] Processando ${updates.length} atualizações de status.`);

            for (const item of updates) {
                // Evolution v2 pode mandar o id em keyId em vez de key.id
                const externalId = item.key?.id || item.keyId || item.id;
                // Evolution v2 pode mandar direto no item.status ou em item.update.status
                const rawStatus = item.status || item.update?.status;
                const evoStatus = rawStatus !== undefined ? String(rawStatus).toUpperCase() : null;

                if (externalId && evoStatus) {
                    const statusMap: Record<string, string> = {
                        '0': 'sent', 'PENDING': 'sent',
                        '1': 'sent', 'SENT': 'sent', 'RECEIVED': 'sent', 'SERVER_ACK': 'sent',
                        '2': 'delivered', 'DELIVERY_ACK': 'delivered', 'DELIVERED': 'delivered', 'DELIV': 'delivered',
                        '3': 'read', 'READ': 'read',
                        '4': 'read', 'PLAYED': 'read'
                    };

                    const newStatus = statusMap[evoStatus];
                    if (newStatus) {
                        const { error } = await supabaseAdmin
                            .from('messages')
                            .update({ status: newStatus })
                            .eq('external_id', externalId);

                        if (error) {
                            console.error(`[STATUS_MSG] Erro ao atualizar mensagem ${externalId}:`, error);
                        } else {
                            console.log(`[STATUS_MSG] Mensagem ${externalId} -> ${newStatus} (${evoStatus})`);
                        }
                    } else {
                        console.log(`[STATUS_MSG] Status ignorado ou não mapeado: ${evoStatus}`);
                    }
                } else {
                    console.log(`[STATUS_MSG] Falha ao extrair externalId ou status. Item: ${JSON.stringify(item)}`);
                }
            }
            return new Response(JSON.stringify({ success: true, type: 'message_update' }), {
                headers: { ...corsHeaders, 'Content-Type': 'application/json' }
            });
        }

        // --- FLUXO 5: NOVA MENSAGEM (OU ENVIADA PELO CELULAR) ---
        if (event === 'MESSAGES_UPSERT' || event === 'SEND_MESSAGE') {
            // Evolution v2 pode enviar em 'message' ou 'messages[0]'
            const message = data.message || (data.messages && data.messages[0]);

            // Ignora se não houver conteúdo ou for broadcast de status ou GRUPOS
            const remoteJid = data.key?.remoteJid || (data.messages && data.messages[0]?.key?.remoteJid);
            if (!message || remoteJid === 'status@broadcast' || remoteJid?.includes('@g.us')) {
                console.log(`[SKIP] Mensagem ignorada. Tipo: ${!message ? 'sem conteúdo' : remoteJid}`);
                return new Response(JSON.stringify({ success: true, skip: 'ignored_type' }), {
                    headers: { ...corsHeaders, 'Content-Type': 'application/json' }
                });
            }

            // Normaliza o acesso aos dados da mensagem (pode vir no data ou no data.messages[0])
            const msgData = data.messages ? data.messages[0] : data;
            const customerNumber = msgData.key.remoteJid.split('@')[0]
            const isFromMe = msgData.key.fromMe
            const deviceSource = (data.source as string | undefined) || null

            // Suporte para múltiplos tipos de mensagem
            const isAudio = !!(message?.audioMessage);
            const isImage = !!(message?.imageMessage);
            const isVideo = !!(message?.videoMessage);
            const isReaction = !!(message?.reactionMessage);
            const reactionText = message?.reactionMessage?.text;

            if (isReaction) {
                console.log(`[REACTION] Detectada. text="${reactionText}" typeof=${typeof reactionText} isEmpty=${reactionText === ''} isNull=${reactionText === null} isUndefined=${reactionText === undefined}`);
                console.log(`[REACTION] Full reactionMessage: ${JSON.stringify(message.reactionMessage)}`);
            }

            let content = message?.conversation ||
                message?.extendedTextMessage?.text ||
                message?.imageMessage?.caption ||
                message?.videoMessage?.caption ||
                (isReaction ? reactionText : null) ||
                "📎 Arquivo/Mídia";

            if (isAudio) content = "🎤 Áudio";

            const externalId = msgData.key.id;

            // Ignorar remoção de reação (text vazio = lead removeu a reação)
            if (isReaction && (reactionText === '' || reactionText === null || reactionText === undefined)) {
                console.log(`[SKIP] Remoção/reação inválida ignorada para ${externalId}`);
                return new Response(JSON.stringify({ success: true, skip: 'reaction_removal' }), {
                    headers: { ...corsHeaders, 'Content-Type': 'application/json' }
                });
            }

            // DEDUPLICAÇÃO: Verificar se mensagem já existe pelo external_id
            const { data: existingMsg } = await supabaseAdmin
                .from('messages')
                .select('id')
                .eq('external_id', externalId)
                .maybeSingle();

            // Resolver UUID da mensagem alvo para reações
            let reactionTargetUUID: string | undefined;
            if (isReaction) {
                const reactionTargetWhatsAppId = message.reactionMessage.key?.id;
                if (reactionTargetWhatsAppId) {
                    const { data: targetMsg } = await supabaseAdmin
                        .from('messages')
                        .select('id')
                        .eq('external_id', reactionTargetWhatsAppId)
                        .maybeSingle();
                    reactionTargetUUID = targetMsg?.id ?? undefined;
                    console.log(`[REACTION] Target WhatsApp ID: ${reactionTargetWhatsAppId} → UUID: ${reactionTargetUUID ?? 'não encontrado'}`);
                }
            }

            if (existingMsg) {
                console.log(`[DEDUPLICAÇÃO] Mensagem ${externalId} já existe. Pulando inserção.`);
                return new Response(JSON.stringify({ success: true, message: 'Duplicate ignored' }), {
                    headers: { ...corsHeaders, 'Content-Type': 'application/json' }
                });
            }

            const pushName = data.pushName || msgData.pushName || customerNumber;
            // Tenta obter a URL da foto pelo webhook. Se não vier e for de cliente, consulta a Evolution.
            let profileUrl = data.profilePicUrl || data.sender?.profilePicUrl || body.data?.profilePicUrl || data.profileUrl || null;

            if (!profileUrl && !isFromMe) {
                try {
                    const evolutionUrl = Deno.env.get('EVOLUTION_API_URL');
                    const evolutionKey = Deno.env.get('EVOLUTION_API_KEY');
                    if (evolutionUrl && evolutionKey) {
                        const req = await fetch(`${evolutionUrl}/chat/fetchProfilePictureUrl/${instanceName}`, {
                            method: 'POST',
                            headers: { 'Content-Type': 'application/json', 'apikey': evolutionKey },
                            body: JSON.stringify({ number: customerNumber })
                        });
                        if (req.ok) {
                            const res = await req.json();
                            if (res && res.profilePictureUrl) {
                                profileUrl = res.profilePictureUrl;
                                console.log(`[AVATAR] Foto dinâmica obtida via API para ${customerNumber}: ${profileUrl}`);
                            }
                        }
                    }
                } catch (e) {
                    console.error("[AVATAR] Erro ao buscar foto dinamicamente:", e);
                }
            }

            // Buscar URL da mídia se for áudio/imagem/vídeo
            let mediaUrl = null;
            if (isAudio || isImage || isVideo) {
                // Em Evolution V2, a URL direta pode vir no payload se configurada a conversão/upload
                mediaUrl = message?.audioMessage?.url || message?.imageMessage?.url || message?.videoMessage?.url || null;

                // Se temos base64 (muito comum em Evolution v2), vamos subir para o Storage do Supabase
                // para garantir que o áudio seja tocável (URLs .enc do WhatsApp não tocam no browser)
                const base64 = body.base64 || msgData?.message?.base64 || body.data?.message?.base64;
                if (base64) {
                    try {
                        console.log(`[MEDIA] Processando base64 para ${isAudio ? 'áudio' : 'imagem/vídeo'}...`);
                        const fileExt = isAudio ? 'opus' : isImage ? 'png' : 'mp4';
                        const fileName = `${externalId}.${fileExt}`;
                        const filePath = `${instanceName || 'unknown'}/${customerNumber}/${fileName}`;

                        // Converter base64 para bytes
                        const binaryStr = atob(base64);
                        const bytes = new Uint8Array(binaryStr.length);
                        for (let i = 0; i < binaryStr.length; i++) {
                            bytes[i] = binaryStr.charCodeAt(i);
                        }

                        const bucketName = 'lead-images';
                        const { error: uploadError } = await supabaseAdmin.storage
                            .from(bucketName)
                            .upload(filePath, bytes, {
                                contentType: isAudio ? 'audio/ogg; codecs=opus' : isImage ? 'image/png' : 'video/mp4',
                                upsert: true
                            });

                        if (!uploadError) {
                            // Gerar URL assinada de longa duração (2 semanas) já que o bucket é privado
                            const { data: signedData, error: signedError } = await supabaseAdmin.storage
                                .from(bucketName)
                                .createSignedUrl(filePath, 60 * 60 * 24 * 14); // 14 dias

                            if (!signedError && signedData) {
                                mediaUrl = signedData.signedUrl;
                                console.log(`[MEDIA] Upload concluído e URL assinada gerada: ${mediaUrl}`);
                            } else {
                                console.error("[MEDIA] Erro ao gerar URL assinada:", signedError);
                                // Fallback para publicUrl caso o bucket seja público
                                const { data: { publicUrl } } = supabaseAdmin.storage.from(bucketName).getPublicUrl(filePath);
                                mediaUrl = publicUrl;
                            }
                        } else {
                            console.error("[MEDIA] Erro ao subir para Storage:", uploadError);
                        }
                    } catch (e) {
                        console.error("[MEDIA] Erro crítico ao processar base64:", e);
                    }
                }
            }

            // Buscar qual clínica/integração pertence
            const { data: connection } = await supabaseAdmin
                .from('whatsapp_connections')
                .select('id, clinic_id')
                .eq('instance_name', instanceName)
                .single()

            if (!connection) throw new Error("Instância não encontrada no banco.")

            // 3. Gerenciar o CHAT (Busca ou Cria)
            let { data: chat, error: chatError } = await supabaseAdmin
                .from('chats')
                .select('id, unread_count, device_source')
                .eq('clinic_id', connection.clinic_id)
                .eq('customer_number', customerNumber)
                .maybeSingle()

            if (chatError) throw chatError

            if (!chat) {
                // Para mensagens enviadas pelo celular (fromMe: true), pushName é o SEU próprio nome,
                // não o do destinatário. Buscamos o perfil real do contato para ter nome e foto corretos.
                let contactName = pushName;
                let contactAvatar = profileUrl;
                if (isFromMe) {
                    try {
                        const evoUrl = Deno.env.get('EVOLUTION_API_URL');
                        const evoKey = Deno.env.get('EVOLUTION_API_KEY');
                        if (evoUrl && evoKey) {
                            const profileReq = await fetch(`${evoUrl}/chat/fetchProfile/${instanceName}`, {
                                method: 'POST',
                                headers: { 'Content-Type': 'application/json', 'apikey': evoKey },
                                body: JSON.stringify({ number: customerNumber })
                            });
                            if (profileReq.ok) {
                                const profileRes = await profileReq.json();
                                if (profileRes?.name) contactName = profileRes.name;
                                if (profileRes?.profilePictureUrl) contactAvatar = profileRes.profilePictureUrl;
                                console.log(`[AVATAR-OUTGOING] Perfil do destinatário obtido: ${contactName}`);
                            }
                        }
                    } catch (e) {
                        console.error("[AVATAR-OUTGOING] Erro ao buscar perfil do destinatário:", e);
                    }
                }

                const { data: newChat, error: newChatErr } = await supabaseAdmin
                    .from('chats')
                    .insert({
                        clinic_id: connection.clinic_id,
                        whatsapp_instance_id: connection.id,
                        customer_number: customerNumber,
                        customer_name: contactName,
                        customer_avatar: contactAvatar,
                        last_message: content,
                        unread_count: isFromMe ? 0 : 1,
                        conversation_status: 'open',
                        device_source: !isFromMe ? deviceSource : null,
                    })
                    .select()
                    .single()

                if (newChatErr) {
                    console.error("[ERRO] Falha ao criar chat:", newChatErr);
                    throw newChatErr;
                }
                chat = newChat
            } else {
                const updateData: any = {
                    last_message: content,
                    updated_at: new Date().toISOString(),
                    is_typing: false // Sempre desliga o digitando ao chegar mensagem
                }

                // Atualiza nome/foto se mudaram ou eram nulos (Apenas se for mensagem do cliente)
                if (!isFromMe) {
                    if (pushName && pushName !== customerNumber && (!chat?.customer_name || chat.customer_name === customerNumber)) {
                        updateData.customer_name = pushName;
                    }

                    if (profileUrl && (!chat?.customer_avatar || chat.customer_avatar.includes('default'))) {
                        updateData.customer_avatar = profileUrl;
                    }

                    // Sempre reabre a conversa se chegar mensagem do cliente
                    updateData.conversation_status = 'open';
                }

                // Incrementa unread se for mensagem recebida, zera se for enviada
                if (!isFromMe) {
                    updateData.unread_count = (chat.unread_count || 0) + 1;
                } else {
                    updateData.unread_count = 0;
                }

                // Preenche device_source apenas se ainda não foi definido (primeira mensagem recebida)
                if (!isFromMe && deviceSource && !chat.device_source) {
                    updateData.device_source = deviceSource;
                }

                const { error: updErr } = await supabaseAdmin.from('chats').update(updateData).eq('id', chat.id)
                if (updErr) console.error("[ERRO] Falha ao atualizar chat:", updErr);
            }

            // 4. Salvar a MENSAGEM
            let { data: lead } = await supabaseAdmin
                .from('leads')
                .select('id')
                .eq('clinic_id', connection.clinic_id)
                .eq('phone', customerNumber)
                .maybeSingle()

            // Automação: Criar Lead se não existir
            if (!lead && !isFromMe) {
                console.log(`[AUTOMAÇÃO] Criando novo lead para o número: ${customerNumber}`);

                // Buscar id e nome da etapa 'Novo Lead' (slug 'new' ou 'novo_lead' ou 'nova_etapa')
                const { data: stage } = await supabaseAdmin
                    .from('pipeline_stages')
                    .select('id, name')
                    .eq('clinic_id', connection.clinic_id)
                    .or('slug.eq.new,slug.eq.novo_lead,slug.eq.nova_etapa,name.ilike.%Nova%Etapa%,name.ilike.%Novo%Lead%')
                    .order('order', { ascending: true })
                    .limit(1)
                    .maybeSingle();

                if (stage) {
                    const { data: newLead, error: leadErr } = await supabaseAdmin
                        .from('leads')
                        .insert({
                            clinic_id: connection.clinic_id,
                            pipeline_stage_id: stage.id,
                            name: pushName || customerNumber,
                            phone: customerNumber,
                            avatar_url: profileUrl,
                            status: stage.name || 'Novo Lead',
                            source: 'Whatsapp'
                        })
                        .select('id')
                        .single();

                    if (!leadErr && newLead) {
                        lead = newLead;
                        console.log(`[SUCESSO] Lead criado: ${lead.id}`);
                    } else {
                        console.error("[ERRO] Falha ao criar lead automático:", leadErr);
                    }
                } else {
                    console.error("[AVISO] Etapa 'novo_lead' não encontrada para esta clínica.");
                }
            } else if (lead && !isFromMe) {
                // Se o lead já existe, opcionalmente atualizar o avatar se estiver vazio
                if (profileUrl) {
                    await supabaseAdmin
                        .from('leads')
                        .update({ avatar_url: profileUrl })
                        .eq('id', lead.id)
                        .is('avatar_url', null);
                }
            }

            // Não salvar mensagem se não há lead (ocorre quando fromMe:true e contato ainda não é lead)
            // O lead será criado quando o contato responder, e as mensagens futuras serão salvas normalmente
            if (!lead?.id) {
                console.log(`[SKIP-MSG] Sem lead para ${customerNumber} (fromMe:${isFromMe}). Mensagem ${msgData.key.id} não salva.`);
                return new Response(JSON.stringify({ success: true }), {
                    headers: { ...corsHeaders, 'Content-Type': 'application/json' }, status: 200
                });
            }

            const { error: msgError } = await supabaseAdmin
                .from('messages')
                .insert({
                    clinic_id: connection.clinic_id,
                    chat_id: chat.id,
                    whatsapp_instance_id: connection.id,
                    lead_id: lead.id,
                    content: content,
                    sender: isFromMe ? 'staff' : 'user',
                    direction: isFromMe ? 'clinica' : 'paciente',
                    channel: 'whatsapp',
                    status: 'sent',
                    external_id: msgData.key.id,
                    metadata: {
                        pushName: data.pushName,
                        isFromMe,
                        type: isAudio ? 'audio' : isImage ? 'image' : isVideo ? 'video' : isReaction ? 'reaction' : 'text',
                        url: mediaUrl,
                        fileType: isAudio ? 'audio' : isImage ? 'image' : isVideo ? 'video' : undefined,
                        reactionEmoji: isReaction ? message.reactionMessage.text : undefined,
                        reactionTargetId: isReaction ? message.reactionMessage.key?.id : undefined,
                        deviceSource,
                    }
                })

            if (msgError) console.error("Erro ao salvar mensagem:", msgError)
        }

        return new Response(JSON.stringify({ success: true }), {
            headers: { ...corsHeaders, 'Content-Type': 'application/json' },
            status: 200
        })

    } catch (error: any) {
        console.error("Webhook Error:", error.message)
        return new Response(JSON.stringify({ error: error.message }), {
            headers: { ...corsHeaders, 'Content-Type': 'application/json' },
            status: 400
        })
    }
})
