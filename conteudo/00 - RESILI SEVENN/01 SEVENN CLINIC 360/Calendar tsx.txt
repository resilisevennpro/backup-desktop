    title: string;
    start: Date;
    end: Date;
    resourceId?: string;
    status: string;

const LIST_VIEW = 'list';

const CalendarPage = () => {
    const { clinic, user, profile } = useAuth();
    const navigate = useNavigate();
    const { moveLead } = useCrmStore();
    const { setHeaderContent, resetHeader } = useUIStore();

    // Core State
    const [view, setView] = useState<View>(Views.WEEK);
    const [date, setDate] = useState(new Date());
    const [currentTime, setCurrentTime] = useState(new Date());
    const [events, setEvents] = useState<Appointment[]>([]);
    const [listAppointments, setListAppointments] = useState<any[]>([]);
    const [loading, setLoading] = useState(true);
    const [periodSummary, setPeriodSummary] = useState({ total: 0, confirmed: 0, finished: 0, pending: 0 });

    // Modals & Menu State
    const [showActionMenu, setShowActionMenu] = useState(false);
    const [isModalOpen, setIsModalOpen] = useState(false);
    const [isBlockModalOpen, setIsBlockModalOpen] = useState(false);
    const [manageBlockModalOpen, setManageBlockModalOpen] = useState(false);
    const [waitingListOpen, setWaitingListOpen] = useState(false);
    const [isAppointmentDetailsOpen, setIsAppointmentDetailsOpen] = useState(false);

    const [dropConfirmOpen, setDropConfirmOpen] = useState(false);

    // Automation Modal State
    const [finishConfirmationOpen, setFinishConfirmationOpen] = useState(false);
    const [pendingFinishId, setPendingFinishId] = useState<string | null>(null);

    // Selection State
    const [selectedSlot, setSelectedSlot] = useState<any>(null);
    const [selectedAppointment, setSelectedAppointment] = useState<any>(null);
    const [editingAppointment, setEditingAppointment] = useState<any>(null);
    const [selectedBlockEvent, setSelectedBlockEvent] = useState<any>(null);
    const [editingBlock, setEditingBlock] = useState<any>(null);
    const [dropDetails, setDropDetails] = useState<any>(null);
    const [isDeleteConfirmOpen, setIsDeleteConfirmOpen] = useState(false);
    const [eventToDelete, setEventToDelete] = useState<any>(null);
    const [initialModalData, setInitialModalData] = useState<any>(null);

    // State for Filter Data (Real)
    const [professionalsList, setProfessionalsList] = useState<any[]>([]);
    const [proceduresList, setProceduresList] = useState<any[]>([]);

    // Filter Selection State
    const [selectedProfessionalIds, setSelectedProfessionalIds] = useState<string[]>([]);
    const [selectedProcedureIds, setSelectedProcedureIds] = useState<string[]>([]);
    const [showFilterMenu, setShowFilterMenu] = useState(false);

    const location = useLocation();

    // Initial Fetch Setup & Location Sate
    useEffect(() => {
        if (clinic?.id) {
            fetchAppointments();
            fetchFiltersData();
        } else {
            setLoading(false);
        }

        const state = location.state as any;

        // LEGACY: Old navigation method (keeping for potential backward compat)
        if (state && state.openAddModal) {
            const { leadId, leadName, professionalId, procedureName, procedureId } = state;
            setInitialModalData({
                leadId,
                title: leadName,
                professionalId,
                procedureName,
                procedureId,
                start: new Date()
            });
            setIsModalOpen(true);
            window.history.replaceState({}, document.title);
        }

        // NEW: Action-based navigation from Lead Details
        if (state && state.action === 'new_appointment' && state.leadId) {
            console.log("🚀 CRM Navigation: New Appointment for Lead", state.leadId);

            setInitialModalData({
                leadId: state.leadId
            });

            setIsModalOpen(true);

            // Clean state to prevent reopening on refresh
            navigate(location.pathname, { replace: true, state: {} });
        }
    }, [clinic?.id, location.state, navigate]);

    // Effect to open appointment details if requested
    useEffect(() => {
        if (location.state && (location.state as any).openAppointmentId && events.length > 0) {
            const { openAppointmentId } = location.state as any;
            const targetEvent = events.find(e => e.id === openAppointmentId);
            if (targetEvent) {
                setSelectedAppointment(targetEvent);
                setIsAppointmentDetailsOpen(true);
                // Clean state to avoid reopening on re-renders
                window.history.replaceState({}, document.title);
            }
        }
    }, [events, location.state]);

    const fetchFiltersData = async () => {
        if (!clinic?.id) {
            console.error('❌ fetchFiltersData: clinic.id is undefined');
            return;
        }

        console.log('🔄 Carregando filtros para clinic_id:', clinic.id);

        // Fetch Professionals
        const { data: pros, error: prosError } = await supabase
            .from('professionals')
            .select('*')
            .eq('clinic_id', clinic.id);

        if (prosError) {
            console.error('❌ Erro ao carregar profissionais:', prosError);
        } else {
            console.log('✅ Profissionais carregados:', pros?.length || 0);
            if (pros) setProfessionalsList(pros);
        }

        // Fetch Services (Procedures)
        const { data: servs, error: servsError } = await supabase
            .from('services')
            .select('*')
            .eq('clinic_id', clinic.id);

        if (servsError) {
            console.error('❌ Erro ao carregar serviços:', servsError);
        } else {
            console.log('✅ Serviços carregados:', servs?.length || 0);
            if (servs) setProceduresList(servs);
        }
    };

    const updateAppointmentStatus = async (id: string, newStatus: string) => {
        // Intercept "Concluído" for Automation
        if (newStatus === 'Concluído') {
            setPendingFinishId(id);
            setFinishConfirmationOpen(true);
            return;
        }

        await executeStatusUpdate(id, newStatus);
    };

    const executeStatusUpdate = async (id: string, newStatus: string, quiet = false) => {
        // Enforce Allowed Values (Sanity Check)
        const allowedStatuses = ['Pendente', 'Confirmado', 'Chegou', 'Em Atendimento', 'Concluído', 'Cancelado', 'Faltou'];
        if (!allowedStatuses.includes(newStatus)) {
            const map: any = { 'CONFIRMADO': 'Confirmado', 'PENDENTE': 'Pendente', 'CHEGOU': 'Chegou' };
            if (map[newStatus]) newStatus = map[newStatus];
        }

        try {
            // Optimistic Update
            setEvents(prev => prev.map(ev => ev.id === id ? { ...ev, status: newStatus } : ev));
            setListAppointments(prev => prev.map(app => app.id === id ? { ...app, status: newStatus } : app));
            if (selectedAppointment && selectedAppointment.id === id) {
                setSelectedAppointment({ ...selectedAppointment, status: newStatus });
            }

            const { error } = await supabase
                .from('appointments')
                .update({ status: newStatus })
                .eq('id', id);

            if (error) {
                console.error("Error updating status:", error);
                // Revert on error
                fetchAppointments();
                toast.error(`Erro ao atualizar: ${error.message}`);
            } else if (!quiet) {
                toast.success("Status Atualizado!");
            }
        } catch (err) {
            console.error(err);
            toast.error("Erro inesperado ao atualizar status.");
        }
    };

    const handleConfirmAutomation = async (type: 'automation' | 'simple') => {
        if (!pendingFinishId) return;

        if (type === 'simple') {
            await executeStatusUpdate(pendingFinishId, 'Concluído');
            setFinishConfirmationOpen(false);
            setPendingFinishId(null);
            return;
        }

        // Full Automation
        const event = events.find(e => e.id === pendingFinishId);
        if (!event) {
            toast.error("Agendamento não encontrado.");
            return;
        }

        try {
            toast.loading("Processando automação...");

            // 1. CRM - Move para Venda Realizada e atualiza Stage (moveLead robusto agora localiza o Stage ID)
            if (event.leadId) {
                await moveLead(event.leadId, 'Venda Realizada', { quiet: true });
            }

            // 2. Financeiro (Se ainda não criado pelo moveLead, mas o moveLead agora já cria se houver valor)
            // Para evitar duplicidade, vamos deixar o moveLead cuidar do financeiro se houver estimated_value.
            // Aqui na Agenda, se tivermos um preço específico do agendamento que difere do estimado, podemos manter.
            // Mas o prompt focou na transição do Lead.

            // 3. Update Status do Agendamento
            await executeStatusUpdate(pendingFinishId, 'Concluído', true);

            toast.dismiss();
            toast.success(`Venda registrada por ${user?.user_metadata?.name || user?.email || 'Usuário'}!`);

        } catch (error) {
            console.error(error);
            toast.error("Erro na automação.");
        } finally {
            setFinishConfirmationOpen(false);
            setPendingFinishId(null);
        }
    };

    const fetchAppointments = async () => {
        if (!clinic?.id) {
            console.error('❌ fetchAppointments: clinic.id is undefined');
            setLoading(false);
            return;
        }

        try {
            setLoading(true);
            console.log('🔄 Carregando agendamentos para clinic_id:', clinic.id);

            const { data, error } = await supabase
                .from('appointments')
                .select(`
                    id,
                    start_time,
                    end_time,
                    status,
                    notes,
                    lead_id,
                    professional_id,
                    procedure_id,
                    leads!lead_id (id, name, procedure, budget, total_ltv, status),
                    professionals!professional_id (id, name, color),
                    services!procedure_id (*)
                `)
                .eq('clinic_id', clinic.id);

            if (error) throw error;

            console.log('✅ Conexão com Procedure_ID estabelecida com sucesso');

            let formattedEvents: Appointment[] = [];

            if (data) {
                formattedEvents = data.map(app => {
                    // Tratamento robusto de nulos para relacionamentos
                    const professional = Array.isArray(app.professionals) ? app.professionals[0] : app.professionals;
                    const lead = Array.isArray(app.leads) ? app.leads[0] : app.leads;
                    const service = Array.isArray(app.services) ? app.services[0] : app.services;

                    return {
                        id: app.id,
                        title: `${lead?.name || 'Paciente'}`,
                        leadId: lead?.id || app.lead_id,
                        leadStatus: lead?.status || 'Novo',
                        start: new Date(app.start_time),
                        end: new Date(app.end_time),
                        status: app.status,
                        professionalId: professional?.id || app.professional_id,
                        professionalName: professional?.name || 'Profissional',
                        procedureId: app.procedure_id,
                        procedureName: service?.name || (lead as any)?.procedure || '',
                        procedurePrice: service?.base_price || 0,
                        leadBudget: lead?.budget || 0,
                        leadTotalLtv: lead?.total_ltv || 0,
                        color: professional?.color || '#3b82f6',
                        notes: app.notes || ''
                    };
                });
            }

            // Sync List View
            const listData = formattedEvents.map(ev => ({
                id: ev.id,
                time: format(ev.start, 'HH:mm'),
                patient: ev.title,
                procedure: (ev as any).procedureName || 'Consulta',
                professional: (ev as any).professionalName || 'Profissional',
                status: ev.status,
                date: ev.start,
                color: ev.color,
                leadId: ev.leadId,
                leadStatus: (ev as any).leadStatus
            }));
            setListAppointments(listData);

            setEvents(formattedEvents);
            console.log('✅ Agendamentos carregados com sucesso:', formattedEvents.length);
            console.log('📋 Agendamentos formatados:', formattedEvents);
        } catch (error) {
            console.error("❌ Erro ao buscar agendamentos reais:", error);
            toast.error("Erro ao carregar agendamentos");
        } finally {
            setLoading(false);
        }
    };

    const handleNavigate = (newDate: Date) => setDate(newDate);
    const handleViewChange = (newView: View) => setView(newView);

    // =============================================================================
    // NAVIGATION HANDLERS (useCallback for stable refs)
    // =============================================================================
    const handlePrevPeriod = useCallback(() => {
        setDate(prev => {
            const newDate = new Date(prev);
            if (view === Views.MONTH) {
                newDate.setMonth(prev.getMonth() - 1);
            } else if (view === Views.WEEK) {
                newDate.setDate(prev.getDate() - 7);
            } else {
                newDate.setDate(prev.getDate() - 1);
            }
            return newDate;
        });
    }, [view]);

    const handleNextPeriod = useCallback(() => {
        setDate(prev => {
            const newDate = new Date(prev);
            if (view === Views.MONTH) {
                newDate.setMonth(prev.getMonth() + 1);
            } else if (view === Views.WEEK) {
                newDate.setDate(prev.getDate() + 7);
            } else {
                newDate.setDate(prev.getDate() + 1);
            }
            return newDate;
        });
    }, [view]);

    const handleGoToToday = useCallback(() => {
        setDate(new Date());
    }, []);

    const toggleFilterMenu = useCallback(() => {
        setShowFilterMenu(prev => !prev);
    }, []);

    const handleNewAppointment = useCallback(() => {
        setShowActionMenu(prev => !prev);
    }, []);

    const eventStyleGetter = (event: Appointment) => {
        const style: any = {
            backgroundColor: event.color || '#3b82f6',
            borderRadius: '4px',
            opacity: 1,
            color: 'white',
            border: '0px',
            display: 'block'
        };

        if (event.status === 'Cancelado') {
            style.backgroundColor = '#fee2e2';
            style.color = '#b91c1c';
            style.textDecoration = 'line-through';
            style.opacity = 0.7;
        }

        return { style };
    };

    // Filter Logic
    const filteredEvents = useMemo(() => {
        return events.filter(e => {
            const matchesProfessional = selectedProfessionalIds.length === 0 || selectedProfessionalIds.includes((e as any).professionalId);
            // Match procedure likely by name if we don't have ID in event, or match by ID if added later. 
            // Current map uses Service Name. Let's filter by name assuming lists are synced or IDs if needed.
            // Simplified: Filter by ID if we had it, but for now lets try to match via name or update fetch to include service_id
            // Better: Let's assume selection uses ID, but check against event.procedureName?? No, let's just filter by text if IDs missing or update fetch.
            // UPDATE: fetchAppointments doesn't put service_id in event top level. Ideally should.
            // For now, let's allow all procedures if logic is complex, OR fix the matching.
            // Fix: let's match by procedure NAME for now since `proceduresList` has names.

            const matchesProcedure = selectedProcedureIds.length === 0 ||
                selectedProcedureIds.some(id => {
                    const proc = proceduresList.find(p => p.id === id);
                    return proc && proc.name === (e as any).procedureName;
                });

            return matchesProfessional && matchesProcedure;
        });
    }, [events, selectedProfessionalIds, selectedProcedureIds, proceduresList]);

    // Update List View Filter
    const filteredListEvents = useMemo(() => {
        if (!listAppointments) return [];
        return listAppointments.filter(e => {
            const matchesDate = !e.date || isSameDay(new Date(e.date), date);
            const matchesProfessional = selectedProfessionalIds.length === 0 || selectedProfessionalIds.some(id => {
                const p = professionalsList.find(p => p.id === id);
                return p && p.name === e.professional;
            });
            const matchesProcedure = selectedProcedureIds.length === 0 || selectedProcedureIds.some(id => {
                const proc = proceduresList.find(p => p.id === id);
                return proc && proc.name === e.procedure;
            });
            const isNotBlocked = e.status !== 'blocked';
            return matchesProfessional && matchesProcedure && matchesDate && isNotBlocked;
        });
    }, [listAppointments, selectedProfessionalIds, selectedProcedureIds, professionalsList, proceduresList, date]);

    // Calculate Dynamic Period Summary
    useEffect(() => {
        let start = startOfDay(date);
        let end = endOfDay(date);

        if (view === Views.WEEK) {
            start = startOfWeek(date, { locale: ptBR });
            end = endOfDay(addDays(start, 6));
        } else if (view === Views.MONTH) {
            start = startOfMonth(date); // Need to import startOfMonth
            end = endOfMonth(date);     // Need to import endOfMonth
        }

        const eventsInView = filteredEvents.filter(ev =>
            isWithinInterval(ev.start, { start, end }) || isWithinInterval(ev.end, { start, end })
        );

        const summary = {
            total: eventsInView.length,
            confirmed: eventsInView.filter(e => e.status === 'Confirmado').length,
            finished: eventsInView.filter(e => e.status === 'Concluído').length,
            pending: eventsInView.filter(e => e.status === 'Pendente').length
        };
        setPeriodSummary(summary);

    }, [filteredEvents, view, date]);

    // =============================================================================
    // DYNAMIC HEADER EFFECT (with setTimeout to break sync cycle)
    // =============================================================================
    useEffect(() => {
        if (!clinic?.id || !profile?.id) return;

        const currentMonthYear = format(date, "MMMM yyyy", { locale: ptBR });

        const timeoutId = setTimeout(() => {
            setHeaderContent(
                <div className="flex items-center justify-between w-full h-full relative">
                    {/* Bloco 1: Esquerda - Apenas Título + Divisor */}
                    <div className="flex items-center gap-6 pl-4">
                        {/* Identidade da Página */}

                    <div className="absolute left-1/2 -translate-x-1/2 hidden lg:flex items-center gap-3 bg-slate-50/50 border border-slate-200/60 px-4 py-1.5 rounded-full shadow-sm backdrop-blur-sm">
                        {/* Pacientes */}

                        <span className="w-1 h-1 rounded-full bg-slate-300" />
                        {/* Confirmados */}
                        <div className="flex items-center gap-1.5 text-[12px] tracking-[0.1em] uppercase">
                            <span className="font-extrabold text-indigo-600" style={{ fontFamily: 'Manrope, sans-serif' }}>{periodSummary.confirmed}</span>
                            <span className="font-semibold text-slate-500">CONFIRMADO{periodSummary.confirmed !== 1 ? 'S' : ''}</span>
                        </div>
                        <span className="w-1 h-1 rounded-full bg-slate-300" />
                        {/* Concluídos - SEMPRE VISÍVEL */}
                        <div className="flex items-center gap-1.5 text-[12px] tracking-[0.1em] uppercase">
                            <span className="font-extrabold text-emerald-600" style={{ fontFamily: 'Manrope, sans-serif' }}>{periodSummary.finished}</span>
                            <span className="font-semibold text-slate-500">CONCLUÍDO{periodSummary.finished !== 1 ? 'S' : ''}</span>
                        </div>
                        {periodSummary.pending > 0 && (
                            <>
                                <span className="w-1 h-1 rounded-full bg-slate-300" />
                                {/* Pendentes */}
                                <div className="flex items-center gap-1.5 text-[12px] tracking-[0.1em] uppercase">
                                    <span className="font-extrabold text-amber-500" style={{ fontFamily: 'Manrope, sans-serif' }}>{periodSummary.pending}</span>
                                    <span className="font-semibold text-slate-500">PENDENTE{periodSummary.pending !== 1 ? 'S' : ''}</span>
                                </div>
                            </>
                        )}
                    </div>

                    {/* Bloco 3: Direita - Filtrar + Novo Agendamento */}
                    <div className="flex items-center gap-2 pr-2">
                        <Button
                            variant="outline"
                            size="sm"
                            className={cn(
                                "h-9 px-3 text-xs font-semibold gap-2 transition-all shadow-sm",
                                selectedProfessionalIds.length > 0
                                    ? 'bg-slate-800 text-white border-slate-800 hover:bg-slate-900 hover:text-white'
                                    : 'bg-white text-slate-700 border-slate-200 hover:bg-slate-50'
                            )}
                            onClick={toggleFilterMenu}
                        >
                            <Filter className="h-3.5 w-3.5" />
                            Filtrar
                        </Button>
                        <Button
                            variant="default"
                            size="sm"
                            className="gap-2 h-9 px-4 text-xs font-bold shadow-md bg-emerald-600 hover:bg-emerald-700 text-white"
                            onClick={handleNewAppointment}
                        >
                            <CalendarPlus className="h-3.5 w-3.5" />
                            Novo Agendamento
                        </Button>
                    </div>
                </div>

        }, 0);

        return () => {
            clearTimeout(timeoutId);
            resetHeader();
        };
    }, [
        clinic?.id,
        profile?.id,
        periodSummary.total,
        periodSummary.confirmed,
        periodSummary.finished,
        periodSummary.pending,
        selectedProfessionalIds.length
    ]); // Only primitive dependencies


    // Custom Event Component
    const EventComponent = ({ event }: any) => {
        const isBlocked = event.status === 'blocked';
        const isCancelled = event.status === 'Cancelado';
        const isHighValue = (event.procedurePrice || 0) > 3000;

        if (isBlocked) {
            const displayProfessionals = event.groupedProfessionals || [
                { id: event.professionalId, name: event.professionalName, color: event.color }
            ];
            return (
                <div
                    className="flex flex-col h-full w-full overflow-hidden p-1 leading-none justify-start pt-1.5 hover:brightness-90 transition-all cursor-pointer relative group"
                    onClick={(e) => {
                        e.stopPropagation();
                        setSelectedBlockEvent(event);
                        setManageBlockModalOpen(true);
                    }}
                >
                    <div className="font-bold font-manrope text-[10px] truncate flex items-center gap-1 z-10 mb-1.5">
                        <Lock className="w-3 h-3 text-slate-500 inline flex-shrink-0" />
                        <span className="truncate">{event.title}</span>
                    </div>
                    <div className="flex flex-wrap gap-1 z-10 w-full">
                        {displayProfessionals.map((prof: any) => (
                            <div
                                key={prof.id}
                                className="text-[9px] font-bold bg-white/95 px-2 py-1 rounded shadow-sm border border-slate-200/50 truncate"
                                style={{ color: prof.color || '#475569' }}
                            >
                                {prof.name}
                            </div>
                        ))}
                    </div>
                    <div className="absolute inset-0 border-2 border-transparent group-hover:border-slate-400/30 rounded transition-colors pointer-events-none" />

                {/* Delete Confirmation Modal (Sophisticated) */}
                <Dialog open={isDeleteConfirmOpen} onOpenChange={setIsDeleteConfirmOpen}>
                    <DialogContent className="sm:max-w-[400px] p-0 overflow-hidden border-0 shadow-2xl rounded-2xl font-manrope">
                        <div className="bg-white">
                            <div className="h-1.5 bg-rose-500 w-full" />
                            <div className="p-6">
                                <div className="flex flex-col items-center text-center space-y-4">
                                    <div className="w-16 h-16 bg-rose-50 rounded-full flex items-center justify-center mb-2">
                                        <AlertTriangle className="w-8 h-8 text-rose-500" />
                                    </div>
                                    <div className="space-y-2">
                                        <DialogTitle className="text-xl font-bold text-slate-900">Confirmar Exclusão</DialogTitle>
                                        <p className="text-sm text-slate-500 leading-relaxed px-4">
                                            Tem certeza que deseja excluir o agendamento de <span className="font-bold text-slate-700">"{eventToDelete?.title}"</span>? Esta ação não poderá ser desfeita.
                                        </p>
                                    </div>
                                </div>

                                <div className="flex flex-col gap-3 mt-8">
                                    <Button
                                        className="h-12 bg-rose-600 hover:bg-rose-700 text-white font-bold flex items-center justify-center gap-2 rounded-xl shadow-lg shadow-rose-200"
                                        onClick={async () => {
                                            if (eventToDelete) {
                                                await handleDeleteEvent(eventToDelete.id);
                                                setIsDeleteConfirmOpen(false);
                                                setEventToDelete(null);
                                                if (selectedAppointment?.id === eventToDelete.id) setSelectedAppointment(null);
                                                if (selectedBlockEvent?.id === eventToDelete.id) setSelectedBlockEvent(null);
                                            }
                                        }}
                                    >
                                        <Trash2 className="w-4 h-4" />
                                        Sim, Excluir Agora
                                    </Button>
                                    <Button
                                        variant="ghost"
                                        className="h-12 text-slate-500 hover:text-slate-700 font-bold rounded-xl"
                                        onClick={() => {
                                            setIsDeleteConfirmOpen(false);
                                            setEventToDelete(null);
                                        }}
                                    >
                                        Cancelar
                                    </Button>
                                </div>
                            </div>
                        </div>
                    </DialogContent>
                </Dialog>
            </TooltipProvider >
        </div >
    );
}

export default CalendarPage;
