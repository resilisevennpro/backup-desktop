import { useState, useEffect, useRef } from "react";
import { FileText, Plus, Search, Download, Trash2, Eye, Loader2, Save, Printer, Bold, Italic, List, ListOrdered, PenTool, Eraser, Check, X, Share2, MessageCircle } from "lucide-react";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Card, CardContent } from "@/components/ui/card";
import { Badge } from "@/components/ui/badge";
import {
    Dialog,
    DialogContent,
    DialogHeader,
    DialogTitle,
    DialogFooter,
    DialogDescription,
    DialogClose,
} from "@/components/ui/dialog";
import {
    Select,
    SelectContent,
    SelectItem,
    SelectTrigger,
    SelectValue,
} from "@/components/ui/select";
import { supabase } from "@/lib/supabase";
import { useAuth } from "@/contexts/AuthContext";
import { toast } from "sonner";
import { cn } from "@/lib/utils";

interface ClinicDocument {
    id: string;
    title: string;
    created_at: string;
    status: 'signed' | 'pending' | 'draft' | 'awaiting_signature';
    template_id?: string;
    content?: string;
    signature_url?: string;
    signed_at?: string;
    public_token?: string;
    expires_at?: string;
}

interface DocumentTemplate {
    id: string;
    title: string;
    content: string;
}

interface DocumentsTabProps {
    leadId?: string;
    clinicId?: string;
    leadName?: string;
}

const VARIABLES = [
    { label: "Nome do Paciente", value: "[Nome do Paciente]" },
    { label: "CPF", value: "[CPF]" },
    { label: "Data", value: "[Data]" },
    { label: "Profissional", value: "[Profissional]" },
    { label: "Clínica", value: "[Nome da Clínica]" }
];

export function DocumentsTab({ leadId }: DocumentsTabProps) {
    const { clinic, profile } = useAuth();
    const [documents, setDocuments] = useState<ClinicDocument[]>([]);
    const [templates, setTemplates] = useState<DocumentTemplate[]>([]);
    const [loading, setLoading] = useState(true);

    // Create State
    const [isNewDocOpen, setIsNewDocOpen] = useState(false);
    const [selectedTemplateId, setSelectedTemplateId] = useState<string>('');
    const [creating, setCreating] = useState(false);

    // Delete State
    const [documentToDelete, setDocumentToDelete] = useState<string | null>(null);

    // Editor/View State
    const [isEditorOpen, setIsEditorOpen] = useState(false);
    const [currentDoc, setCurrentDoc] = useState<ClinicDocument | null>(null);
    const [editorContent, setEditorContent] = useState('');
    const [lead, setLead] = useState<any>(null);

    // Signature State
    const [isSigning, setIsSigning] = useState(false);
    const canvasRef = useRef<HTMLCanvasElement>(null);
    const [isDrawing, setIsDrawing] = useState(false);

    const editorRef = useRef<HTMLDivElement>(null);

    // Fetch Lead Data
    useEffect(() => {
        if (leadId) {
            supabase.from('leads').select('*').eq('id', leadId).single().then(({ data }) => {
                if (data) setLead(data);
            });
        }
    }, [leadId]);

    // Fetch Documents & Templates
    useEffect(() => {
        if (!leadId || !clinic?.id) return;

        const fetchData = async () => {
            setLoading(true);
            try {
                // 1. Fetch Documents
                const { data: docsData, error: docsError } = await supabase
                    .from('clinic_documents')
                    .select('*')
                    .eq('lead_id', leadId)
                    .order('created_at', { ascending: false });

                if (docsError) throw docsError;
                setDocuments(docsData || []);

                // 2. Fetch Templates
                const { data: tmplData, error: tmplError } = await supabase
                    .from('document_templates')
                    .select('id, title, content')
                    .eq('clinic_id', clinic.id);

                if (tmplError) throw tmplError;
                setTemplates(tmplData || []);

            } catch (error) {
                console.error("Erro ao carregar dados:", error);
                toast.error("Erro ao carregar documentos.");
            } finally {
                setLoading(false);
            }
        };

        fetchData();
    }, [leadId, clinic?.id]);

    // --- Toolbar Actions ---
    const execCmd = (command: string, value: string | undefined = undefined) => {
        document.execCommand(command, false, value);
        editorRef.current?.focus();
    };

    const insertVariable = (variable: string) => {
        if (!editorRef.current) return;
        editorRef.current.focus();
        document.execCommand("insertText", false, variable);
    };

    const handleCreateDocument = async () => {
        if (!selectedTemplateId || !leadId || !clinic?.id) return;

        const template = templates.find(t => t.id === selectedTemplateId);
        if (!template) return;

        setCreating(true);
        try {
            const { data, error } = await supabase
                .from('clinic_documents')
                .insert([
                    {
                        clinic_id: clinic.id,
                        lead_id: leadId,
                        title: template.title,
                        template_id: template.id,
                        content: template.content,
                        status: 'pending'
                    }
                ])
                .select()
                .single();

            if (error) throw error;

            setDocuments(prev => [data, ...prev]);
            toast.success("Documento criado com sucesso!");
            setIsNewDocOpen(false);
            setSelectedTemplateId('');
        } catch (error) {
            console.error("Erro ao criar documento:", error);
            toast.error("Erro ao criar documento.");
        } finally {
            setCreating(false);
        }
    };

    const confirmDeleteDocument = async () => {
        if (!documentToDelete) return;

        try {
            const { error } = await supabase
                .from('clinic_documents')
                .delete()
                .eq('id', documentToDelete);

            if (error) throw error;

            setDocuments(prev => prev.filter(doc => doc.id !== documentToDelete));
            toast.success("Documento removido.");
            setDocumentToDelete(null);
        } catch (error) {
            console.error("Erro ao deletar documento:", error);
            toast.error("Erro ao remover documento.");
        }
    };

    // Editor & Parsing Logic
    const parseContent = (content: string) => {
        if (!content) return '';
        let parsed = content;

        const today = new Date().toLocaleDateString('pt-BR');
        const professionalName = profile?.full_name || "Equipe Sevenn";
        const clinicName = clinic?.name || "Clínica";
        const patientName = lead?.name || "Paciente";
        const patientCpf = lead?.cpf || "CPF não informado";
        const patientPhone = lead?.mobile || lead?.phone || "";

        parsed = parsed.replace(/\[Nome do Paciente\]/g, patientName);
        parsed = parsed.replace(/\[CPF\]/g, patientCpf);
        parsed = parsed.replace(/\[Telefone\]/g, patientPhone);
        parsed = parsed.replace(/\[Data\]/g, today);
        parsed = parsed.replace(/\[Profissional\]/g, professionalName);
        parsed = parsed.replace(/\[Nome da Clínica\]/g, clinicName);

        return parsed;
    };

    const handleOpenEditor = async (doc: ClinicDocument) => {
        let contentToUse = doc.content || '';

        // Fallback: If doc has no content but has a template, try to fetch template content
        if (!contentToUse && doc.template_id) {
            const tmpl = templates.find(t => t.id === doc.template_id);
            if (tmpl) contentToUse = tmpl.content;
        }

        const parsedContent = parseContent(contentToUse);
        setEditorContent(parsedContent);
        setCurrentDoc(doc);
        setIsEditorOpen(true);
        // Reset signing state when opening
        setIsSigning(false);

        // Timeout to set innerHTML after render
        setTimeout(() => {
            if (editorRef.current) {
                editorRef.current.innerHTML = parsedContent;
            }
        }, 100);
    };

    const handleSaveContent = async () => {
        if (!currentDoc || !editorRef.current) return;
        const newContent = editorRef.current.innerHTML;

        try {
            const { error } = await supabase
                .from('clinic_documents')
                .update({ content: newContent })
                .eq('id', currentDoc.id);

            if (error) throw error;

            setDocuments(prev => prev.map(d => d.id === currentDoc.id ? { ...d, content: newContent } : d));
            toast.success("Documento atualizado com sucesso!");
        } catch (error) {
            console.error("Erro ao salvar documento:", error);
            toast.error("Erro ao salvar alterações.");
        }
    };

    const handlePrint = (doc?: ClinicDocument) => {
        if (doc) {
            handleOpenEditor(doc);
            setTimeout(() => window.print(), 500);
        } else {
            window.print();
        }
    };

    const handleSendLink = async (doc: ClinicDocument) => {
        try {
            let token = doc.public_token;
            let expiration = doc.expires_at;
            let needsUpdate = false;

            // Generate token if not exists
            if (!token) {
                token = crypto.randomUUID();
                // Expire in 7 days
                const date = new Date();
                date.setDate(date.getDate() + 7);
                expiration = date.toISOString();
                needsUpdate = true;
            }

            if (needsUpdate) {
                const { error } = await supabase
                    .from('clinic_documents')
                    .update({
                        public_token: token,
                        expires_at: expiration,
                        status: doc.status === 'draft' ? 'pending' : doc.status
                    })
                    .eq('id', doc.id);

                if (error) throw error;

                // Update local state
                setDocuments(prev => prev.map(d =>
                    d.id === doc.id ? { ...d, public_token: token, expires_at: expiration, status: 'awaiting_signature' as const } : d
                ));
                if (currentDoc?.id === doc.id) {
                    setCurrentDoc(prev => prev ? { ...prev, public_token: token, expires_at: expiration, status: 'awaiting_signature' as const } : null);
                }
            } else {
                if (doc.status !== 'signed') {
                    // Optimistically update status to awaiting_signature if not signed
                    await supabase.from('clinic_documents').update({ status: 'awaiting_signature' }).eq('id', doc.id);
                    setDocuments(prev => prev.map(d => d.id === doc.id ? { ...d, status: 'awaiting_signature' as const } : d));
                }
            }

            // Re-read token just in case
            const finalToken = token || doc.public_token;

            const link = `${window.location.origin}/assinatura/${finalToken}`;
            await navigator.clipboard.writeText(link);

            const expDate = new Date(expiration || Date.now()).toLocaleDateString('pt-BR');
            const message = `Olá ${lead?.name || ''}! Aqui está o link seguro para assinar seu documento: ${link} (Válido até ${expDate})`;

            // Open WhatsApp
            let phone = lead?.mobile || lead?.phone || '';
            phone = phone.replace(/\D/g, ''); // Remove formatting
            if (phone && !phone.startsWith('55')) phone = `55${phone}`; // Add Brazil code if missing

            const whatsappUrl = `https://web.whatsapp.com/send?phone=${phone}&text=${encodeURIComponent(message)}`;
            window.open(whatsappUrl, '_blank');

            toast.success("Link copiado e WhatsApp aberto!");

        } catch (error) {
            console.error("Erro ao gerar link:", error);
            toast.error("Erro ao gerar link de assinatura.");
        }
    };

    // --- Signature Logic ---
    const startSigning = () => {
        setIsSigning(true);
        // We need to wait for canvas to be rendered to set context
        setTimeout(() => {
            const canvas = canvasRef.current;
            if (canvas) {
                // Resize for high DPI
                const ratio = Math.max(window.devicePixelRatio || 1, 1);
                canvas.width = canvas.offsetWidth * ratio;
                canvas.height = canvas.offsetHeight * ratio;
                const ctx = canvas.getContext('2d');
                if (ctx) ctx.scale(ratio, ratio);
            }
        }, 100);
    };

    const getTouchPos = (canvasDom: HTMLCanvasElement, touchEvent: React.TouchEvent) => {
        const rect = canvasDom.getBoundingClientRect();
        return {
            x: touchEvent.touches[0].clientX - rect.left,
            y: touchEvent.touches[0].clientY - rect.top
        };
    };

    const getMousePos = (canvasDom: HTMLCanvasElement, mouseEvent: React.MouseEvent) => {
        const rect = canvasDom.getBoundingClientRect();
        return {
            x: mouseEvent.clientX - rect.left,
            y: mouseEvent.clientY - rect.top
        };
    };

    const startDraw = (e: React.MouseEvent | React.TouchEvent) => {
        setIsDrawing(true);
        const canvas = canvasRef.current;
        if (!canvas) return;
        const ctx = canvas.getContext('2d');
        if (!ctx) return;

        ctx.strokeStyle = "#000";
        ctx.lineWidth = 2;
        ctx.lineJoin = "round";
        ctx.lineCap = "round";

        const pos = 'touches' in e ? getTouchPos(canvas, e as React.TouchEvent) : getMousePos(canvas, e as React.MouseEvent);
        ctx.beginPath();
        ctx.moveTo(pos.x, pos.y);
    };

    const draw = (e: React.MouseEvent | React.TouchEvent) => {
        if (!isDrawing) return;
        const canvas = canvasRef.current;
        if (!canvas) return;
        const ctx = canvas.getContext('2d');
        if (!ctx) return;

        e.preventDefault(); // Prevent scrolling on touch
        const pos = 'touches' in e ? getTouchPos(canvas, e as React.TouchEvent) : getMousePos(canvas, e as React.MouseEvent);
        ctx.lineTo(pos.x, pos.y);
        ctx.stroke();
    };

    const endDraw = () => {
        setIsDrawing(false);
    };

    const clearSignature = () => {
        const canvas = canvasRef.current;
        if (canvas) {
            const ctx = canvas.getContext('2d');
            ctx?.clearRect(0, 0, canvas.width, canvas.height);
        }
    };

    const confirmSignature = async () => {
        if (!canvasRef.current || !currentDoc) return;

        try {
            const canvas = canvasRef.current;
            // Convert to Blob
            const blob = await new Promise<Blob | null>(resolve => canvas.toBlob(resolve, 'image/png'));
            if (!blob) throw new Error("Falha ao gerar imagem.");

            const fileName = `signatures/${currentDoc.id}_${Date.now()}.png`;

            // Upload
            const { error: uploadError } = await supabase.storage
                .from('lead-images')
                .upload(fileName, blob);

            if (uploadError) throw uploadError;

            // Get Public URL
            const { data: { publicUrl } } = supabase.storage
                .from('lead-images')
                .getPublicUrl(fileName);

            // Update Doc
            const finalContent = editorRef.current?.innerHTML || "";
            const now = new Date().toISOString();

            const { error: dbError } = await supabase
                .from('clinic_documents')
                .update({
                    status: 'signed',
                    signature_url: publicUrl,
                    signed_at: now,
                    content: finalContent
                })
                .eq('id', currentDoc.id);

            if (dbError) throw dbError;

            setDocuments(prev => prev.map(d =>
                d.id === currentDoc.id
                    ? { ...d, status: 'signed', signature_url: publicUrl, signed_at: now, content: finalContent }
                    : d
            ));

            setCurrentDoc(prev => prev ? { ...prev, status: 'signed', signature_url: publicUrl, signed_at: now, content: finalContent } : null);
            setIsSigning(false);
            toast.success("Assinatura coletada com sucesso!");

        } catch (error) {
            console.error("Erro ao salvar assinatura:", error);
            toast.error("Erro ao salvar assinatura.");
        }
    };

    return (
        <div className="flex flex-col h-full bg-slate-50/50 p-6 space-y-6">
            <style>
                {`
                @media print {
                    @page { margin: 0; }
                    body { visibility: hidden; background: white; }
                    .print\\:hidden, .print\\:hidden * { display: none !important; }
                    #printable-document {
                        visibility: visible;
                        position: absolute;
                        left: 0;
                        top: 0;
                        width: 100%;
                        margin: 0;
                        padding: 2cm;
                        box-shadow: none !important;
                        background: white !important;
                    }
                    #printable-document * { visibility: visible; }
                }
                `}
            </style>

            <div className="flex items-center justify-between">
                <div className="flex items-center gap-2">
                    <FileText className="w-5 h-5 text-primary" />
                    <h3 className="font-space text-lg font-bold text-slate-900">Documentos do Paciente</h3>
                </div>
                <div className="flex items-center gap-2">
                    <div className="relative w-64">
                        <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-slate-400" />
                        <Input placeholder="Buscar documentos..." className="pl-9 h-9 text-sm bg-white" />
                    </div>
                    <Button className="h-9 gap-2" onClick={() => setIsNewDocOpen(true)}>
                        <Plus className="w-4 h-4" /> Novo Documento
                    </Button>
                </div>
            </div>

            {loading ? (
                <div className="flex items-center justify-center py-20">
                    <Loader2 className="w-8 h-8 animate-spin text-primary" />
                </div>
            ) : (
                <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
                    {documents.map((doc) => (
                        <Card key={doc.id} className="group hover:shadow-md transition-all border-slate-200">
                            <CardContent className="p-4 space-y-4">
                                <div className="flex items-start justify-between">
                                    <div className="p-2 bg-primary/5 rounded-lg">
                                        <FileText className="w-6 h-6 text-primary" />
                                    </div>
                                    <Badge variant={doc.status === 'signed' ? 'default' : doc.status === 'awaiting_signature' ? 'outline' : 'secondary'} className={
                                        doc.status === 'signed'
                                            ? 'bg-emerald-100 text-emerald-700 hover:bg-emerald-100'
                                            : doc.status === 'awaiting_signature'
                                                ? 'bg-blue-50 text-blue-700 border-blue-200 hover:bg-blue-100'
                                                : 'bg-amber-100 text-amber-700 hover:bg-amber-100'
                                    }>
                                        {doc.status === 'signed' ? 'Assinado' : doc.status === 'awaiting_signature' ? 'Aguardando' : 'Pendente'}
                                    </Badge>
                                </div>

                                <div>
                                    <h4 className="font-manrope font-bold text-slate-900 truncate" title={doc.title}>
                                        {doc.title}
                                    </h4>
                                    <p className="text-xs text-slate-500 font-medium mt-1">
                                        {new Date(doc.created_at).toLocaleDateString()}
                                    </p>
                                </div>

                                <div className="flex items-center gap-2 pt-2">
                                    <Button
                                        variant="outline"
                                        size="sm"
                                        className="flex-1 h-8 gap-1.5 text-xs text-slate-600"
                                        onClick={() => handleOpenEditor(doc)}
                                    >
                                        <Eye className="w-3.5 h-3.5" /> Ver / Editar
                                    </Button>
                                    <Button
                                        variant="outline"
                                        size="sm"
                                        className="h-8 w-8 p-0 text-slate-400 hover:text-primary"
                                        onClick={() => handleSendLink(doc)}
                                        title="Enviar para Assinatura"
                                    >
                                        <Share2 className="w-3.5 h-3.5" />
                                    </Button>
                                    <Button
                                        variant="outline"
                                        size="sm"
                                        className="h-8 w-8 p-0 text-slate-400 hover:text-primary"
                                        onClick={() => handlePrint(doc)}
                                        title="Baixar Documento"
                                    >
                                        <Download className="w-3.5 h-3.5" />
                                    </Button>
                                    <Button
                                        variant="outline"
                                        size="sm"
                                        className="h-8 w-8 p-0 text-slate-400 hover:text-destructive"
                                        onClick={() => setDocumentToDelete(doc.id)}
                                    >
                                        <Trash2 className="w-3.5 h-3.5" />
                                    </Button>
                                </div>
                            </CardContent>
                        </Card>
                    ))}
                </div>
            )}

            {!loading && documents.length === 0 && (
                <div className="flex-1 flex flex-col items-center justify-center text-slate-400 space-y-2">
                    <FileText className="w-12 h-12 opacity-20" />
                    <p className="text-sm font-medium">Nenhum documento encontrado.</p>
                </div>
            )}

            {/* Modal Novo Documento */}
            <Dialog open={isNewDocOpen} onOpenChange={setIsNewDocOpen}>
                <DialogContent>
                    <DialogHeader>
                        <DialogTitle>Gerar Novo Documento</DialogTitle>
                        <DialogDescription>
                            Selecione um modelo para criar o documento para este paciente.
                        </DialogDescription>
                    </DialogHeader>

                    <div className="py-4 space-y-4">
                        <div className="space-y-2">
                            <label className="text-sm font-medium leading-none peer-disabled:cursor-not-allowed peer-disabled:opacity-70">
                                Modelo de Documento
                            </label>
                            <Select value={selectedTemplateId} onValueChange={setSelectedTemplateId}>
                                <SelectTrigger>
                                    <SelectValue placeholder="Selecione um template..." />
                                </SelectTrigger>
                                <SelectContent>
                                    {templates.map(t => (
                                        <SelectItem key={t.id} value={t.id}>{t.title}</SelectItem>
                                    ))}
                                </SelectContent>
                            </Select>
                        </div>
                    </div>

                    <DialogFooter>
                        <Button variant="outline" onClick={() => setIsNewDocOpen(false)}>Cancelar</Button>
                        <Button
                            onClick={handleCreateDocument}
                            disabled={!selectedTemplateId || creating}
                            className="bg-primary text-white"
                        >
                            {creating ? <Loader2 className="w-4 h-4 animate-spin mr-2" /> : null}
                            Criar Documento
                        </Button>
                    </DialogFooter>
                </DialogContent>
            </Dialog>

            {/* Modal Delete Confirmation */}
            <Dialog open={!!documentToDelete} onOpenChange={(open) => !open && setDocumentToDelete(null)}>
                <DialogContent>
                    <DialogHeader>
                        <DialogTitle>Excluir Documento</DialogTitle>
                        <DialogDescription>
                            Tem certeza que deseja remover este documento? Esta ação não pode ser desfeita.
                        </DialogDescription>
                    </DialogHeader>
                    <DialogFooter>
                        <Button variant="ghost" onClick={() => setDocumentToDelete(null)}>Cancelar</Button>
                        <Button variant="destructive" onClick={confirmDeleteDocument}>Excluir</Button>
                    </DialogFooter>
                </DialogContent>
            </Dialog>

            {/* Editor Modal (Full Screen with Signature) */}
            <Dialog open={isEditorOpen} onOpenChange={setIsEditorOpen}>
                <DialogContent className="max-w-5xl h-[95vh] flex flex-col p-0 gap-0 overflow-hidden outline-none print:shadow-none print:border-none">
                    {/* Header - Hidden on Print */}
                    <div className="flex shrink-0 items-center justify-between p-4 border-b bg-card print:hidden gap-4">
                        <div className="flex items-center gap-2 overflow-hidden flex-1">
                            <FileText className="w-5 h-5 text-primary shrink-0" />
                            <DialogTitle className="text-lg truncate">{currentDoc?.title}</DialogTitle>
                            {currentDoc?.status === 'signed' ? (
                                <Badge variant="default" className="bg-emerald-100 text-emerald-700 hover:bg-emerald-100 ml-2 shrink-0">
                                    Assinado
                                </Badge>
                            ) : currentDoc?.status === 'awaiting_signature' && (
                                <Badge variant="outline" className="text-blue-600 border-blue-200 ml-2 shrink-0">
                                    Aguardando
                                </Badge>
                            )}
                        </div>
                        <div className="flex items-center gap-2 shrink-0">
                            {!isSigning && (
                                <>
                                    <Button variant="outline" size="sm" onClick={handlePrint} className="gap-2 hidden sm:flex">
                                        <Printer className="w-4 h-4" />
                                    </Button>
                                    <Button onClick={handleSaveContent} size="sm" className="gap-2 bg-primary">
                                        <Save className="w-4 h-4" /> Salvar
                                    </Button>
                                    {/* Intentionally removed manual close button as DialogContent provides one */}
                                </>
                            )}
                        </div>
                    </div>

                    {/* Toolbar - Hidden on Print or Signing */}
                    {!isSigning && (
                        <div className="flex shrink-0 items-center gap-1 p-2 border-b bg-muted/30 overflow-x-auto print:hidden">
                            <Button variant="ghost" size="sm" className="h-8 w-8 p-0" onClick={() => execCmd('bold')}>
                                <Bold className="h-4 w-4" />
                            </Button>
                            <Button variant="ghost" size="sm" className="h-8 w-8 p-0" onClick={() => execCmd('italic')}>
                                <Italic className="h-4 w-4" />
                            </Button>
                            <div className="w-px h-4 bg-border mx-1" />
                            <Button variant="ghost" size="sm" className="h-8 w-8 p-0" onClick={() => execCmd('insertUnorderedList')}>
                                <List className="h-4 w-4" />
                            </Button>
                            <Button variant="ghost" size="sm" className="h-8 w-8 p-0" onClick={() => execCmd('insertOrderedList')}>
                                <ListOrdered className="h-4 w-4" />
                            </Button>
                            <div className="w-px h-4 bg-border mx-1" />
                            {VARIABLES.map(v => (
                                <Badge
                                    key={v.value}
                                    variant="secondary"
                                    className="cursor-pointer hover:bg-primary/20 hover:text-primary transition-colors font-mono text-[10px] h-6 whitespace-nowrap"
                                    onClick={() => insertVariable(v.value)}
                                >
                                    {v.label}
                                </Badge>
                            ))}
                        </div>
                    )}

                    {/* Content Area */}
                    <div className="flex-1 bg-slate-100 p-4 md:p-8 overflow-y-auto flex justify-center print:p-0 print:bg-white print:overflow-visible">
                        <div
                            id="printable-document"
                            className="w-full md:w-[21cm] min-h-[29.7cm] bg-white shadow-sm md:shadow-lg p-8 md:p-[2.5cm] text-slate-900 print:shadow-none print:p-0"
                        >
                            {/* Document Body */}
                            <div
                                className="prose max-w-none text-justify text-sm leading-relaxed outline-none empty:after:content-['Comece_a_editar...'] empty:after:text-slate-300"
                                ref={editorRef}
                                contentEditable={!isSigning && currentDoc?.status !== 'signed'}
                            />

                            {/* Signature Display (If Signed) */}
                            {currentDoc?.status === 'signed' && currentDoc.signature_url && (
                                <div className="mt-12 border-t pt-4 w-64">
                                    <img src={currentDoc.signature_url} alt="Assinatura" className="h-16 object-contain mb-2" />
                                    <p className="text-xs text-slate-500">Assinado digitalmente em {new Date(currentDoc.signed_at!).toLocaleDateString()} às {new Date(currentDoc.signed_at!).toLocaleTimeString()}</p>
                                </div>
                            )}

                            {/* Signature Pad (Signing Mode) */}
                            {isSigning && (
                                <div className="mt-8 p-4 border-2 border-dashed border-primary/20 rounded-xl bg-slate-50 print:hidden">
                                    <p className="text-sm font-bold text-center mb-2 text-primary">Área de Assinatura</p>
                                    <div className="border bg-white rounded-lg shadow-inner overflow-hidden flex justify-center touch-none">
                                        <canvas
                                            ref={canvasRef}
                                            className="w-full h-40 cursor-crosshair touch-none"
                                            onMouseDown={startDraw}
                                            onMouseMove={draw}
                                            onMouseUp={endDraw}
                                            onMouseLeave={endDraw}
                                            onTouchStart={startDraw}
                                            onTouchMove={draw}
                                            onTouchEnd={endDraw}
                                        />
                                    </div>
                                    <div className="flex justify-center mt-2">
                                        <Button size="sm" variant="ghost" onClick={clearSignature} className="text-xs text-muted-foreground gap-1">
                                            <Eraser className="w-3 h-3" /> Limpar
                                        </Button>
                                    </div>
                                </div>
                            )}
                        </div>
                    </div>

                    {/* Footer Actions - Hidden on Print */}
                    <div className="p-4 border-t bg-card flex justify-between items-center print:hidden">
                        {isSigning ? (
                            <div className="flex items-center gap-2 w-full justify-end">
                                <Button variant="ghost" onClick={() => setIsSigning(false)}>Cancelar</Button>
                                <Button variant="default" onClick={confirmSignature} className="gap-2 bg-emerald-600 hover:bg-emerald-700">
                                    <Check className="w-4 h-4" /> Confirmar Assinatura
                                </Button>
                            </div>
                        ) : (
                            <div className="flex items-center justify-between w-full">
                                <p className="text-xs text-muted-foreground hidden md:block">
                                    {currentDoc?.status === 'signed' ? 'Documento bloqueado para edição.' : 'Edite o documento antes de solicitar assinatura.'}
                                </p>
                                <div className="flex items-center gap-2 ml-auto">

                                    {currentDoc?.status !== 'signed' && (
                                        <>
                                            <Button
                                                variant="outline"
                                                className="gap-2 text-blue-600 border-blue-200 hover:bg-blue-50"
                                                onClick={() => currentDoc && handleSendLink(currentDoc)}
                                            >
                                                <Share2 className="w-4 h-4" /> Enviar p/ Assinar
                                            </Button>
                                            <Button
                                                onClick={startSigning}
                                                className="gap-2"
                                                variant="brand"
                                            >
                                                <PenTool className="w-4 h-4" /> Coletar Assinatura
                                            </Button>
                                        </>
                                    )}
                                </div>
                            </div>
                        )}
                    </div>
                </DialogContent>
            </Dialog>
        </div>
    );
}
