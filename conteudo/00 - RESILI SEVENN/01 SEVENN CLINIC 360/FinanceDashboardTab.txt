import { useState, useEffect, useMemo } from "react";
import { Card, CardContent, CardHeader, CardTitle, CardDescription } from "@/components/ui/card";
import { Button } from "@/components/ui/button";
import { Progress } from "@/components/ui/progress";
import { startOfMonth, endOfMonth, endOfDay } from "date-fns";
import { DateRangePicker } from "@/components/kanban/DateRangePicker";
import { TrendingUp, DollarSign, Users, Activity, ExternalLink } from "lucide-react";
import {
    BarChart,
    Bar,
    XAxis,
    YAxis,
    CartesianGrid,
    Tooltip,
    ResponsiveContainer,
    AreaChart,
    Area,
    PieChart,
    Pie,
    Cell
} from "recharts";

import { useFinanceStore } from "@/store/useFinanceStore";
import { useServicesStore } from "@/store/useServicesStore";
import { useAuth } from "@/contexts/AuthContext";

// --- MOCK DATA ---
const PROJECTION_DATA = [
    { name: 'Fev', values: 2900, guaranteed: 2900 },
    { name: 'Mar', values: 2900, guaranteed: 2900 },
    { name: 'Abr', values: 2000, guaranteed: 1950 },
    { name: 'Mai', values: 900, guaranteed: 880 },
    { name: 'Jun', values: 600, guaranteed: 550 },
    { name: 'Jul', values: 200, guaranteed: 200 },
    { name: 'Ago', values: 200, guaranteed: 200 },
    { name: 'Set', values: 200, guaranteed: 200 },
    { name: 'Out', values: 200, guaranteed: 200 },
    { name: 'Nov', values: 200, guaranteed: 200 },
    { name: 'Dez', values: 200, guaranteed: 200 },
    { name: 'Jan', values: 200, guaranteed: 200 },
];

interface FinanceDashboardTabProps {
    onNavigateToTransactions: () => void;
}

export function FinanceDashboardTab({ onNavigateToTransactions }: FinanceDashboardTabProps) {
    const { clinic } = useAuth();
    const { transactions, metrics, fetchFinanceData, fetchTransactions } = useFinanceStore();
    const { services, fetchServices } = useServicesStore();

    const [salesView, setSalesView] = useState<'daily' | 'monthly'>('daily');
    const [profitabilityView, setProfitabilityView] = useState<'net' | 'gross'>('net');

    // Date Range State
    const [dateRange, setDateRange] = useState<{ from: Date; to: Date }>({
        from: startOfMonth(new Date()),
        to: endOfMonth(new Date()),
    });

    // Carregar dados ao montar ou alterar datas
    useEffect(() => {
        if (clinic?.id && dateRange?.from && dateRange?.to) {
            const end = endOfDay(dateRange.to);
            fetchFinanceData(clinic.id, 'custom', dateRange.from, end);
            fetchServices(clinic.id);
        }
    }, [clinic?.id, dateRange, fetchFinanceData, fetchServices]);

    // Estado derivado para os dados do gráfico de vendas
    const currentSalesData = useMemo(() => {
        if (!transactions || transactions.length === 0) return [];

        const grouped: Record<string, { value: number; date: Date }> = {};

        transactions.forEach(t => {
            if (t.type !== 'income' || t.status !== 'paid') return;

            const date = new Date(t.transaction_date);
            let key = '';

            if (salesView === 'daily') {
                key = date.toLocaleDateString('pt-BR', { day: '2-digit', month: '2-digit' });
            } else {
                const monthNames = ["Jan", "Fev", "Mar", "Abr", "Mai", "Jun", "Jul", "Ago", "Set", "Out", "Nov", "Dez"];
                key = monthNames[date.getMonth()];
            }

            if (!grouped[key]) {
                grouped[key] = { value: 0, date };
            }
            grouped[key].value += Number(t.amount);
        });

        // Ordenação
        return Object.entries(grouped)
            .map(([day, data]) => ({ day, value: data.value, date: data.date }))
            .sort((a, b) => a.date.getTime() - b.date.getTime());

    }, [transactions, salesView]);

    // --- Lógica 1: Top 5 Pacientes ---
    const topPatientsData = useMemo(() => {
        if (!transactions) return [];

        // Filtra receitas pagas
        const income = transactions.filter(t => t.type === 'income' && t.status === 'paid');

        // Agrupa por paciente
        const grouped = income.reduce((acc, curr) => {
            const name = curr.patient || "Desconhecido";
            acc[name] = (acc[name] || 0) + Number(curr.amount);
            return acc;
        }, {} as Record<string, number>);

        // Ordena e pega top 5
        return Object.entries(grouped)
            .map(([name, value]) => ({ name, value }))
            .sort((a, b) => b.value - a.value)
            .slice(0, 5);
    }, [transactions]);

    // --- Lógica 2: Lucratividade Real ---
    const profitabilityData = useMemo(() => {
        if (!transactions) return [];

        const income = transactions.filter(t => t.type === 'income' && t.status === 'paid');

        const officialServices = services.map(s => s.name);

        // Passo B: Agrupar por procedimento (Whitelist)
        const grouped = income.reduce((acc, curr) => {
            let procName = curr.procedure || curr.description || "Outros";
            if (!officialServices.includes(procName)) procName = "Outros";

            if (!acc[procName]) acc[procName] = { gross: 0, count: 0 };

            acc[procName].gross += Number(curr.amount);
            acc[procName].count += 1;
            return acc;
        }, {} as Record<string, { gross: number, count: number }>);

        // Passo C: Calcular custo e líquido
        return Object.entries(grouped).map(([procName, data]) => {
            let totalCost = 0;
            if (procName !== "Outros") {
                const service = services.find(s => s.name === procName);
                totalCost = Number((service as any)?.custo_material || 0) * data.count;
            }

            return {
                name: procName,
                gross: data.gross,
                net: data.gross - totalCost
            };
        }).sort((a, b) => b.gross - a.gross) // Ordena por faturamento bruto
            .slice(0, 8); // Top 8 para não quebrar layout
    }, [transactions, services]);

    // Calcula o valor máximo de faturamento para servir de base 100% para as barras
    const maxGrossValue = useMemo(() => {
        if (profitabilityData.length === 0) return 1;
        return Math.max(...profitabilityData.map(d => d.gross));
    }, [profitabilityData]);

    return (
        <div className="space-y-6 animate-in slide-in-from-bottom-4 duration-500">

            {/* Header com DatePicker */}
            <div className="flex flex-col md:flex-row items-start md:items-center justify-between gap-4 mb-2">
                <div>
                    <h2 className="text-3xl font-bold tracking-tight">Visão Geral</h2>
                    <p className="text-muted-foreground">Acompanhe seus indicadores financeiros.</p>
                </div>
                <div className="flex items-center gap-2 pr-4">
                    <DateRangePicker
                        value={dateRange}
                        onChange={setDateRange}
                        className="w-[120px]"
                    />
                </div>
            </div>

            {/* 0. Meta de Faturamento (New Feature) */}
            <Card className="border-none shadow-md">
                <CardContent className="pt-6 pb-6">
                    <div className="space-y-4">
                        <div className="flex justify-between items-end">
                            <div>
                                <h3 className="text-lg font-bold  tracking-tight mb-1">Meta de Faturamento</h3>
                                <div className="flex items-baseline gap-2">
                                    <span className="text-3xl font-bold text-[#59A52C]">R$ 42.377,00</span>
                                    <span className="text-lg text-muted-foreground font-medium">/ R$ 100.000,00</span>
                                </div>
                            </div>
                            <div className="text-right">
                                <span className="text-2xl font-bold text-zinc-800">42.4%</span>
                                <p className="text-xs text-muted-foreground">concluído</p>
                            </div>
                        </div>
                        <Progress value={42.4} className="h-4 bg-zinc-100 [&>div]:bg-[#59A52C]" />
                    </div>
                </CardContent>
            </Card>

            {/* 1. Header Section - KPIs Principais */}
            <div className="grid gap-4 md:grid-cols-4">
                <Card className="border-l-4 border-l-green-500 shadow-sm transition-all hover:shadow-md">
                    <CardHeader className="flex flex-row items-center justify-between space-y-0 pb-2">
                        <CardTitle className="text-sm font-medium">Faturamento Bruto</CardTitle>
                        <DollarSign className="h-4 w-4 text-muted-foreground" />
                    </CardHeader>
                    <CardContent>
                        <div className="text-2xl font-bold text-green-600">
                            {new Intl.NumberFormat('pt-BR', { style: 'currency', currency: 'BRL' }).format(metrics?.totalRevenue || 0)}
                        </div>
                        <p className="text-xs text-muted-foreground flex items-center mt-1">
                            <TrendingUp className="h-3 w-3 text-green-500 mr-1" />
                            +41% desde o último mês
                        </p>
                    </CardContent>
                </Card>
                <Card className="border-l-4 border-l-emerald-500 shadow-sm transition-all hover:shadow-md">
                    <CardHeader className="flex flex-row items-center justify-between space-y-0 pb-2">
                        <CardTitle className="text-sm font-medium">Lucro Líquido</CardTitle>
                        <Activity className="h-4 w-4 text-muted-foreground" />
                    </CardHeader>
                    <CardContent>
                        <div className="text-2xl font-bold text-emerald-600">
                            {new Intl.NumberFormat('pt-BR', { style: 'currency', currency: 'BRL' }).format(metrics?.totalProfit || 0)}
                        </div>
                        <p className="text-xs text-muted-foreground flex items-center mt-1">
                            <TrendingUp className="h-3 w-3 text-emerald-500 mr-1" />
                            {metrics?.profitMargin?.toFixed(1)}% margem real
                        </p>
                    </CardContent>
                </Card>
                <Card className="border-l-4 border-l-blue-500 shadow-sm transition-all hover:shadow-md">
                    <CardHeader className="flex flex-row items-center justify-between space-y-0 pb-2">
                        <CardTitle className="text-sm font-medium">Margem Líquida</CardTitle>
                        <PieChart className="h-4 w-4 text-muted-foreground" />
                    </CardHeader>
                    <CardContent>
                        <div className="text-2xl font-bold">{metrics?.profitMargin?.toFixed(1)}%</div>
                        <p className="text-xs text-muted-foreground mt-1">Estável</p>
                    </CardContent>
                </Card>
                <Card className="border-l-4 border-l-purple-500 shadow-sm transition-all hover:shadow-md">
                    <CardHeader className="flex flex-row items-center justify-between space-y-0 pb-2">
                        <CardTitle className="text-sm font-medium">Ticket Médio</CardTitle>
                        <Users className="h-4 w-4 text-muted-foreground" />
                    </CardHeader>
                    <CardContent>
                        <div className="text-2xl font-bold">
                            {new Intl.NumberFormat('pt-BR', { style: 'currency', currency: 'BRL' }).format(
                                transactions?.length > 0 ? (metrics?.totalRevenue || 0) / transactions.length : 0
                            )}
                        </div>
                        <p className="text-xs text-muted-foreground mt-1">Por paciente</p>
                    </CardContent>
                </Card>
            </div>

            {/* 2. Planejamento & Projeções */}
            <Card className="col-span-4 bg-white border-none shadow-md">
                <CardHeader>
                    <div className="flex items-center gap-2">
                        <TrendingUp className="h-5 w-5 text-green-600" />
                        <CardTitle className="text-lg">Planejamento Financeiro & Projeções</CardTitle>
                    </div>
                    <CardDescription>Análise dos recebíveis (Crédito) projetado para os próximos 12 meses.</CardDescription>
                </CardHeader>
                <CardContent>
                    <div className="grid md:grid-cols-3 gap-6 mb-8">
                        <div className="bg-[#59A52C]/10 p-4 rounded-xl border border-[#59A52C]/20">
                            <span className="text-xs font-semibold text-[#59A52C] uppercase tracking-wider">Total já Garantido (12 Meses)</span>
                            <div className="text-3xl font-bold text-[#448022] mt-1">R$ 10.580,00</div>
                        </div>
                        <div className="p-4 bg-muted/30 rounded-xl border border-border/50">
                            <span className="text-xs text-muted-foreground font-bold uppercase tracking-wider">Pico de Recebimento</span>
                            <div className="text-xl font-bold text-zinc-700 mt-1">Fev. <span className="text-sm font-medium text-muted-foreground">(R$ 3.110)</span></div>
                        </div>
                        <div className="p-4 bg-muted/30 rounded-xl border border-border/50">
                            <span className="text-xs text-muted-foreground font-bold uppercase tracking-wider">Média Mensal</span>
                            <div className="text-xl font-bold text-zinc-700 mt-1">R$ 881,67</div>
                        </div>
                    </div>

                    <div className="grid md:grid-cols-3 gap-8">
                        <div className="md:col-span-2 h-[280px]">
                            <h4 className="text-sm font-medium mb-4 text-zinc-600">Cronograma de Entradas - Valores Garantidos</h4>
                            <ResponsiveContainer width="100%" height="100%">
                                <BarChart data={PROJECTION_DATA}>
                                    <defs>
                                        <linearGradient id="barGradient" x1="0" y1="0" x2="0" y2="1">
                                            <stop offset="0%" stopColor="#4ade80" stopOpacity={1} />
                                            <stop offset="100%" stopColor="#86efac" stopOpacity={0.6} />
                                        </linearGradient>
                                    </defs>
                                    <CartesianGrid strokeDasharray="3 3" vertical={false} opacity={0.2} />
                                    <XAxis dataKey="name" axisLine={false} tickLine={false} tick={{ fontSize: 12, fill: '#888' }} />
                                    <Tooltip
                                        cursor={{ fill: 'transparent' }}
                                        contentStyle={{ borderRadius: '8px', border: 'none', boxShadow: '0 4px 12px rgba(0,0,0,0.1)' }}
                                    />
                                    <Bar dataKey="guaranteed" fill="url(#barGradient)" radius={[6, 6, 0, 0]} barSize={32} />
                                </BarChart>
                            </ResponsiveContainer>
                            <div className="flex items-center gap-2 mt-2 ml-2">
                                <div className="w-3 h-3 rounded-full bg-[#4ade80]" />
                                <span className="text-xs text-muted-foreground">Recebíveis de Crédito (Já vendido)</span>
                            </div>
                        </div>
                        <div className="space-y-6">
                            <div className="h-[120px]">
                                <h4 className="text-sm font-medium mb-2 text-zinc-600">Curva de Recebimento (Garantido)</h4>
                                <ResponsiveContainer width="100%" height="100%">
                                    <AreaChart data={PROJECTION_DATA}>
                                        <defs>
                                            <linearGradient id="colorGuaranteed" x1="0" y1="0" x2="0" y2="1">
                                                <stop offset="5%" stopColor="#22c55e" stopOpacity={0.2} />
                                                <stop offset="95%" stopColor="#22c55e" stopOpacity={0} />
                                            </linearGradient>
                                        </defs>
                                        <Area type="monotone" dataKey="guaranteed" stroke="#22c55e" fillOpacity={1} fill="url(#colorGuaranteed)" strokeWidth={2} />
                                    </AreaChart>
                                </ResponsiveContainer>
                            </div>
                            <div className="space-y-3 pt-4 border-t border-dashed">
                                <h4 className="text-xs font-bold text-muted-foreground uppercase tracking-wider">Por Trimestre</h4>
                                <div className="flex justify-between text-sm py-1 border-b border-zinc-50">
                                    <span className="text-muted-foreground">Q1 (Jan-Mar)</span>
                                    <span className="font-bold text-zinc-700">R$ 6.345,00</span>
                                </div>
                                <div className="flex justify-between text-sm py-1 border-b border-zinc-50">
                                    <span className="text-muted-foreground">Q2 (Abr-Jun)</span>
                                    <span className="font-bold text-zinc-700">R$ 3.485,00</span>
                                </div>
                                <div className="flex justify-between text-sm py-1">
                                    <span className="text-muted-foreground">Q3 (Jul-Set)</span>
                                    <span className="font-bold text-zinc-700">R$ 375,00</span>
                                </div>
                            </div>
                        </div>
                    </div>
                </CardContent>
            </Card>

            {/* 3. Evolução de Vendas & Lucratividade */}
            <div className="grid md:grid-cols-1 gap-6">
                <Card>
                    <CardHeader>
                        <div className="flex items-center justify-between">
                            <div>
                                <CardTitle>Evolução de Vendas</CardTitle>
                                <CardDescription>Acompanhe a receita diária ou mensal do período.</CardDescription>
                            </div>
                            <div className="flex bg-muted p-1 rounded-md">
                                <Button
                                    variant={salesView === 'daily' ? 'default' : 'ghost'}
                                    size="sm"
                                    onClick={() => setSalesView('daily')}
                                    className={`h-7 text-xs ${salesView === 'daily' ? 'bg-[#59A52C] hover:bg-[#59A52C]/90 text-white shadow-sm' : 'text-muted-foreground'}`}
                                >
                                    Diário
                                </Button>
                                <Button
                                    variant={salesView === 'monthly' ? 'default' : 'ghost'}
                                    size="sm"
                                    onClick={() => setSalesView('monthly')}
                                    className={`h-7 text-xs ${salesView === 'monthly' ? 'bg-[#59A52C] hover:bg-[#59A52C]/90 text-white shadow-sm' : 'text-muted-foreground'}`}
                                >
                                    Mensal
                                </Button>
                            </div>
                        </div>
                    </CardHeader>
                    <CardContent>
                        <div className="h-[300px]">
                            <ResponsiveContainer width="100%" height="100%">
                                <AreaChart data={currentSalesData}>
                                    <defs>
                                        <linearGradient id="colorSales" x1="0" y1="0" x2="0" y2="1">
                                            <stop offset="5%" stopColor="#59A52C" stopOpacity={0.3} />
                                            <stop offset="95%" stopColor="#59A52C" stopOpacity={0} />
                                        </linearGradient>
                                    </defs>
                                    <CartesianGrid strokeDasharray="3 3" vertical={false} opacity={0.3} />
                                    <XAxis dataKey="day" axisLine={false} tickLine={false} tick={{ fontSize: 12 }} />
                                    <YAxis axisLine={false} tickLine={false} tick={{ fontSize: 12 }} tickFormatter={(value) => `R$${value / 1000}k`} />
                                    <Tooltip
                                        formatter={(value) => [`R$ ${value}`, "Vendas"]}
                                        contentStyle={{ borderRadius: '8px', border: 'none', boxShadow: '0 4px 12px rgba(0,0,0,0.1)' }}
                                    />
                                    <Area type="monotone" dataKey="value" stroke="#59A52C" strokeWidth={3} fillOpacity={1} fill="url(#colorSales)" />
                                </AreaChart>
                            </ResponsiveContainer>
                        </div>
                    </CardContent>
                </Card>
            </div>

            {/* 4. Bottom Row: Lucratividade, Despesas & Top Pacientes */}
            <div className="grid md:grid-cols-2 lg:grid-cols-3 gap-6">
                {/* Lucratividade */}
                <Card className="lg:col-span-3 xl:col-span-1 border-none shadow-md">
                    <CardHeader>
                        <div className="flex items-center justify-between">
                            <CardTitle className="text-base font-bold text-zinc-800">Lucratividade por Procedimento</CardTitle>
                            <div className="flex bg-muted p-0.5 rounded-md">
                                <button
                                    onClick={() => setProfitabilityView('net')}
                                    className={`text-[10px] px-2 py-1 rounded-sm font-medium transition-all ${profitabilityView === 'net' ? 'bg-white shadow text-[#59A52C]' : 'text-muted-foreground'}`}
                                >
                                    Lucro Liq.
                                </button>
                                <button
                                    onClick={() => setProfitabilityView('gross')}
                                    className={`text-[10px] px-2 py-1 rounded-sm font-medium transition-all ${profitabilityView === 'gross' ? 'bg-white shadow text-[#59A52C]' : 'text-muted-foreground'}`}
                                >
                                    Fat. Bruto
                                </button>
                            </div>
                        </div>
                        <CardDescription>Resultado após descontar custos diretos</CardDescription>
                    </CardHeader>
                    <CardContent>
                        <div className="space-y-6">
                            {profitabilityData.length > 0 ? (
                                profitabilityData.map((item) => (
                                    <div key={item.name} className="space-y-2">
                                        <div className="flex justify-between text-xs font-semibold text-zinc-700">
                                            <span>{item.name}</span>
                                            <span className="text-zinc-500 font-normal">
                                                {new Intl.NumberFormat('pt-BR', { style: 'currency', currency: 'BRL' }).format(profitabilityView === 'net' ? item.net : item.gross)}
                                            </span>
                                        </div>
                                        <div className="relative pt-1">
                                            <Progress
                                                value={((profitabilityView === 'net' ? item.net : item.gross) / maxGrossValue) * 100}
                                                className="h-3 bg-zinc-100 [&>div]:bg-[#59A52C] transition-all duration-500"
                                            />
                                        </div>
                                    </div>
                                ))
                            ) : (
                                <div className="text-sm text-center text-muted-foreground py-8">
                                    Nenhum dado de venda encontrado.
                                </div>
                            )}
                        </div>
                    </CardContent>
                </Card>

                {/* Despesas */}
                <Card>
                    <CardHeader>
                        <CardTitle className="text-base">Despesas e Custos</CardTitle>
                        <CardDescription>Composição do faturamento</CardDescription>
                    </CardHeader>
                    <CardContent>
                        <div className="flex items-center justify-center relative h-[200px]">
                            <ResponsiveContainer width="100%" height="100%">
                                <PieChart>
                                    <Pie
                                        data={[
                                            { name: 'Comissões', value: metrics?.totalCommissions || 0, color: '#3b82f6' },
                                            { name: 'Materiais', value: metrics?.totalMaterials || 0, color: '#8b5cf6' },
                                            { name: 'Custos Fixos', value: metrics?.totalFixedCosts || 0, color: '#f59e0b' },
                                        ]}
                                        innerRadius={60}
                                        outerRadius={80}
                                        paddingAngle={5}
                                        dataKey="value"
                                        stroke="none"
                                    >
                                        {[
                                            { name: 'Comissões', value: metrics?.totalCommissions || 0, color: '#3b82f6' },
                                            { name: 'Materiais', value: metrics?.totalMaterials || 0, color: '#8b5cf6' },
                                            { name: 'Custos Fixos', value: metrics?.totalFixedCosts || 0, color: '#f59e0b' },
                                        ].map((entry, index) => (
                                            <Cell key={`cell-${index}`} fill={entry.color} />
                                        ))}
                                    </Pie>
                                    <Tooltip formatter={(value) => new Intl.NumberFormat('pt-BR', { style: 'currency', currency: 'BRL' }).format(Number(value))} />
                                </PieChart>
                            </ResponsiveContainer>
                            <div className="absolute inset-0 flex items-center justify-center pointer-events-none">
                                <span className="font-bold text-xl">{((1 - (metrics?.profitMargin || 0) / 100) * 100).toFixed(0)}%</span>
                            </div>
                        </div>
                        <div className="flex justify-center gap-4 mt-4 text-xs flex-wrap">
                            <div className="flex items-center gap-1">
                                <div className="w-2 h-2 rounded-full bg-blue-500"></div>
                                <span>Comissões: {new Intl.NumberFormat('pt-BR', { style: 'currency', currency: 'BRL' }).format(metrics?.totalCommissions || 0)}</span>
                            </div>
                            <div className="flex items-center gap-1">
                                <div className="w-2 h-2 rounded-full bg-purple-500"></div>
                                <span>Materiais: {new Intl.NumberFormat('pt-BR', { style: 'currency', currency: 'BRL' }).format(metrics?.totalMaterials || 0)}</span>
                            </div>
                            <div className="flex items-center gap-1">
                                <div className="w-2 h-2 rounded-full bg-amber-500"></div>
                                <span>Custos Fixos: {new Intl.NumberFormat('pt-BR', { style: 'currency', currency: 'BRL' }).format(metrics?.totalFixedCosts || 0)}</span>
                            </div>
                        </div>
                        <div className="mt-6 p-3 bg-yellow-50 dark:bg-yellow-900/10 border border-yellow-100 rounded-lg">
                            <div className="flex items-center gap-2 text-yellow-700 dark:text-yellow-500 font-semibold text-xs mb-1">
                                <Activity className="h-3 w-3" /> Resumo de Custos
                            </div>
                            <p className="text-[10px] text-yellow-600/80">
                                24% do faturamento são custos operacionais e comissões.
                            </p>
                        </div>
                    </CardContent>
                </Card>

                {/* Top Pacientes */}
                <Card>
                    <CardHeader>
                        <CardTitle className="text-base">Top 5 Pacientes</CardTitle>
                        <CardDescription>Maiores valores faturados</CardDescription>
                    </CardHeader>
                    <CardContent>
                        <div className="space-y-4">
                            {topPatientsData.length > 0 ? (
                                topPatientsData.map((patient, i) => (
                                    <div key={i} className="flex items-center justify-between p-2 hover:bg-muted/50 rounded-lg transition-colors group cursor-pointer" onClick={onNavigateToTransactions}>
                                        <div className="flex items-center gap-3">
                                            <div className="flex items-center justify-center w-6 h-6 rounded-full bg-green-100 text-green-700 text-xs font-bold">
                                                {i + 1}
                                            </div>
                                            <span className="text-sm font-medium truncate max-w-[120px]" title={patient.name}>
                                                {patient.name}
                                            </span>
                                        </div>
                                        <div className="flex items-center gap-2">
                                            <span className="text-sm font-semibold text-muted-foreground group-hover:text-primary transition-colors">
                                                {new Intl.NumberFormat('pt-BR', { style: 'currency', currency: 'BRL' }).format(patient.value)}
                                            </span>
                                            <ExternalLink className="h-3 w-3 opacity-0 group-hover:opacity-100 transition-opacity" />
                                        </div>
                                    </div>
                                ))
                            ) : (
                                <div className="text-sm text-center text-muted-foreground py-8">
                                    Nenhum dado de venda encontrado.
                                </div>
                            )}
                        </div>
                        <Button variant="ghost" className="w-full mt-4 text-xs" onClick={onNavigateToTransactions}>
                            Ver todos os pacientes
                        </Button>
                    </CardContent>
                </Card>
            </div>
        </div>
    );
}
