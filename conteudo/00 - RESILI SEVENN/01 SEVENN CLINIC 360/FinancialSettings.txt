import { useEffect, useState } from "react";
import { Card, CardContent, CardDescription, CardHeader, CardTitle } from "@/components/ui/card";
import { Input } from "@/components/ui/input";
import { Button } from "@/components/ui/button";
import { toast } from "sonner";
import { supabase } from "@/lib/supabase";
import { useAuth } from "@/contexts/AuthContext";
import { Loader2, Save, Target, CreditCard, CheckCircle2 } from "lucide-react";

interface PaymentFee {
    metodo_pagamento: string;
    taxa_percentual: number;
}

// FORMATADOR DE MOEDA (Estilo Banco)
// Transforma "1234" em "12,34" e "123456" em "1.234,56"
const formatMoney = (value: string) => {
    // Remove tudo que não é dígito
    const cleanValue = value.replace(/\D/g, "");

    // Converte para número e divide por 100 para considerar centavos
    const numberValue = Number(cleanValue) / 100;

    // Formata para BRL
    return new Intl.NumberFormat("pt-BR", {
        minimumFractionDigits: 2,
        maximumFractionDigits: 2,
    }).format(numberValue);
};

// CONVERSOR REVERSO (Texto formatado -> Float para o Banco)
const parseMoney = (value: string) => {
    if (!value) return 0;
    // Remove pontos e troca vírgula por ponto
    const clean = value.replace(/\./g, "").replace(",", ".");
    return parseFloat(clean) || 0;
};

export function FinancialSettings() {
    const { clinic } = useAuth();
    const [loading, setLoading] = useState(false);
    const [savingMeta, setSavingMeta] = useState(false);

    // Estados como STRING para permitir a formatação visual (pontos e vírgulas)
    const [metaFaturamento, setMetaFaturamento] = useState<string>("");
    const [verbaMarketing, setVerbaMarketing] = useState<string>("");
    const [fees, setFees] = useState<PaymentFee[]>([]);

    useEffect(() => {
        if (clinic?.id) loadSettings();
    }, [clinic?.id]);

    const loadSettings = async () => {
        if (!clinic?.id) return;
        setLoading(true);
        try {
            // 1. Buscar Config Global
            const { data: clinicData } = await supabase
                .from('clinics')
                .select('meta_faturamento, verba_marketing')
                .eq('id', clinic.id)
                .single();

            // Carrega e já formata para o padrão visual BR
            const meta = clinicData?.meta_faturamento ? clinicData.meta_faturamento.toFixed(2) : "000";
            const mkt = clinicData?.verba_marketing ? clinicData.verba_marketing.toFixed(2) : "000";

            setMetaFaturamento(formatMoney(String(meta).replace('.', ''))); // Trick para passar pelo formatador inicial
            setVerbaMarketing(formatMoney(String(mkt).replace('.', '')));

            // 2. Buscar Taxas
            const { data: feesData } = await supabase
                .from('config_taxas_pagamento')
                .select('metodo_pagamento, taxa_percentual')
                .eq('clinic_id', clinic.id);

            const defaultTypes = ['pix', 'debito', 'credito_1x', 'credito_2x_4x', 'credito_5x_8x', 'credito_9x_12x'];
            const mergedFees = defaultTypes.map(type => {
                const existing = feesData?.find(f => f.metodo_pagamento === type);
                return {
                    metodo_pagamento: type,
                    taxa_percentual: existing ? Number(existing.taxa_percentual) : 0
                };
            });
            setFees(mergedFees);

        } catch (error) {
            console.error("Erro ao carregar configs:", error);
        } finally {
            setLoading(false);
        }
    };

    const handleMetaChange = (value: string) => {
        setMetaFaturamento(formatMoney(value));
    };

    const handleMarketingChange = (value: string) => {
        setVerbaMarketing(formatMoney(value));
    };

    const handleSavePlanning = async () => {
        if (!clinic?.id) return;
        setSavingMeta(true);
        try {
            const { error } = await supabase
                .from('clinics')
                .update({
                    meta_faturamento: parseMoney(metaFaturamento),
                    verba_marketing: parseMoney(verbaMarketing)
                })
                .eq('id', clinic.id);

            if (error) throw error;
            toast.success("Planejamento atualizado com sucesso!");
        } catch (error) {
            toast.error("Erro ao salvar planejamento.");
        } finally {
            setSavingMeta(false);
        }
    };

    const handleFeeChange = (type: string, value: string) => {
        // Taxas não precisam de máscara de milhar, apenas número simples
        setFees(prev => prev.map(f =>
            f.metodo_pagamento === type ? { ...f, taxa_percentual: parseFloat(value) || 0 } : f
        ));
    };

    const handleFeeBlur = async (fee: PaymentFee) => {
        if (!clinic?.id) return;
        try {
            const { error } = await supabase
                .from('config_taxas_pagamento')
                .upsert({
                    clinic_id: clinic.id,
                    metodo_pagamento: fee.metodo_pagamento,
                    taxa_percentual: fee.taxa_percentual
                }, { onConflict: 'clinic_id, metodo_pagamento' });

            if (error) throw error;

            toast.success(`Taxa de ${getLabel(fee.metodo_pagamento)} salva!`, {
                duration: 2000,
                position: 'bottom-right'
            });
        } catch (error) {
            console.error(error);
            toast.error("Erro ao salvar taxa.");
        }
    };

    const getLabel = (type: string) => {
        switch (type) {
            case 'pix': return 'Pix';
            case 'debito': return 'Cartão de Débito';
            case 'credito_1x': return 'Crédito à Vista (1x)';
            case 'credito_2x_4x': return 'Crédito Parcelado (2x a 4x)';
            case 'credito_5x_8x': return 'Crédito Parcelado (5x a 8x)';
            case 'credito_9x_12x': return 'Crédito Parcelado (9x a 12x)';
            default: return type;
        }
    };

    if (loading) return <div className="flex justify-center p-8"><Loader2 className="animate-spin text-slate-400" /></div>;

    return (
        <div className="space-y-6 max-w-5xl">
            <div className="grid grid-cols-1 md:grid-cols-2 gap-8">

                {/* 1. CARTÃO DE PLANEJAMENTO */}
                <Card className="border-none shadow-sm bg-white h-full flex flex-col">
                    <CardHeader>
                        <CardTitle className="flex items-center gap-2 text-base text-slate-800">
                            <Target className="w-4 h-4 text-blue-600" />
                            Planejamento Mensal
                        </CardTitle>
                        <CardDescription>Valores base para suas metas e cálculo de lucro.</CardDescription>
                    </CardHeader>
                    <CardContent className="space-y-5 flex-1">
                        <div className="space-y-2">
                            <label className="text-sm font-medium text-slate-700">Meta de Faturamento (Mensal)</label>
                            <div className="relative">
                                <span className="absolute left-3 top-2.5 text-slate-400 font-medium">R$</span>
                                {/* MUDANÇA: Type text e removido text-lg */}
                                <Input
                                    type="text"
                                    className="pl-9 font-medium"
                                    value={metaFaturamento}
                                    onChange={(e) => handleMetaChange(e.target.value)}
                                    placeholder="0,00"
                                />
                            </div>
                            <p className="text-xs text-slate-400">Ex: 100.000,00</p>
                        </div>

                        <div className="space-y-2">
                            <label className="text-sm font-medium text-slate-700">Verba de Marketing (Fixo/Mês)</label>
                            <div className="relative">
                                <span className="absolute left-3 top-2.5 text-slate-400 font-medium">R$</span>
                                <Input
                                    type="text"
                                    className="pl-9 font-medium"
                                    value={verbaMarketing}
                                    onChange={(e) => handleMarketingChange(e.target.value)}
                                    placeholder="0,00"
                                />
                            </div>
                        </div>
                    </CardContent>
                    <div className="p-6 pt-0 mt-auto">
                        <Button onClick={handleSavePlanning} disabled={savingMeta} className="w-full bg-slate-900 hover:bg-slate-800 text-white">
                            {savingMeta ? <Loader2 className="w-4 h-4 animate-spin mr-2" /> : <Save className="w-4 h-4 mr-2" />}
                            Salvar Metas
                        </Button>
                    </div>
                </Card>

                {/* 2. CARTÃO DE TAXAS */}
                <Card className="border-none shadow-sm bg-white h-full">
                    <CardHeader>
                        <CardTitle className="flex items-center gap-2 text-base text-slate-800">
                            <CreditCard className="w-4 h-4 text-emerald-600" />
                            Taxas de Maquininha
                        </CardTitle>
                        <CardDescription>
                            Taxas descontadas automaticamente. <br />
                            <span className="text-emerald-600 font-medium text-xs flex items-center gap-1 mt-1">
                                <CheckCircle2 className="w-3 h-3" /> Salva automaticamente ao sair do campo
                            </span>
                        </CardDescription>
                    </CardHeader>
                    <CardContent className="space-y-4">
                        {fees.map((fee) => (
                            <div key={fee.metodo_pagamento} className="flex items-center justify-between gap-4">
                                <label className="text-sm font-medium text-slate-600 flex-1">{getLabel(fee.metodo_pagamento)}</label>
                                <div className="relative w-28">
                                    <Input
                                        type="number"
                                        className="pr-8 text-right font-medium"
                                        value={fee.taxa_percentual}
                                        onChange={(e) => handleFeeChange(fee.metodo_pagamento, e.target.value)}
                                        onBlur={() => handleFeeBlur(fee)}
                                        step="0.01"
                                    />
                                    <span className="absolute right-3 top-2.5 text-slate-400 font-medium">%</span>
                                </div>
                            </div>
                        ))}
                    </CardContent>
                </Card>
            </div>
        </div>
    );
}
