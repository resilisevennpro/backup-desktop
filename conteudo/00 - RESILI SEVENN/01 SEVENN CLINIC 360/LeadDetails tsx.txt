export default function LeadDetailsPage() {
    const { leadId } = useParams();
    const navigate = useNavigate();
    const { profile, clinic } = useAuth();
    const { leads, columns, moveLead, fetchBoardData, updateLead, loading, availableTags, fetchAvailableTags, fetchLeadById } = useCrmStore();
    const { services, fetchServices } = useServicesStore();
    const { professionals, fetchProfessionals } = useSettingsStore();

    const [localName, setLocalName] = useState("");
    const [localEmail, setLocalEmail] = useState("");
    const [emailSuggestion, setEmailSuggestion] = useState("");
    const [emailError, setEmailError] = useState(false);
    const [localBudget, setLocalBudget] = useState("");

    const setHeaderContent = useUIStore(state => state.setHeaderContent);



    const { transactions, fetchTransactions } = useFinanceStore();

    // Adicione isto logo no início do componente LeadDetailsPage
    const [localProcedureId, setLocalProcedureId] = useState(lead?.procedure_id);

    // Isso garante que se você mudar de lead, o botão atualiza
    useEffect(() => {
        setLocalProcedureId(lead?.procedure_id);
    }, [lead?.procedure_id]);


    const formatInteger = (val: string | number | undefined) => {
        if (val === undefined || val === null || val === "") return "";
        const clean = String(val).replace(/\D/g, "");
        if (!clean) return "";
        return new Intl.NumberFormat('pt-BR').format(parseInt(clean));
    };

    // --- FETCH SOB DEMANDA ---
    useEffect(() => {
        if (leadId && !lead && !loading) {
            // Validação simples de formato (ex: se é UUID ou string válida)
            if (leadId.length > 5) {
                fetchLeadById(leadId);
            }

    }, [leadId, lead, loading, fetchLeadById]);

    useEffect(() => {
        if (lead) {
            setLocalName(lead.name);
            setLocalEmail(lead.email || "");
            setLocalBudget(formatInteger(lead.budget));
        }

    // Fetch dependencies
    useEffect(() => {
        if (clinic?.id) {

            fetchProfessionals(clinic.id);

    const leadTransactions = transactions.filter(t => t.lead_id === leadId);
    const realLtv = leadTransactions.filter(t => t.status === 'paid').reduce((acc, t) => acc + (t.amount || 0), 0);
    const realDebt = leadTransactions.filter(t => t.status === 'pending').reduce((acc, t) => acc + (t.amount || 0), 0);

    const [showDeleteDialog, setShowDeleteDialog] = useState(false);

    const executeUpdateLead = async (updates: any, showToast = true) => {
        if (!lead?.id) return;
        await updateLead(lead.id, updates);
        if (showToast) {
            toast.success("Salvo com sucesso", {
                duration: 2000,
                icon: <Check className="w-4 h-4 text-emerald-500" />

        if (!lead?.id) return;

        // 1. SALVAR NO BANCO IMEDIATAMENTE (Prioridade Máxima para garantir consistência)
        // Isso remove a trava que impedia a atualização quando havia agendamentos.
        await executeUpdateLead(updates);

        // 2. DEPOIS verificamos efeitos colaterais (agendamentos) para sugerir sync

                    .from('appointments')
                    .select('id')
                    .eq('lead_id', lead.id)
                    .in('status', ['Confirmado', 'Pendente', 'Em Atendimento', 'Chegou'])

                    setActiveAppointmentId(data[0].id);
                    setPendingUpdates(updates);
                    setSyncModalOpen(true);
                    // REMOVIDO: return; (Não bloqueamos mais o fluxo, pois o save já ocorreu)
                }

                console.error("Error checking appointments:", error);
            }

        if (!pendingUpdates || !activeAppointmentId) return;

        // 1. Update Lead

        // 2. Sync Appointment

            const apptUpdates: any = {};

                const { error } = await supabase
                    .from('appointments')
                    .update(apptUpdates)

                    toast.success("Ficha e Agendamento atualizados!", { icon: <CheckCircle className="w-4 h-4 text-emerald-500" /> });

                    toast.error("Erro ao atualizar agendamento.");

        setActiveAppointmentId(null);

            const { error } = await supabase.from('leads').delete().eq('id', lead.id);
            if (error) throw error;

        if (!text) return;

    const validateEmail = (email: string) => {
        const standardRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
        if (!standardRegex.test(email)) return false;

        const [, domain] = email.split('@');
        const allowedDomains = [
            'gmail.com', 'outlook.com', 'hotmail.com', 'live.com', 'msn.com',
            'yahoo.com', 'yahoo.com.br', 'icloud.com', 'me.com',
            'uol.com.br', 'bol.com.br', 'terra.com.br', 'ig.com.br'
        ];
        const allowedExtensions = ['.com', '.com.br', '.net', '.pro', '.dev', '.tech', '.co'];

        const domainMatchesAllowed = allowedDomains.includes(domain.toLowerCase());
        const extMatchesAllowed = allowedExtensions.some(ext => domain.toLowerCase().endsWith(ext));

        return domainMatchesAllowed || extMatchesAllowed;

    const handleEmailChange = (e: React.ChangeEvent<HTMLInputElement>) => {
        const val = e.target.value.toLowerCase().replace(/\s/g, ''); // Normalização: minúscula e sem espaços
        setLocalEmail(val);

        if (val.includes('@')) {
            const [, domainPart] = val.split('@');
            const domains = [
                'gmail.com', 'outlook.com', 'hotmail.com', 'live.com', 'msn.com',
                'yahoo.com', 'yahoo.com.br', 'icloud.com', 'me.com',
                'uol.com.br', 'bol.com.br', 'terra.com.br', 'ig.com.br'
            ];
            const suggestion = domains.find(d => d.startsWith(domainPart) && d !== domainPart);
            if (suggestion && domainPart.length > 1) { // Só sugere após 2 letras do domínio
                setEmailSuggestion(suggestion.slice(domainPart.length));
            } else {
                setEmailSuggestion("");
            }
        } else {
            setEmailSuggestion("");
        }

        setEmailError(val.length > 0 && !validateEmail(val));
    };

        // Correções automáticas comuns
        if (val.endsWith('.con')) val = val.slice(0, -1) + 'm';
        if (val.includes('gamil')) val = val.replace('gamil', 'gmail');
        if (val.includes('gmial')) val = val.replace('gmial', 'gmail');
        if (val.includes('hotmial')) val = val.replace('hotmial', 'hotmail');

        setLocalEmail(val);
        const isValid = val.length > 0 ? !validateEmail(val) : false;
        setEmailError(isValid);
        setEmailSuggestion("");

        if (!isValid && val !== lead?.email) {
            handleUpdateLead({ email: val });
        }
    };

    const [dossierData, setDossierData] = useState<{ sentiment: string, sentimentIcon: any, sentimentColor: string, intent: number, consciousness: string, summary: string, objection?: string } | null>(null);

    const handleFinalizeSale = async () => {
        if (!lead || !quickSaleValue) return;
        setIsProcessingSale(true);

        try {
            const numericValue = Number(quickSaleValue.toString().replace(/[^0-9.]/g, ''));

            if (quickSaleMethod === 'other' && !quickSaleObs.trim()) {

                })(),
                closed_at: new Date().toISOString()
            }).eq('id', lead.id);

            if (syncError) throw new Error("Erro ao sincronizar dados do lead.");

            const { data: dbLead, error: fetchError } = await supabase
                .from('leads')
                .select('total_ltv, stage_history')
                .eq('id', lead.id)
                .single();

            if (fetchError || !dbLead) throw new Error("Erro de segurança: Não foi possível sincronizar o LTV.");

                const now = new Date();
                const yyyy = now.getFullYear();
                const mm = String(now.getMonth() + 1).padStart(2, '0');
                const dd = String(now.getDate()).padStart(2, '0');
                // Força T12:00:00-03:00 para garantir que caia no dia certo no Brasil
                return `${yyyy}-${mm}-${dd}T12:00:00-03:00`;
            };

            // Passo 4: Inserir Transação (Save)
            const { error: finError } = await supabase
                .from('financial_transactions')
                .insert({
                    lead_id: lead.id,
                    clinic_id: lead.clinic_id || clinic?.id,

                        procedure_name: procedureName,

                return;

                type: 'venda',

                created_at: getHardcodedBrazilDate()
            });

            await fetchBoardData(clinic?.id || '');

                duration: 4000

            setIsQuickSaleModalOpen(false);
            setIsProcessingSale(false);

        } catch (error: any) {
            console.error("Erro na Venda Rápida:", error);
            toast.error("Erro ao processar venda: " + (error.message || "Verifique sua conexão."));

    // --- 1. DAILY CREDITS SYSTEM (Frontend Simulation with LocalStorage) ---
    useEffect(() => {
        if (!leadId) return;

        const today = new Date().toDateString();
        const storageKey = `sevenn_credits_${leadId}`;
        const stored = localStorage.getItem(storageKey);

        if (stored) {
            const { date, credits } = JSON.parse(stored);
            if (date === today) {
                setAiCredits(credits);
            } else {
                // New day, reset credits
                setAiCredits(3);
                localStorage.setItem(storageKey, JSON.stringify({ date: today, credits: 3 }));
            }
        } else {
            // First time logic
            setAiCredits(3);
            localStorage.setItem(storageKey, JSON.stringify({ date: today, credits: 3 }));

        }

    const [searchParams, setSearchParams] = useSearchParams();

        const map: Record<string, string> = {

            'prontuario': 'records',

            'galeria': 'gallery',
            'insightsia': 'insights',
            'documentos': 'documents'
        };
        return map[urlParam.toLowerCase()] || "";

            'chat': 'chat',
            'records': 'prontuario',
            'finance': 'financeiro',
            'gallery': 'galeria',
            'insights': 'insightsia',
            'documents': 'documentos'

                                        {/* Lado Esquerdo: SENTIMENTO */}
                                        <div className="p-4 flex flex-col justify-center items-center text-center gap-2 relative">
                                            <div className={cn("w-10 h-10 rounded-full flex items-center justify-center shrink-0 bg-slate-50", dossierData ? "bg-blue-50" : "bg-slate-50")}>
                                                {dossierData ? <dossierData.sentimentIcon className={cn("w-5 h-5", dossierData.sentimentColor)} /> : <Search className="w-4 h-4 text-slate-300" />}
                                            </div>
                                            <div>
                                                <span className="font-space text-[10px] font-bold uppercase text-slate-400 tracking-wider block mb-0.5">Sentimento</span>
                                                <span className={cn("font-manrope text-sm font-bold truncate block", dossierData ? "text-slate-800" : "text-slate-300")}>
                                                    {dossierData ? dossierData.sentiment : 'Indefinido'}
                                                </span>
                                            </div>
                                        </div>

                                        {/* Lado Direito: INTENÇÃO DE COMPRA */}
                                        <div className="p-4 flex flex-col justify-center relative">
                                            <div className="flex justify-between items-center mb-2">
                                                <span className="font-space text-[10px] font-bold uppercase text-slate-400 tracking-wider">Intenção</span>
                                                {dossierData && (
                                                    <span className="font-manrope text-xs font-black text-slate-800">{dossierData.intent}%</span>
                                                )}
                                            </div>
                                            <div className="relative pt-1">
                                                <div className="h-1.5 w-full bg-slate-100 rounded-full overflow-hidden">
                                                    <div
                                                        className="h-full bg-gradient-to-r from-slate-300 to-slate-400 transition-all duration-1000"
                                                        style={{
                                                            width: `${dossierData ? dossierData.intent : 0}%`,
                                                            backgroundImage: dossierData ? 'linear-gradient(to right, #fbbf24, #10b981)' : ''
                                                        }}
                                                    />
                                                </div>
                                                {dossierData && (
                                                    <p className="font-manrope text-[9px] text-slate-400 mt-1.5 text-right">
                                                        <span className="font-bold text-slate-600">{dossierData.consciousness}</span>
                                                    </p>
                                                )}
                                            </div>
                                        </div>
                                    </div>
                                </div>

                                {/* B. PRÓXIMO PASSO RECOMENDADO */}
                                <div className="bg-white p-5 rounded-xl border shadow-sm flex flex-col justify-center">
                                    <h4 className="font-space text-[12px] font-bold uppercase text-slate-600 mb-3 tracking-wider">Próximo Passo Recomendado</h4>
                                    <div className="flex items-center gap-3 p-3 bg-blue-50 border border-blue-100 rounded-lg">
                                        <Calendar className="w-5 h-5 text-blue-600 shrink-0" />
                                        <div className="flex flex-col">
                                            <span className="font-manrope text-xs font-bold text-blue-900">Agendar Avaliação Presencial</span>
                                            <span className="font-manrope text-[10px] text-blue-600 leading-tight">Alta probabilidade de conversão.</span>
                                        </div>
                                    </div>
                                </div>
                            </div>

                            {/* 2. CORPO CENTRAL (Análise Seveen AI Dossiê + Objeções) */}
                            <div className="bg-white p-8 rounded-2xl border shadow-sm relative overflow-hidden flex flex-col justify-center">
                                <div className="absolute top-0 right-0 p-8 opacity-[0.03] pointer-events-none"><Sparkles className="w-64 h-64" /></div>
                                <div className="mb-6 space-y-1">
                                    <h4 className="font-space text-[16px] font-semibold tracking-tight text-slate-900 flex items-center gap-2">
                                        <FileText className="w-3 h-3" /> Análise Sevenn AI
                                    </h4>
                                    <p className="font-manrope text-[10px] text-slate-500 font-medium ml-5">
                                        * A análise é renovada automaticamente ao mudar para próxima etapa (exceto Venda Realizada, Descarte e Nutrição).
                                    </p>
                                </div>

                                {dossierData ? (
                                    <div className="space-y-6 relative z-10 animate-in fade-in duration-700">
                                        <div>
                                            <h5 className="font-space text-[11px] font-bold uppercase text-green-500 mb-2">Análise Geral</h5>
                                            <p className="font-manrope text-sm text-slate-600 leading-relaxed font-medium">
                                                {dossierData.summary}
                                            </p>
                                        </div>

                                        {dossierData.objection && (
                                            <div className="bg-red-50/50 border border-red-100 rounded-lg p-4">
                                                <h5 className="font-space text-[11px] font-bold uppercase text-red-500 mb-2 flex items-center gap-2">
                                                    <div className="w-1.5 h-1.5 rounded-full bg-red-500"></div> Objeções Identificadas
                                                </h5>
                                                <p className="font-manrope text-sm text-slate-700 leading-relaxed">
                                                    {dossierData.objection}
                                                </p>
                                            </div>
                                        )}
                                    </div>
                                ) : (
                                    <div className="text-center py-8">
                                        <Lock className="w-8 h-8 text-slate-200 mx-auto mb-2" />
                                        <p className="font-manrope text-sm text-slate-400">Análise disponível a partir da etapa <span className="font-bold text-slate-500">Orçamento</span>.</p>
                                    </div>
                                )}
                            </div>

                            {/* 3. BLOCO INFERIOR (Sugestão Manual & Créditos) */}
                            <div className={cn("rounded-2xl border p-1 transition-all duration-300", isGeneratingStrategy ? "bg-purple-50 border-purple-200" : "bg-slate-50 border-slate-200")}>
                                <div className="bg-white rounded-xl p-6 shadow-sm">
                                    <div className="flex items-center justify-between mb-4 border-b pb-4 border-slate-100">
                                        <div className="flex items-center gap-3">
                                            <div className={cn("p-2 rounded-lg", isGeneratingStrategy ? "bg-purple-100 text-purple-600 animate-pulse" : "bg-slate-100 text-slate-500")}>
                                                <Sparkles className="w-5 h-5" />
                                            </div>
                                            <div>
                                                <h4 className="font-space text-[14px] font-semibold text-slate-900">Estratégia de Resposta</h4>
                                                <p className="font-manrope text-[11px] text-green-600 font-semibold translate-y-[-1px]">
                                                    {aiCredits > 0
                                                        ? `${aiCredits} créditos disponíveis hoje`
                                                        : <span className="text-red-500 font-bold">Créditos diários esgotados. Volte amanhã.</span>
                                                    }
                                                </p>
                                            </div>
                                        </div>

                                        <Button
                                            disabled={aiCredits <= 0 || isGeneratingStrategy}
                                            onClick={handleUseCredit}
                                            className={cn(
                                                "rounded-full font-manrope font-semibold text-[12px] h-8 shadow-md transition-all",
                                                aiCredits > 0
                                                    ? "bg-gradient-to-r from-purple-600 to-indigo-600 hover:from-purple-700 hover:to-indigo-700 text-white shadow-purple-200"
                                                    : "bg-slate-200 text-slate-400 cursor-not-allowed shadow-none"
                                            )}
                                        >
                                            {isGeneratingStrategy ? (
                                                <><RefreshCw className="w-3.5 h-3.5 mr-2 animate-spin" /> Gerando...</>
                                            ) : (
                                                <><Sparkles className="w-3.5 h-3.5 mr-2 fill-current" /> Gerar Nova Estratégia</>
                                            )}
                                        </Button>
                                    </div>

                                    {/* CONTEÚDO DA RESPOSTA */}
                                    <div className="min-h-[100px] relative">
                                        {isGeneratingStrategy ? (
                                            <div className="space-y-3 p-2 animate-pulse">
                                                <div className="h-4 bg-slate-100 rounded w-3/4"></div>
                                                <div className="h-4 bg-slate-100 rounded w-full"></div>
                                                <div className="h-4 bg-slate-100 rounded w-5/6"></div>
                                                <div className="h-8 bg-slate-50 rounded w-1/3 mt-4"></div>
                                            </div>
                                        ) : (
                                            <>
                                                {/* Exemplo de resultado gerado */}
                                                <div className={cn("transition-all duration-500", activeTab === 'insights' ? "opacity-100" : "opacity-0")}>
                                                    <p className="font-manrope text-[14px] text-slate-700 leading-relaxed mb-6">
                                                        "Olá {lead.name}, tudo bem? Notei que você tem interesse em harmonização e vi que tem um perfil super alinhado com nossos resultados naturais. Que tal agendarmos uma avaliação para você conhecer nossa tecnologia exclusiva?"
                                                    </p>

                                                    <Button
                                                        variant="secondary"
                                                        className="font-manrope h-9 gap-2 text-[12px] font-semibold text-slate-600 bg-slate-100 hover:bg-slate-200 w-full justify-center border border-slate-200"
                                                        onClick={() => copyAndGoToChat(`Olá ${lead.name}, tudo bem? Notei que você tem interesse em harmonização e vi que tem um perfil super alinhado com nossos resultados naturais. Que tal agendarmos uma avaliação para você conhecer nossa tecnologia exclusiva?`)}
                                                    >
                                                        <MessageSquare className="w-4 h-4 ml-0.5" /> Copiar e ir para o Chat
                                                    </Button>
                                                </div>
                                            </>
                                        )}
                                    </div>
                                </div>
                            </div>
                        </div>
                    </TabsContent>

                    {/* PRONTUÁRIO - MEDICAL RECORDS */}
                    <TabsContent value="records" className="flex-1 flex flex-col mt-0 data-[state=inactive]:hidden overflow-hidden">
                        <MedicalRecordTab leadId={lead?.id || ""} clinicId={lead?.clinic_id || ""} />
                    </TabsContent>

                    {/* DOCUMENTS - NEW TAB */}
                    <TabsContent value="documents" className="flex-1 overflow-hidden mt-0 data-[state=inactive]:hidden">
                        {lead && <DocumentsTab leadId={lead.id} clinicId={lead.clinic_id} leadName={lead.name} />}
                    </TabsContent>


                    {/* GALLERY TAB */}
                    <TabsContent value="gallery" className="flex-1 overflow-hidden mt-0 data-[state=active]:flex data-[state=inactive]:hidden flex-col">
                        <LeadGalleryTab
                            leadId={leadId}
                            onSendToChat={(url) => {
                                setMsg(`[Imagem: ${url}]`);
                                setActiveTab("chat");
                            }}
                        />
                    </TabsContent>

                    <TabsContent value="finance" className="flex-1 overflow-hidden mt-0 data-[state=inactive]:hidden flex-col bg-slate-50/50">
                        <LeadFinancialTab />
                    </TabsContent>
                </Tabs>

            </main>

            {/* Sync Modal */}
            <Dialog open={syncModalOpen} onOpenChange={setSyncModalOpen}>
                <DialogContent className="font-manrope sm:max-w-md">
                    <DialogHeader>
                        <DialogTitle className="flex items-center gap-2 text-slate-800">
                            <RefreshCw className="w-5 h-5 text-indigo-600" />
                            Sincronização de Agenda
                        </DialogTitle>
                        <DialogDescription className="pt-2 text-slate-600">
                            Identificamos um agendamento ativo para este paciente. Deseja atualizar as informações no calendário também?
                        </DialogDescription>
                    </DialogHeader>
                    <DialogFooter className="gap-2 sm:justify-between mt-4">
                        <Button variant="outline" onClick={() => handleSyncAction(false)} className="w-full text-slate-600">
                            Não, apenas na ficha
                        </Button>
                        <Button onClick={() => handleSyncAction(true)} className="w-full bg-indigo-600 hover:bg-indigo-700 text-white gap-2">
                            <CheckCircle className="w-4 h-4" />
                            Sim, atualizar ambos
                        </Button>
                    </DialogFooter>
                </DialogContent>
            </Dialog>

            {/* Quick Sale Modal */}
            <Dialog open={isQuickSaleModalOpen} onOpenChange={setIsQuickSaleModalOpen}>
                <DialogContent className="font-manrope sm:max-w-md bg-white border-none shadow-2xl">
                    <DialogHeader className="border-b border-slate-100 pb-4">
                        <DialogTitle className="flex items-center gap-2 text-slate-900 text-lg">
                            <div className="bg-emerald-100 p-2 rounded-full">
                                <DollarSign className="w-5 h-5 text-emerald-600" />
                            </div>
                            Venda Rápida
                        </DialogTitle>
                        <DialogDescription className="text-slate-600">
                            Use esta ação para concluir a venda deste lead (sem agendamento) ou regularizar pagamentos pendentes. Ao confirmar, <span className="font-bold text-emerald-500">a Venda será registrada e o Lead será movido automaticamente </span> para etapa de Venda Realizada no seu CRM.
                        </DialogDescription>
                    </DialogHeader>

                    <div className="py-4 space-y-4">
                        {/* Valor */}
                        <div className="space-y-1.5">
                            <label className="text-xs font-bold uppercase text-slate-500 tracking-wide">Valor da Venda (R$)</label>
                            <div className="relative">
                                <span className="absolute left-3 top-2.5 text-slate-400 font-bold">R$</span>
                                <Input
                                    type="number"
                                    className="pl-9 font-bold text-lg text-slate-800 border-slate-200 focus:border-emerald-500 focus:ring-emerald-500/20"
                                    placeholder="0.00"
                                    value={quickSaleValue}
                                    onChange={(e) => setQuickSaleValue(e.target.value)}
                                    autoFocus
                                />
                            </div>
                        </div>

                        {/* Pagamento */}
                        <div className="grid grid-cols-2 gap-4">
                            <div className="space-y-1.5">
                                <label className="text-xs font-bold uppercase text-slate-500 tracking-wide">Pagamento</label>
                                <Select value={quickSaleMethod} onValueChange={setQuickSaleMethod}>
                                    <SelectTrigger className="font-semibold text-slate-700">
                                        <SelectValue />
                                    </SelectTrigger>
                                    <SelectContent>
                                        <SelectItem value="pix">Pix</SelectItem>
                                        <SelectItem value="money">Dinheiro</SelectItem>
                                        <SelectItem value="credit">Cartão de Crédito</SelectItem>
                                        <SelectItem value="debit">Cartão de Débito</SelectItem>
                                        <SelectItem value="boleto">Boleto Bancário</SelectItem>
                                        <SelectItem value="partnership">Parceria</SelectItem>
                                        <SelectItem value="other">Outro / A definir</SelectItem>
                                    </SelectContent>
                                </Select>
                            </div>

                            {/* Parcelas (Condicional) */}
                            {(quickSaleMethod === 'credit') && (
                                <div className="space-y-1.5 animate-in fade-in slide-in-from-left-2">
                                    <label className="text-xs font-bold uppercase text-slate-500 tracking-wide">Parcelas</label>
                                    <Select
                                        value={String(quickSaleInstallments)}
                                        onValueChange={(val) => setQuickSaleInstallments(Number(val))}
                                    >
                                        <SelectTrigger className="font-semibold text-slate-700">
                                            <SelectValue />
                                        </SelectTrigger>
                                        <SelectContent>
                                            {Array.from({ length: 12 }, (_, i) => i + 1).map(num => (
                                                <SelectItem key={num} value={String(num)}>{num}x</SelectItem>
                                            ))}
                                        </SelectContent>
                                    </Select>
                                </div>
                            )}
                        </div>

                        {/* Observações */}
                        <div className="space-y-1.5">
                            <label className="text-xs font-bold uppercase text-slate-500 tracking-wide">Observações</label>
                            <textarea
                                className={cn(
                                    "w-full min-h-[80px] rounded-md border bg-slate-50 px-3 py-2 text-sm text-slate-700 outline-none transition-all resize-none font-medium placeholder:text-slate-400",
                                    (quickSaleMethod === 'other' || quickSaleMethod === 'partnership') && !quickSaleObs.trim() ? "border-amber-400 ring-1 ring-amber-100" : "border-slate-200 focus:border-emerald-500 focus:ring-1 focus:ring-emerald-500/20"
                                )}
                                placeholder={(quickSaleMethod === 'other' || quickSaleMethod === 'partnership') ? "Descreva os detalhes (Obrigatório)..." : "Detalhes adicionais sobre a venda..."}
                                value={quickSaleObs}
                                onChange={(e) => setQuickSaleObs(e.target.value)}
                            />
                        </div>
                    </div>

                    <DialogFooter className="border-t border-slate-100 pt-4">
                        <Button
                            variant="ghost"
                            onClick={() => setIsQuickSaleModalOpen(false)}
                            disabled={isProcessingSale}
                        >
                            Cancelar
                        </Button>
                        <Button
                            onClick={handleFinalizeSale}
                            className="bg-emerald-600 hover:bg-emerald-700 text-white font-bold gap-2 min-w-[140px] disabled:opacity-50 disabled:cursor-not-allowed"
                            disabled={
                                isProcessingSale ||
                                !quickSaleValue ||
                                ((quickSaleMethod === 'other' || quickSaleMethod === 'partnership') && !quickSaleObs.trim())
                            }
                        >
                            {isProcessingSale ? (
                                <><RefreshCw className="w-4 h-4 animate-spin" /> Processando</>
                            ) : (
                                <><CheckCircle className="w-4 h-4" /> Finalizar Venda</>
                            )}
                        </Button>
                    </DialogFooter>
                </DialogContent>
            </Dialog>

            {/* Delete Confirmation Dialog */}
            <Dialog open={showDeleteDialog} onOpenChange={setShowDeleteDialog}>
                <DialogContent className="sm:max-w-[425px]">
                    <DialogHeader>
                        <DialogTitle className="text-red-600 flex items-center gap-2">
                            <AlertTriangle className="w-5 h-5" />
                            Confirmar Exclusão
                        </DialogTitle>
                        <DialogDescription className="pt-3">
                            Você tem certeza que deseja excluir o lead <strong>{lead?.name}</strong>?
                            <br /><br />
                            Esta ação é irreversível e todos os dados relacionados (histórico, atividades, documentos) serão perdidos.
                        </DialogDescription>
                    </DialogHeader>
                    <DialogFooter className="gap-2 sm:gap-0">
                        <Button
                            variant="outline"
                            onClick={() => setShowDeleteDialog(false)}
                        >
                            Cancelar
                        </Button>
                        <Button
                            variant="destructive"
                            onClick={handleDeleteLead}
                            className="gap-2"
                        >
                            <Trash2 className="w-4 h-4" />
                            Excluir Lead
                        </Button>
                    </DialogFooter>
                </DialogContent>
            </Dialog>
        </div>
    );
}
