{/* SEÇÃO 3: ETIQUETAS */}
                    <div className="space-y-2">
                        <h3 className="font-space text-[12px] font-semibold tracking-[0.06em] text-slate-700 uppercase flex items-center gap-2 border-t border-slate-200/70 pt-5 pb-2"><Tag className="w-3.5 h-3.5 text-slate-400" /> Etiquetas</h3>
                        <div className="flex flex-wrap gap-1.5 pt-0.5">
                            {lead.tags?.map(t => (
                                <Badge key={t} variant="secondary" className="bg-white border text-slate-600 h-auto font-manrope text-[12px] font-semibold px-2.5 py-1 rounded-full group">
                                    {t}
                                    <X
                                        className="w-3 h-3 ml-1 text-slate-300 cursor-pointer hover:text-red-500 hidden group-hover:block"
                                        onClick={() => {
                                            const newTags = lead.tags?.filter(tag => tag !== t) || [];
                                            handleUpdateLead({ tags: newTags });
                                        }}
                                    />
                                </Badge>
                            ))}
                            <div className="relative inline-block">
                                <div
                                    className="flex items-center h-6 rounded-full text-slate-400 border border-dashed border-slate-300 hover:text-primary hover:border-primary px-2 font-manrope cursor-pointer bg-slate-50 hover:bg-white transition-all text-[10px]"
                                    onClick={(e) => {
                                        const container = e.currentTarget.nextElementSibling;
                                        if (container) container.classList.toggle('hidden');
                                        // Focus input
                                        const input = container?.querySelector('input');
                                        if (input) setTimeout(() => input.focus(), 50);
                                    }}
                                >
                                    + Tag
                                </div>
                                <div className="hidden absolute top-0 left-0 z-50 bg-white border border-slate-200 shadow-lg p-2 rounded-lg flex gap-2 items-center min-w-[200px]">
                                    <input
                                        className="text-xs border-b border-slate-200 outline-none w-full pb-1 text-slate-700"
                                        placeholder="Nova tag..."
                                        onKeyDown={(e) => {
                                            if (e.key === 'Enter') {
                                                const val = e.currentTarget.value.trim();
                                                if (val) {
                                                    const currentTags = lead.tags || [];
                                                    if (!currentTags.includes(val)) {
                                                        handleUpdateLead({ tags: [...currentTags, val] });
                                                    }
                                                }
                                                e.currentTarget.value = '';
                                                e.currentTarget.parentElement?.classList.add('hidden');
                                            }
                                            if (e.key === 'Escape') {
                                                e.currentTarget.parentElement?.classList.add('hidden');
                                            }
                                        }}
                                        onBlur={(e) => {
                                            // Optional: close on blur or keep open? 
                                            // Let's close after short delay to allow clicks
                                            setTimeout(() => e.target.parentElement?.classList.add('hidden'), 200);
                                        }}
                                    />
                                    <Check className="w-3 h-3 text-emerald-500 cursor-pointer" />
                                </div>
                            </div>
                        </div>
                    </div>
