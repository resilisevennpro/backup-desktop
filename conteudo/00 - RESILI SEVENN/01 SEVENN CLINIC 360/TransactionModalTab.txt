import { useState, useEffect } from "react";
import { useAuth } from "@/contexts/AuthContext";
import { DataTable } from "@/components/ui/data-table";
import type { ColumnDef } from "@tanstack/react-table";
import { Badge } from "@/components/ui/badge";
import { Plus, Pencil, Trash2, Search, FileSpreadsheet, AlertTriangle } from "lucide-react";
import { Button } from "@/components/ui/button";
import { Card, CardContent, CardHeader, CardTitle, CardDescription } from "@/components/ui/card";
import { Input } from "@/components/ui/input";
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@/components/ui/select";
import { Tooltip, TooltipContent, TooltipProvider, TooltipTrigger } from "@/components/ui/tooltip";
import { AddTransactionModal } from "@/components/finance/AddTransactionModal";
import { Dialog, DialogContent, DialogDescription, DialogFooter, DialogHeader, DialogTitle } from "@/components/ui/dialog";
import * as XLSX from 'xlsx';
import { startOfMonth, endOfMonth, subDays, subMonths, format, startOfWeek, endOfWeek, parseISO } from 'date-fns';
import { ptBR } from 'date-fns/locale';
import { cn } from "@/lib/utils";

import { useFinanceStore } from "@/store/useFinanceStore";
import type { Transaction } from "@/components/finance/types";

export function TransactionsManagerTab() {
    const { clinic } = useAuth();
    const { transactions, fetchTransactions, deleteTransaction } = useFinanceStore();
    const [searchTerm, setSearchTerm] = useState("");
    const [periodFilter, setPeriodFilter] = useState("this_month");
    const [customStartDate, setCustomStartDate] = useState("");
    const [customEndDate, setCustomEndDate] = useState("");

    // Modal States
    const [isAddModalOpen, setIsAddModalOpen] = useState(false);
    const [editingTransaction, setEditingTransaction] = useState<Transaction | null>(null);
    const [deleteId, setDeleteId] = useState<string | null>(null);

    useEffect(() => {
        if (clinic?.id) {
            fetchTransactions(clinic.id);
        }
    }, [clinic?.id, fetchTransactions]);

    const handleDelete = (id: string) => {
        setDeleteId(id);
    };

    const confirmDelete = async () => {
        if (deleteId) {
            await deleteTransaction(deleteId);
            setDeleteId(null);
        }
    };

    const handleEdit = (transaction: Transaction) => {
        setEditingTransaction(transaction);
        setIsAddModalOpen(true);
    };

    const handleSuccess = () => {
        if (clinic?.id) fetchTransactions(clinic.id);
        setIsAddModalOpen(false);
        setEditingTransaction(null);
    };

    // --- FILTER LOGIC ---
    const getFilteredTransactions = () => {
        const now = new Date();
        let start = startOfMonth(now);
        let end = now; // Default to now for "until today"

        switch (periodFilter) {
            case 'today':
                start = now;
                end = now;
                break;
            case 'yesterday':
                start = subDays(now, 1);
                end = subDays(now, 1);
                break;
            case 'week': // This week
                start = startOfWeek(now, { weekStartsOn: 0 });
                end = endOfWeek(now);
                break;
            case 'last_week': // Last 7 days or Last Week? "Ultimos 7 dias".
                // User prompt example: "05 a 12 de Janeiro". 
                start = subDays(now, 7);
                end = now;
                break;
            case 'this_month':
                start = startOfMonth(now);
                end = now;
                break;
            case 'last_month':
                start = startOfMonth(subMonths(now, 1));
                end = endOfMonth(subMonths(now, 1));
                break;
            case 'last_3_months':
                start = subMonths(now, 3);
                end = now;
                break;
            case 'last_6_months':
                start = subMonths(now, 6);
                end = now;
                break;
            case 'last_365_days':
                start = subDays(now, 365);
                end = now;
                break;
            case 'custom':
                if (customStartDate) start = parseISO(customStartDate);
                if (customEndDate) end = parseISO(customEndDate);
                break;
            default:
                // custom or all? Default to this month if unknown
                break;
        }

        // Apply Date Filter + Search
        return transactions.filter(t => {
            const tDate = parseISO(t.transaction_date);
            // Check Date (ignoring time for pure date comparison usually, but isWithinInterval handles instances)
            // Adjust start to 00:00:00 and end to 23:59:59
            const s = new Date(start); s.setHours(0, 0, 0, 0);
            const e = new Date(end); e.setHours(23, 59, 59, 999);

            const matchesDate = tDate >= s && tDate <= e;

            const matchesSearch =
                (t.description?.toLowerCase() || "").includes(searchTerm.toLowerCase()) ||
                (t.patient?.toLowerCase() || "").includes(searchTerm.toLowerCase()) ||
                (t.procedure?.toLowerCase() || "").includes(searchTerm.toLowerCase()) ||
                (t.payment_method?.toLowerCase() || "").includes(searchTerm.toLowerCase());

            return matchesDate && matchesSearch;
        });
    };

    const filteredData = getFilteredTransactions();

    // --- EXPORT LOGIC ---
    const generatePeriodLabel = () => {
        const now = new Date();
        let label = "";

        const cap = (s: string) => s.charAt(0).toUpperCase() + s.slice(1);
        // FORCE LOCALE PT-BR
        const formatPt = (date: Date, pattern: string) => format(date, pattern, { locale: ptBR });

        if (periodFilter === 'this_month') {
            // "01_a_27Janeiro"
            label = `${formatPt(startOfMonth(now), 'dd')}_a_${formatPt(now, 'dd')}${cap(formatPt(now, 'MMMM'))}`;
        } else if (periodFilter === 'last_week' || periodFilter === 'week') {
            const start = subDays(now, 7);
            label = `${formatPt(start, 'dd')}_a_${formatPt(now, 'dd')}${cap(formatPt(now, 'MMMM'))}`;
        } else if (periodFilter === 'last_3_months') {
            // "01Dez-2025_a_01Mar-2026"
            const start = subMonths(now, 3);
            // Format: ddMMM-yyyy
            const sStr = `${formatPt(start, 'ddMMM-yyyy')}`;
            const eStr = `${formatPt(now, 'ddMMM-yyyy')}`;
            label = `${cap(sStr)}_a_${cap(eStr)}`;
        } else if (periodFilter === 'custom' && customStartDate && customEndDate) {
            label = `${formatPt(parseISO(customStartDate), 'ddMMMyy')}_a_${formatPt(parseISO(customEndDate), 'ddMMMyy')}`;
        } else {
            // Fallback
            label = `Relatorio_${periodFilter}_${formatPt(now, 'dd-MM-yyyy')}`;
        }

        return label.replace(/\s/g, '').replace(/\./g, ''); // Ensure no spaces or dots
    };

    const handleExport = () => {
        const periodLabel = generatePeriodLabel();

        // 1. Prepare Data with matching keys to desired Headers (Uppercase)
        const dataToExport = filteredData.map(t => ({
            "DATA": new Date(t.transaction_date).toLocaleDateString('pt-BR'),
            "PACIENTE": t.patient || "",
            "PROFISSIONAL": t.professional || "",
            "SERVIÇO": t.procedure || "",
            "VALOR": t.amount.toLocaleString('pt-BR', { style: 'currency', currency: 'BRL' }),
            "STATUS": t.status === 'paid' ? "PAGO" : "PENDENTE",
            "PAGAMENTO": t.payment_method || "",
            "OBSERVAÇÕES": t.notes || ""
        }));

        const wb = XLSX.utils.book_new();

        // 2. Create Sheet with Explicit Headers
        const header = ["DATA", "PACIENTE", "PROFISSIONAL", "SERVIÇO", "VALOR", "STATUS", "PAGAMENTO", "OBSERVAÇÕES"];
        const ws = XLSX.utils.json_to_sheet(dataToExport.length > 0 ? dataToExport : [], {
            header
        });

        // Auto-width columns
        const wscols = [
            { wch: 12 }, // DATA
            { wch: 30 }, // PACIENTE
            { wch: 20 }, // PROFISSIONAL
            { wch: 25 }, // SERVIÇO
            { wch: 15 }, // VALOR
            { wch: 12 }, // STATUS
            { wch: 15 }, // PAGAMENTO
            { wch: 40 }  // OBS
        ];
        ws['!cols'] = wscols;

        // 3. Append Sheet using periodLabel as Name
        XLSX.utils.book_append_sheet(wb, ws, periodLabel);

        const fileName = `Relatório_de_Vendas_${periodLabel}.xlsx`;

        XLSX.writeFile(wb, fileName);
    };

    const columns: ColumnDef<Transaction>[] = [
        {
            accessorKey: "transaction_date",
            header: "DATA",
            cell: ({ row }) => {
                try {
                    return (
                        <span className="text-sm text-zinc-600 font-medium whitespace-nowrap">
                            {new Date(row.original.transaction_date).toLocaleDateString('pt-BR')}
                        </span>
                    );
                } catch (e) {
                    return <span className="text-zinc-500 font-medium">{new Date(row.original.transaction_date).toLocaleDateString('pt-BR')}</span>;
                }
            }
        },
        {
            accessorKey: "patient",
            header: "PACIENTE",
            cell: ({ row }) => (
                <div className="flex items-center gap-2">
                    <span className="font-bold text-sm text-zinc-800">{row.original.patient || "N/A"}</span>
                    {row.original.is_edited && (
                        <Badge variant="outline" className="text-[10px] bg-amber-50 text-amber-600 border-amber-200 gap-1 px-1.5 py-0">
                            <AlertTriangle className="w-3 h-3" /> EDITADO
                        </Badge>
                    )}
                </div>
            )
        },
        {
            accessorKey: "professional",
            header: "PROFISSIONAL",
            cell: ({ row }) => (
                <Badge variant="secondary" className="font-normal bg-zinc-100 text-zinc-700 hover:bg-zinc-200 border-zinc-200">
                    {row.original.professional === 'unassigned' ? 'Sem Resp.' : row.original.professional}
                </Badge>
            )
        },
        {
            accessorKey: "procedure",
            header: "SERVIÇO",
            cell: ({ row }) => (
                <span className="text-sm text-zinc-600">{row.original.procedure || row.original.description}</span>
            )
        },
        {
            accessorKey: "payment_method",
            header: "PAGAMENTO",
            cell: ({ row }) => {
                const methodKey = row.original.payment_method || "other";

                const labels: Record<string, string> = {
                    pix: 'Pix',
                    money: 'Dinheiro',
                    credit: 'Crédito',
                    debit: 'Débito',
                    boleto: 'Boleto',
                    partnership: 'Parceria',
                    other: 'Outro'
                };

                const label = labels[methodKey] || methodKey;
                const installments = (methodKey === 'credit' || methodKey === 'debit') && row.original.installments && row.original.installments > 1
                    ? ` - ${row.original.installments}x`
                    : "";

                const getBadgeColors = (key: string) => {
                    switch (key) {
                        case 'pix': return "bg-emerald-100 text-emerald-700 border-emerald-200";
                        case 'money': return "bg-green-100 text-green-800 border-green-200";
                        case 'credit': return "bg-amber-100 text-amber-700 border-amber-200";
                        case 'debit': return "bg-blue-100 text-blue-700 border-blue-200";
                        case 'boleto': return "bg-indigo-100 text-indigo-700 border-indigo-200";
                        case 'partnership': return "bg-purple-100 text-purple-700 border-purple-200";
                        default: return "bg-zinc-100 text-zinc-600 border-zinc-200";
                    }
                };

                return (
                    <Badge variant="outline" className={cn("font-medium text-[11px] px-2 py-0 border", getBadgeColors(methodKey))}>
                        {label}{installments}
                    </Badge>
                );
            }
        },
        {
            accessorKey: "amount",
            header: "VALOR",
            cell: ({ row }) => {
                const amount = row.original.amount;
                const formatted = new Intl.NumberFormat("pt-BR", {
                    style: "currency",
                    currency: "BRL",
                }).format(amount)

                return (
                    <div className="font-bold text-sm text-zinc-700">
                        {formatted}
                    </div>
                )
            },
        },
        {
            accessorKey: "status",
            header: "STATUS",
            cell: ({ row }) => (
                <Badge variant="outline" className={`uppercase text-[10px] font-bold border-none px-2 py-0.5 ${row.original.status === 'paid' ? 'bg-emerald-100 text-emerald-700' : 'bg-amber-100 text-amber-700'}`}>
                    {row.original.status === 'paid' ? 'CONCLUÍDO' : 'PENDENTE'}
                </Badge>
            )
        },
        {
            id: "actions",
            header: "AÇÕES",
            cell: ({ row }) => {
                return (
                    <div className="flex items-center gap-1 justify-end">
                        <Button
                            variant="ghost"
                            size="icon"
                            className="h-8 w-8 text-blue-500 hover:text-blue-600 hover:bg-blue-50 rounded-full"
                            onClick={() => handleEdit(row.original)}
                        >
                            <Pencil className="h-4 w-4" />
                        </Button>
                        <Button
                            variant="ghost"
                            size="icon"
                            className="h-8 w-8 text-red-400 hover:text-red-600 hover:bg-red-50 rounded-full"
                            onClick={() => handleDelete(row.original.id)}
                        >
                            <Trash2 className="h-4 w-4" />
                        </Button>
                    </div>
                )
            },
        },
    ];

    return (
        <div className="space-y-6 animate-in slide-in-from-right-4 duration-500">

            <Card className="border-none shadow-sm bg-white">
                <CardHeader className="pb-4">
                    <div className="flex flex-col md:flex-row justify-between items-start md:items-center gap-4">
                        <div>
                            <CardTitle>Extrato Detalhado de Cobranças</CardTitle>
                            <CardDescription>Pesquise rapidamente filtrando por "Paciente, Procedimento, Forma de Pagamento ou Data"</CardDescription>
                        </div>
                        <div className="flex gap-2">
                            <Button variant="outline" className="gap-2 text-zinc-600" onClick={handleExport}>
                                <FileSpreadsheet className="h-4 w-4" /> Exportar
                            </Button>
                            <TooltipProvider>
                                <Tooltip>
                                    <TooltipTrigger asChild>
                                        <Button
                                            className="bg-[#59A52C] hover:bg-[#4d8f26] shadow-md gap-2 text-white font-semibold"
                                            onClick={() => setIsAddModalOpen(true)}
                                        >
                                            <Plus className="h-4 w-4" /> Nova Cobrança
                                        </Button>
                                    </TooltipTrigger>
                                    <TooltipContent className="bg-slate-800 text-white border-slate-700">
                                        <p>Gera pendência financeira. Ideal para orçamentos e pagamentos futuros.</p>
                                    </TooltipContent>
                                </Tooltip>
                            </TooltipProvider>
                        </div>
                    </div>
                </CardHeader>
                <CardContent>

                    {/* Toolbar de Filtros */}
                    <div className="flex flex-col md:flex-row gap-4 mb-6 p-4 bg-zinc-50/80 rounded-lg border border-zinc-100">
                        <div className="flex-1 relative">
                            <Search className="absolute left-3 top-2.5 h-4 w-4 text-muted-foreground" />
                            <Input
                                placeholder="Pesquise por Paciente, Procedimento, Pagamento..."
                                className="pl-9 bg-white border-zinc-200"
                                value={searchTerm}
                                onChange={(e) => setSearchTerm(e.target.value)}
                            />
                        </div>
                        <div className="flex gap-2 items-center">
                            <Select value={periodFilter} onValueChange={setPeriodFilter}>
                                <SelectTrigger className="w-[180px] bg-white border-zinc-200">
                                    <SelectValue placeholder="Período" />
                                </SelectTrigger>
                                <SelectContent>
                                    <SelectItem value="today">Hoje</SelectItem>
                                    <SelectItem value="yesterday">Ontem</SelectItem>
                                    <SelectItem value="week">Esta Semana</SelectItem>
                                    <SelectItem value="last_week">Últimos 7 Dias</SelectItem>
                                    <SelectItem value="this_month">Este Mês</SelectItem>
                                    <SelectItem value="last_month">Mês Passado</SelectItem>
                                    <SelectItem value="last_3_months">Últimos 3 Meses</SelectItem>
                                    <SelectItem value="last_6_months">Últimos 6 Meses</SelectItem>
                                    <SelectItem value="last_365_days">Últimos 365 Dias</SelectItem>
                                    <SelectItem value="custom">Personalizado</SelectItem>
                                </SelectContent>
                            </Select>

                            {periodFilter === 'custom' && (
                                <div className="flex gap-2 items-center animate-in fade-in slide-in-from-right-2">
                                    <Input
                                        type="date"
                                        className="w-[140px] h-10 bg-white border-zinc-200"
                                        value={customStartDate}
                                        onChange={(e) => setCustomStartDate(e.target.value)}
                                    />
                                    <span className="text-zinc-400">até</span>
                                    <Input
                                        type="date"
                                        className="w-[140px] h-10 bg-white border-zinc-200"
                                        value={customEndDate}
                                        onChange={(e) => setCustomEndDate(e.target.value)}
                                    />
                                </div>
                            )}

                            <Button variant="ghost" className="text-zinc-500 hover:text-zinc-700" onClick={() => {
                                setPeriodFilter("this_month");
                                setSearchTerm("");
                                setCustomStartDate("");
                                setCustomEndDate("");
                            }}>
                                Limpar
                            </Button>
                        </div>
                    </div>

                    {/* Counter */}
                    <div className="mb-2 flex items-center justify-between px-1">
                        <p className="text-xs font-bold text-zinc-500 uppercase tracking-wide">
                            Exibindo {filteredData.length} lançamentos neste período
                        </p>
                    </div>

                    {/* Tabela */}
                    <div className="rounded-xl border border-zinc-100 overflow-hidden">
                        <DataTable columns={columns} data={filteredData} />
                    </div>

                </CardContent>
            </Card>

            <AddTransactionModal
                isOpen={isAddModalOpen}
                onClose={() => {
                    setIsAddModalOpen(false);
                    setEditingTransaction(null); // Reset on close
                }}
                onSuccess={handleSuccess}
                transactionToEdit={editingTransaction}
            />

            {/* Custom Delete Modal */}
            <Dialog open={!!deleteId} onOpenChange={(open) => !open && setDeleteId(null)}>
                <DialogContent>
                    <DialogHeader>
                        <DialogTitle>Excluir Lançamento</DialogTitle>
                        <DialogDescription>
                            Tem certeza que deseja remover este registro financeiro? Esta ação não pode ser desfeita.
                        </DialogDescription>
                    </DialogHeader>
                    <DialogFooter>
                        <Button variant="outline" onClick={() => setDeleteId(null)}>Cancelar</Button>
                        <Button variant="destructive" onClick={confirmDelete} className="bg-red-600 hover:bg-red-700">
                            Sim, excluir
                        </Button>
                    </DialogFooter>
                </DialogContent>
            </Dialog>
        </div>
    );
}
