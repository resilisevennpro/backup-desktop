export interface Message {
    id: string;
    text: string;
    sender: 'user' | 'me' | 'ai' | 'system';
    timestamp: string;
    status: 'sent' | 'delivered' | 'read';
    metadata?: {
        type: 'media' | 'text' | string;
        url?: string;
        fileName?: string;
        fileType?: 'image' | 'video' | 'audio' | 'document' | string;
    };
}

    id: string;

    color: string;
    pipeline_id: string;
    position: number;
}

    id: string;
    clinic_id: string;
    titulo: string;
    conteudo: string;
    categoria: string;
    status: string;
}

interface ChatWindowProps {
    contact: Contact | null;
    messages: Message[];
    onSendMessage: (text: string) => void;
    isAiActive: boolean;
    onToggleAi: () => void;
    onResolveConversation?: () => void;
    onAssignConversation?: () => void;
    clinicId?: string;
    operatorName?: string | null;
}

export function ChatWindow({
    contact,
    messages,
    onSendMessage,
    isAiActive,
    onToggleAi,
    onResolveConversation,
    onAssignConversation,
    clinicId,
}: ChatWindowProps) {
    const { profile, clinic } = useAuth();
    const navigate = useNavigate();
    const messagesEndRef = useRef<HTMLDivElement>(null);

    // Verification: Data Isolation & Session
    // Security: Only use clinic ID if it looks like a valid UUID (not "1")
    const activeClinicId = (clinic?.id && clinic.id !== "1") ? clinic.id : (clinicId && clinicId !== "1" ? clinicId : null);

    // Search state
    const [searchOpen, setSearchOpen] = useState(false);
    const [searchQuery, setSearchQuery] = useState("");
    const [searchIndex, setSearchIndex] = useState(0);

    // Stage selector state
    const [stageOpen, setStageOpen] = useState(false);
    const [allStages, setAllStages] = useState<PipelineStage[]>([]);
    const [expandedPipe, setExpandedPipe] = useState<string | null>(null);

    // Drawer states
    const [prontuarioOpen, setProntuarioOpen] = useState(false);
    const [midiasOpen, setMidiasOpen] = useState(false);
    const [documentosOpen, setDocumentosOpen] = useState(false);

    // Prontuário data
    const [clinicalNotes, setClinicalNotes] = useState<ClinicalNote[]>([]);
    const [loadingNotes, setLoadingNotes] = useState(false);
    const [leadDetails, setLeadDetails] = useState<{ allergies?: string; complaint?: string } | null>(null);

    // Mídias data
    const [galleryItems, setGalleryItems] = useState<GalleryItem[]>([]);
    const [loadingGallery, setLoadingGallery] = useState(false);

    // Documentos data
    const [documents, setDocuments] = useState<ClinicDocument[]>([]);
    const [loadingDocs, setLoadingDocs] = useState(false);

    // Next appointment data (fetched live)
    const [nextAppointment, setNextAppointment] = useState<NextAppointmentData | null>(null);
    const [loadingAppointment, setLoadingAppointment] = useState(false);

    // Quick Replies (Slash Command)
    const [replies, setReplies] = useState<QuickReply[]>([]);
    const [slashOpen, setSlashOpen] = useState(false);
    const [slashQuery, setSlashQuery] = useState("");
    const [inputValue, setInputValue] = useState("");

    // Support Media Modal
    const [isMediaModalOpen, setIsMediaModalOpen] = useState(false);
    const [supportMedia, setSupportMedia] = useState<SupportMedia[]>([]);
    const [supportFolders, setSupportFolders] = useState<SupportFolder[]>([]);
    const [selectedSupportFolder, setSelectedSupportFolder] = useState<SupportFolder | null>(null);
    const [selectedMedia, setSelectedMedia] = useState<SupportMedia[]>([]);
    const [loadingMedia, setLoadingMedia] = useState(false);
    const [mediaSearchQuery, setMediaSearchQuery] = useState("");
    const [mediaTypeFilter, setMediaTypeFilter] = useState("all");
    const [mediaLayout, setMediaLayout] = useState<'grid' | 'list'>('grid');
    const [isConfirmingMedia, setIsConfirmingMedia] = useState(false);
    const [sendingMedia, setSendingMedia] = useState(false);

    // Emoji and File Attachments
    const [emojiPickerOpen, setEmojiPickerOpen] = useState(false);
    const [selectedFiles, setSelectedFiles] = useState<File[]>([]);
    const [uploadingFiles, setUploadingFiles] = useState(false);
    const fileInputRef = useRef<HTMLInputElement>(null);
    const textInputRef = useRef<HTMLInputElement>(null);

    // Audio Recording
    const [isRecording, setIsRecording] = useState(false);
    const [isPaused, setIsPaused] = useState(false);
    const [isPreviewMode, setIsPreviewMode] = useState(false);
    const [recordingTime, setRecordingTime] = useState(0);
    const [audioPreviewUrl, setAudioPreviewUrl] = useState<string | null>(null);
    const [recordedBlob, setRecordedBlob] = useState<Blob | null>(null);
    const mediaRecorderRef = useRef<MediaRecorder | null>(null);
    const audioChunksRef = useRef<Blob[]>([]);
    const timerRef = useRef<number | null>(null);
    const audioStreamRef = useRef<MediaStream | null>(null);
    const isRecordingCanceled = useRef(false);
    const isStopForPreview = useRef(false);

    // Check if any drawer is open (for dynamic layout)
    // const anyDrawerOpen = prontuarioOpen || midiasOpen || documentosOpen;

    const sentimentIcons = useMemo(() => ({
        interested: { icon: Heart, color: 'text-green-500', label: 'Interessado' },
        doubting: { icon: HelpCircle, color: 'text-yellow-500', label: 'Dúvida' },
        angry: { icon: AlertCircle, color: 'text-red-500', label: 'Insatisfeito' },
        booking: { icon: CalendarCheck, color: 'text-blue-500', label: 'Agendamento' },
        neutral: { icon: MessageSquare, color: 'text-zinc-400', label: 'Neutro' },
        question: { icon: Search, color: 'text-purple-400', label: 'Sondagem' }
    }), []);

    // =========================================================================
    // FETCH: Pipeline Stages
    // =========================================================================
    useEffect(() => {
        if (!activeClinicId) return;
        const fetchStages = async () => {
            const { data, error } = await supabase
                .from('pipeline_stages')
                .select('id, name, color, pipeline_id, position')
                .eq('clinic_id', activeClinicId)
                .eq('is_active', true)
                .order('position', { ascending: true });
            if (!error && data) setAllStages(data);
        };
        fetchStages();
    }, [activeClinicId]);

    // =========================================================================
    // FETCH: Next Appointment for this lead
    // =========================================================================
    useEffect(() => {
        if (contact?.id) {
            setNextAppointment(null);
            return;
        }
    }, [contact?.id]);

    // =========================================================================
    // KEYBOARD SHORTCUTS
    // =========================================================================
    useEffect(() => {
        const handleKeyDown = (e: KeyboardEvent) => {
            // Space: Toggle Record/Pause
            if (e.code === 'Space') {
                const target = e.target as HTMLElement;
                const isInput = target.tagName === 'INPUT' || target.tagName === 'TEXTAREA' || target.isContentEditable;

                if (isRecording && !isInput) {
                    e.preventDefault();
                    if (isPaused) resumeRecording();
                    else pauseRecording();
                }
            }

            // Enter: Send Audio in Preview Mode
            if (e.code === 'Enter' && isPreviewMode && recordedBlob) {
                e.preventDefault();
                handleSendAudioPreview();
            }
        };

        window.addEventListener('keydown', handleKeyDown);
        return () => window.removeEventListener('keydown', handleKeyDown);
    }, [isRecording, isPaused, isPreviewMode, recordedBlob]);

    // =========================================================================
    // FETCH: Next Appointment for this lead
    // =========================================================================
    const fetchProntuarioData = useCallback(async () => {
        if (!contact?.id || contact.id === '1') return;
        setLoadingNotes(true);
        try {
            const { data: notes, error: notesError } = await supabase
                .from('clinical_records')
                .select('*, professional:profiles!professional_id(name:full_name)')
                .eq('lead_id', contact.id)
                .order('created_at', { ascending: false })
                .limit(10);

            if (!notesError && notes) setClinicalNotes(notes as any);

            const { data: leadData } = await supabase
                .from('leads')
                .select('custom_fields')
                .eq('id', contact.id)
                .single();

            if (leadData?.custom_fields) {
                const cf = leadData.custom_fields as any;
                setLeadDetails({
                    allergies: cf.allergies || '',
                    complaint: cf.complaint || '',
                });
            } else {
                setLeadDetails(null);
            }
        } catch (err) {
            console.error("Fetch prontuario error:", err);
        } finally {
            setLoadingNotes(false);
        }
    }, [contact?.id]);

    useEffect(() => {
        if (prontuarioOpen) fetchProntuarioData();
    }, [prontuarioOpen, fetchProntuarioData]);

    // =========================================================================
    // FETCH: Mídias (lead_gallery) when drawer opens
    // =========================================================================
    const fetchGalleryData = useCallback(async () => {
        if (!contact?.id || contact.id === '1') return;
        setLoadingGallery(true);
        try {
            const { data, error } = await supabase
                .from('lead_gallery')
                .select('id, url, thumbnail, category, created_at, description')
                .eq('lead_id', contact.id)
                .order('created_at', { ascending: false })
                .limit(30);

            if (!error && data) setGalleryItems(data);
            else setGalleryItems([]);
        } catch (err) {
            console.error("Fetch gallery error:", err);
            setGalleryItems([]);
        } finally {
            setLoadingGallery(false);
        }
    }, [contact?.id]);

    useEffect(() => {
        if (midiasOpen) fetchGalleryData();
    }, [midiasOpen, fetchGalleryData]);

    // =========================================================================
    // FETCH: Documentos (clinic_documents) when drawer opens
    // =========================================================================
    const fetchDocumentsData = useCallback(async () => {
        if (!contact?.id || contact.id === '1' || !activeClinicId) return;
        setLoadingDocs(true);
        try {
            const { data, error } = await supabase
                .from('clinic_documents')
                .select('id, title, created_at, status, content')
                .eq('clinic_id', activeClinicId)
                .eq('lead_id', contact.id)
                .order('created_at', { ascending: false })
                .limit(20);

            if (!error && data) setDocuments(data);
            else setDocuments([]);
        } catch (err) {
            console.error("Fetch documents error:", err);
            setDocuments([]);
        } finally {
            setLoadingDocs(false);
        }
    }, [contact?.id, activeClinicId]);

    useEffect(() => {
        if (!documentosOpen) return;
        fetchDocumentsData();
    }, [documentosOpen, fetchDocumentsData]);

    // =========================================================================
    // FETCH: Quick Replies
    // =========================================================================
    useEffect(() => {
        if (!activeClinicId) return;
        const fetchReplies = async () => {
            const { data, error } = await supabase
                .from('respostas_rapidas')
                .select('*')
                .eq('clinic_id', activeClinicId)
                .eq('is_active', true);
            if (!error && data) setReplies(data);
        };
        fetchReplies();
    }, [activeClinicId]);

    // =========================================================================
    // FETCH: Support Media
    // =========================================================================
    useEffect(() => {
        if (!isMediaModalOpen || !activeClinicId || activeClinicId === '1') return;

        const fetchSupportMedia = async () => {
            setLoadingMedia(true);
            try {
                const [mediaRes, foldersRes] = await Promise.all([
                    supabase
                        .from('apoio_midias')
                        .select('*')
                        .eq('clinic_id', activeClinicId)
                        .eq('is_active', true)
                        .is('deleted_at', null)
                        .order('created_at', { ascending: false }),
                    supabase
                        .from('apoio_pastas')
                        .select('*')
                        .eq('clinic_id', activeClinicId)
                        .eq('is_active', true)
                        .is('deleted_at', null)
                        .order('is_fixed', { ascending: false })
                        .order('nome', { ascending: true })
                ]);

                if (mediaRes.error) throw mediaRes.error;
                if (foldersRes.error) throw foldersRes.error;

                setSupportMedia(mediaRes.data || []);
                setSupportFolders(foldersRes.data || []);
            } catch (error) {
                console.error("Error fetching support media:", error);
                toast.error("Erro ao carregar biblioteca de mídias.");
            } finally {
                setLoadingMedia(false);
            }
        };

        fetchSupportMedia();
    }, [isMediaModalOpen, activeClinicId]);

    // =========================================================================
    const handleSendMedia = async (files: SupportMedia[], isApoio: boolean) => {
        if (!contact || !activeClinicId) return;

        // Security: Ensure operator is logged in
        if (!profile?.id) {
            toast.error("Sessão expirada. Por favor, faça login novamente.");
            return;
        }

        setSendingMedia(true);
        try {
            // Disparar o envio para cada arquivo
            for (const file of files) {
                let mediaUrl = file.url;

                // Lógica de Segurança (LGPD) - Refresh Signed URL on Send
                if (isPrivateBucket(mediaUrl)) {
                    console.log("Preparing fresh 24h signed URL for private media:", file.nome);
                    const info = parseSupabaseUrl(mediaUrl);
                    if (info) {
                        try {
                            const { url } = await getSecureUrl(info.bucket, info.path);
                            mediaUrl = url;
                        } catch (e) {
                            console.error("Failed to sign URL for dispatch:", e);
                            // Fallback to original URL, but it might fail on delivery
                        }
                    }
                }

                // Disparo via Evolution API (Simulado)
                console.log(`[${CURRENT_PROVIDER}] Enviando mídia para ${contact.name}:`, mediaUrl);

                // Registra na tabela de mensagens do Supabase para manter histórico
                await supabase.from('messages').insert({
                    clinic_id: activeClinicId,
                    lead_id: contact.id,
                    content: `[Mídia: ${file.nome}]`,
                    metadata: {
                        type: 'media',
                        url: mediaUrl,
                        fileName: file.nome,
                        fileType: file.tipo_arquivo,
                        provider: CURRENT_PROVIDER
                    },
                    direction: 'clinica',
                    operator_id: profile.id
                });
            }

            toast.success(`${files.length} mídia(s) enviada(s) com sucesso!`);
            setIsMediaModalOpen(false);
            setIsConfirmingMedia(false);
            setSelectedMedia([]);
        } catch (error) {
            console.error("Error sending media:", error);
            toast.error("Falha ao enviar mídias. Tente novamente.");
        } finally {
            setSendingMedia(false);
        }
    };

    const filteredSupportMedia = useMemo(() => {
        return supportMedia.filter(item => {
            if (selectedSupportFolder && item.folder_id !== selectedSupportFolder.id) return false;
            if (!selectedSupportFolder) return false;

            const matchesSearch = item.nome.toLowerCase().includes(mediaSearchQuery.toLowerCase());
            const matchesType = mediaTypeFilter === 'all' || item.tipo_arquivo === mediaTypeFilter;
            return matchesSearch && matchesType;
        });
    }, [supportMedia, mediaSearchQuery, mediaTypeFilter, selectedSupportFolder]);

    const filteredSupportFolders = useMemo(() => {
        // Deduplication: keep only first is_fixed folder per tipo
        const seen = new Set<string>();
        const deduped = supportFolders.filter(folder => {
            if (folder.is_fixed) {
                const key = `fixed-${folder.tipo}`;
                if (seen.has(key)) return false;
                seen.add(key);
            }
            return true;
        });
        if (selectedSupportFolder) return [];
        return deduped.filter(folder =>
            folder.nome.toLowerCase().includes(mediaSearchQuery.toLowerCase())
        );
    }, [supportFolders, mediaSearchQuery, selectedSupportFolder]);

    // Media count per folder for the folder grid badges
    const mediaCountByFolder = useMemo(() => {
        const map: Record<string, number> = {};
        supportMedia.forEach(item => {
            map[item.folder_id] = (map[item.folder_id] || 0) + 1;
        });
        return map;
    }, [supportMedia]);

    // Helper: detect if a media URL comes from a private bucket
    const isPrivateBucket = (url: string) => {
        if (!url) return false;
        return (
            url.includes('/lead-images/') ||
            url.includes('/patient-photos/') ||
            url.includes('/clinic-documents/')
        );
    };

    const toggleMediaSelection = (file: SupportMedia) => {
        setSelectedMedia(prev =>
            prev.find(f => f.id === file.id)
                ? prev.filter(f => f.id !== file.id)
                : [...prev, file]
        );
    };

    const handleMediaSearch = (e: React.ChangeEvent<HTMLInputElement>) => {
        setMediaSearchQuery(e.target.value);
    };

    const handleConfirmMediaSend = () => {
        if (selectedMedia.length > 0) {
            setIsConfirmingMedia(true);
        }
    };

    const resetMediaModal = () => {
        setIsMediaModalOpen(false);
        setIsConfirmingMedia(false);
        setSelectedMedia([]);
        setMediaSearchQuery("");
        setMediaTypeFilter("all");
        setSelectedSupportFolder(null);
    };

    const handleFolderBack = () => {
        setSelectedSupportFolder(null);
        setMediaSearchQuery("");
        setMediaTypeFilter("all");
    };

    // =========================================================================
    // COMPUTED
    // =========================================================================
    const groupedPipelines = useMemo(() => {
        const grouped: { id: string; label: string; stages: PipelineStage[] }[] = [];
        for (const pipe of PIPELINES_META) {
            const pipeStages = allStages.filter(s => s.pipeline_id === pipe.id);
            if (pipeStages.length > 0) {
                grouped.push({ id: pipe.id, label: pipe.label, stages: pipeStages });
            }
        }
        return grouped;
    }, [allStages]);

    // Find which pipeline the current contact's stage belongs to
    const currentPipelineId = useMemo(() => {
        if (!contact?.stage) return null;
        const matchingStage = allStages.find(s => s.name === contact.stage);
        return matchingStage?.pipeline_id || null;
    }, [contact?.stage, allStages]);

    // Auto-expand the lead's current pipeline when popover opens
    useEffect(() => {
        if (stageOpen && currentPipelineId) {
            setExpandedPipe(currentPipelineId);
        }
    }, [stageOpen, currentPipelineId]);

    // =========================================================================
    // SEARCH: Filter + highlight + navigate results
    // =========================================================================
    const searchResults = useMemo(() => {
        if (!searchQuery.trim()) return [];
        const q = searchQuery.toLowerCase();
        return messages
            .map((m, idx) => ({ msg: m, originalIndex: idx }))
            .filter(({ msg }) => msg.text.toLowerCase().includes(q));
    }, [messages, searchQuery]);

    const filteredMessages = useMemo(() => {
        return messages;
    }, [messages]);

    const currentSearchResultId = searchResults.length > 0 ? searchResults[searchIndex]?.msg.id : null;

    useEffect(() => {
        if (currentSearchResultId) {
            const el = document.getElementById(`msg-${currentSearchResultId}`);
            el?.scrollIntoView({ behavior: 'smooth', block: 'center' });
        }
    }, [currentSearchResultId]);

    const handleSearchPrev = () => {
        setSearchIndex(prev => (prev > 0 ? prev - 1 : searchResults.length - 1));
    };
    const handleSearchNext = () => {
        setSearchIndex(prev => (prev < searchResults.length - 1 ? prev + 1 : 0));
    };

    useEffect(() => {
        setSearchIndex(0);
    }, [searchQuery]);

    const highlightText = (text: string, query: string) => {
        if (!query.trim()) return text;
        const regex = new RegExp(`(${query.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')})`, 'gi');
        const parts = text.split(regex);
        return parts.map((part, i) =>
            regex.test(part)
                ? <mark key={i} className="bg-yellow-300/80 text-slate-900 rounded-sm px-0.5">{part}</mark>
                : part
        );
    };

    // =========================================================================
    // AUTO-SCROLL & SEND
    // =========================================================================
    const scrollToBottom = () => {
        messagesEndRef.current?.scrollIntoView({ behavior: "smooth" });
    };

    useEffect(() => {
        if (!searchQuery.trim()) scrollToBottom();
    }, [messages, searchQuery]);

    const handleSend = async () => {
        const text = inputValue.trim();
        const hasFiles = selectedFiles.length > 0;

        if (text || hasFiles) {
            // Security Check
            if (!profile?.id || !activeClinicId) {
                toast.error("Sessão expirada. Por favor, faça login novamente.");
                return;
            }

            try {
                // If there are files, upload them first
                if (hasFiles) {
                    setUploadingFiles(true);
                    const uploadedSupportMedia: SupportMedia[] = [];

                    for (const file of selectedFiles) {
                        const fileExt = file.name.split('.').pop();
                        const fileName = `${crypto.randomUUID()}.${fileExt}`;
                        const filePath = `${activeClinicId}/${contact?.id}/${fileName}`;

                        const { error: uploadError } = await supabase.storage
                            .from('lead-images')
                            .upload(filePath, file);

                        if (uploadError) throw uploadError;

                        const { url } = await getSecureUrl('lead-images', filePath);

                if (e.data.size > 0) audioChunksRef.current.push(e.data);

                                                className="flex items-center gap-3 p-3 rounded-lg border border-slate-200 bg-white hover:shadow-sm transition-shadow cursor-pointer group"
                                            >
                                                <div className="w-9 h-9 rounded-lg bg-slate-100 flex items-center justify-center shrink-0 group-hover:bg-primary/10 transition-colors">
                                                    <FileText className="w-4 h-4 text-slate-500 group-hover:text-primary" />
                                                </div>
                                                <div className="flex-1 min-w-0">
                                                    <p className="text-xs font-semibold text-slate-900 truncate">{doc.title}</p>
                                                    <div className="flex items-center gap-2 mt-0.5">
                                                        <Badge className={cn("text-[8px] font-bold h-4 px-1.5 border-none shadow-none", status.color)}>
                                                            {status.label}
                                                        </Badge>
                                                        <span className="text-[9px] text-slate-400">
                                                            {new Date(doc.created_at).toLocaleDateString('pt-BR')}
                                                        </span>
                                                    </div>
                                                </div>
                                                <Download className="w-3.5 h-3.5 text-slate-400 opacity-0 group-hover:opacity-100 transition-opacity" />
                                            </div>
                                        );
                                    })}
                                </div>
                            )}
                        </div>

                        <div className="border-t p-3 shrink-0">
                            <Button
                                variant="outline"
                                className="w-full gap-2 text-xs font-semibold"
                                onClick={() => { setDocumentosOpen(false); navigate(buildExpandLink('documentos')); }}
                            >
                                <ExternalLink className="w-3.5 h-3.5" />
                                Ver Documentos Completos
                            </Button>
                        </div>
                    </div>
                )}

                {/* ========================= MODALS ========================= */}

                {/* Support Media Modal */}
                <Dialog open={isMediaModalOpen} onOpenChange={(open) => !open && resetMediaModal()}>
                    <DialogContent className="max-w-3xl h-[80vh] flex flex-col p-0 overflow-hidden border-primary/10">
                        {!isConfirmingMedia ? (
                            <>
                                {/* Header */}
                                <DialogHeader className="p-5 pb-3 border-b bg-white shrink-0">
                                    <div className="flex items-center justify-between">
                                        <div className="flex items-center gap-2 min-w-0">
                                            <DialogTitle className="text-lg font-bold font-space flex items-center gap-2">
                                                {selectedSupportFolder && (
                                                    <button
                                                        onClick={handleFolderBack}
                                                        className="p-1.5 -ml-1 rounded-lg hover:bg-slate-100 text-slate-500 hover:text-slate-800 transition-colors"
                                                    >
                                                        <ArrowLeft className="w-4 h-4" />
                                                    </button>
                                                )}
                                                <div className={cn(
                                                    "p-1.5 rounded-lg",
                                                    selectedSupportFolder
                                                        ? selectedSupportFolder.tipo === 'institucional' ? "bg-emerald-50" : selectedSupportFolder.tipo === 'servico' ? "bg-blue-50" : "bg-slate-50"
                                                        : "bg-emerald-50"
                                                )}>
                                                    <Folder className={cn(
                                                        "w-4 h-4",
                                                        selectedSupportFolder
                                                            ? selectedSupportFolder.tipo === 'institucional' ? "text-emerald-600" : selectedSupportFolder.tipo === 'servico' ? "text-blue-600" : "text-slate-500"
                                                            : "text-emerald-600"
                                                    )} />
                                                </div>
                                                <span className="truncate">
                                                    {selectedSupportFolder ? selectedSupportFolder.nome : 'Biblioteca de Mídias'}
                                                </span>
                                            </DialogTitle>
                                            <DialogDescription className="sr-only">
                                                {selectedSupportFolder ? 'Selecione mídias para enviar' : 'Escolha uma pasta'}
                                            </DialogDescription>
                                        </div>
                                        <div className="flex items-center gap-2 shrink-0">
                                            <div className="relative w-56">
                                                <Search className="absolute left-3 top-2.5 h-3.5 w-3.5 text-muted-foreground" />
                                                <input
                                                    className="w-full h-9 pl-9 pr-4 rounded-full border border-zinc-200 text-sm focus:outline-none focus:ring-2 focus:ring-primary/20 transition-shadow"
                                                    placeholder={selectedSupportFolder ? 'Buscar arquivo...' : 'Buscar pasta...'}
                                                    value={mediaSearchQuery}
                                                    onChange={handleMediaSearch}
                                                />
                                            </div>
                                        </div>
                                    </div>

                                    {/* Type Filters — only shown inside a folder */}
                                    {selectedSupportFolder && (
                                        <div className="flex items-center justify-between mt-3 pt-2">
                                            <Tabs value={mediaTypeFilter} onValueChange={setMediaTypeFilter} className="flex-1">
                                                <TabsList className="bg-muted/50 p-1 h-9 rounded-full">
                                                    {MEDIA_TABS.map(tab => (
                                                        <TabsTrigger
                                                            key={tab.value}
                                                            value={tab.value}
                                                            className="rounded-full px-3.5 text-[11px] font-bold uppercase tracking-wider data-[state=active]:shadow-sm"
                                                        >
                                                            <tab.icon className="w-3.5 h-3.5 mr-1.5 hidden sm:inline" />
                                                            {tab.label}
                                                        </TabsTrigger>
                                                    ))}
                                                </TabsList>
                                            </Tabs>

                                            <div className="flex items-center bg-muted rounded-lg p-1 shrink-0 ml-3">
                                                <Button
                                                    variant={mediaLayout === 'grid' ? 'secondary' : 'ghost'}
                                                    size="icon"
                                                    className="h-7 w-7 rounded-md"
                                                    onClick={() => setMediaLayout('grid')}
                                                >
                                                    <LayoutGrid className="h-3.5 w-3.5" />
                                                </Button>
                                                <Button
                                                    variant={mediaLayout === 'list' ? 'secondary' : 'ghost'}
                                                    size="icon"
                                                    className="h-7 w-7 rounded-md"
                                                    onClick={() => setMediaLayout('list')}
                                                >
                                                    <List className="h-3.5 w-3.5" />
                                                </Button>
                                            </div>
                                        </div>
                                    )}
                                </DialogHeader>

                                {/* Content Area */}
                                <div className="flex-1 overflow-y-auto p-5 scrollbar-hide">
                                    {loadingMedia ? (
                                        <div className="flex flex-col items-center justify-center py-20 text-muted-foreground">
                                            <Loader2 className="w-8 h-8 animate-spin mb-3 text-primary/40" />
                                            <p className="text-xs font-medium animate-pulse">Carregando...</p>
                                        </div>
                                    ) : !selectedSupportFolder ? (
                                        /* ===== LEVEL 1: Folder Grid ===== */
                                        <div className="animate-in fade-in slide-in-from-left-2 duration-200">
                                            <div className="grid grid-cols-2 md:grid-cols-3 lg:grid-cols-4 gap-3">
                                                {filteredSupportFolders.map(folder => {
                                                    const count = mediaCountByFolder[folder.id] || 0;
                                                    const typeColor = folder.tipo === 'institucional'
                                                        ? { bg: 'bg-emerald-50', text: 'text-emerald-600', hoverBg: 'hover:border-emerald-200', badge: 'bg-emerald-100 text-emerald-700' }
                                                        : folder.tipo === 'servico'
                                                            ? { bg: 'bg-blue-50', text: 'text-blue-600', hoverBg: 'hover:border-blue-200', badge: 'bg-blue-100 text-blue-700' }
                                                            : { bg: 'bg-slate-50', text: 'text-slate-500', hoverBg: 'hover:border-slate-300', badge: 'bg-slate-100 text-slate-600' };
                                                    return (
                                                        <div
                                                            key={folder.id}
                                                            onClick={() => { setSelectedSupportFolder(folder); setMediaSearchQuery(''); setMediaTypeFilter('all'); }}
                                                            className={cn(
                                                                "group relative border rounded-xl p-4 flex flex-col items-center gap-2.5 transition-all cursor-pointer bg-white",
                                                                "hover:shadow-md", typeColor.hoverBg
                                                            )}
                                                        >
                                                            {/* Folder Icon */}
                                                            <div className={cn("p-3.5 rounded-2xl transition-colors", typeColor.bg)}>
                                                                <Folder className={cn("w-7 h-7", typeColor.text)} />
                                                            </div>

                                                            {/* Folder Name */}
                                                            <div className="text-center w-full">
                                                                <p className="text-sm font-bold text-slate-800 truncate px-1">{folder.nome}</p>
                                                                {folder.descricao && (
                                                                    <p className="text-[10px] text-muted-foreground truncate px-1 mt-0.5">{folder.descricao}</p>
                                                                )}
                                                            </div>

                                                            {/* Footer: tipo badge + count */}
                                                            <div className="flex items-center gap-1.5">
                                                                <span className={cn("text-[9px] px-2 py-0.5 rounded-full font-bold uppercase tracking-wider", typeColor.badge)}>
                                                                    {folder.tipo}
                                                                </span>
                                                                {count > 0 && (
                                                                    <span className="text-[9px] px-1.5 py-0.5 rounded-full bg-slate-100 text-slate-500 font-bold">
                                                                        {count}
                                                                    </span>
                                                                )}
                                                            </div>
                                                        </div>
                                                    );
                                                })}
                                            </div>

                                            {filteredSupportFolders.length === 0 && (
                                                <div className="py-16 text-center text-muted-foreground opacity-50">
                                                    <FolderSearch className="w-12 h-12 mx-auto mb-3" />
                                                    <p className="font-medium">Nenhuma pasta encontrada.</p>
                                                    {mediaSearchQuery && <p className="text-xs mt-1">Tente outro termo de busca.</p>}
                                                </div>
                                            )}
                                        </div>
                                    ) : filteredSupportMedia.length === 0 ? (
                                        /* ===== LEVEL 2: Empty State ===== */
                                        <div className="flex flex-col items-center justify-center py-20 text-muted-foreground opacity-60 animate-in fade-in duration-200">
                                            <div className="relative mb-4">
                                                <FolderSearch className="w-16 h-16 text-slate-200" />
                                                <div className="absolute -bottom-1 -right-1 bg-white rounded-full p-1 border shadow-sm">
                                                    <Search className="w-4 h-4 text-primary" />
                                                </div>
                                            </div>
                                            <p className="text-sm font-bold text-slate-600">
                                                {mediaTypeFilter === 'all'
                                                    ? 'Nenhuma mídia nesta pasta'
                                                    : `Nenhum(a) ${MEDIA_TABS.find(t => t.value === mediaTypeFilter)?.label.toLowerCase()} encontrado(a)`
                                                }
                                            </p>
                                            <p className="text-xs mt-1">Tente outros termos ou mude o filtro.</p>
                                        </div>
                                    ) : mediaLayout === 'grid' ? (
                                        /* ===== LEVEL 2: Grid View ===== */
                                        <div className="grid grid-cols-2 md:grid-cols-4 lg:grid-cols-5 gap-3 animate-in fade-in slide-in-from-right-2 duration-200">
                                            {filteredSupportMedia.map((item) => {
                                                const isSelected = selectedMedia.some(f => f.id === item.id);
                                                const isPrivate = isPrivateBucket(item.url);
                                                return (
                                                    <div
                                                        key={item.id}
                                                        onClick={() => !isPrivate && toggleMediaSelection(item)}
                                                        className={cn(
                                                            "group relative bg-white border rounded-xl overflow-hidden transition-all",
                                                            isPrivate
                                                                ? "opacity-50 cursor-not-allowed"
                                                                : "cursor-pointer hover:shadow-md",
                                                            isSelected ? "ring-2 ring-primary border-transparent shadow-lg" : "border-slate-200"
                                                        )}
                                                    >
                                                        <div className="aspect-square bg-slate-50 flex items-center justify-center relative overflow-hidden">
                                                            {isPrivate ? (
                                                                <div className="flex flex-col items-center gap-1.5 p-3 text-center">
                                                                    <Loader2 className="w-6 h-6 text-slate-300 animate-spin" />
                                                                    <span className="text-[9px] text-slate-400 font-medium">Bucket privado</span>
                                                                </div>
                                                            ) : item.tipo_arquivo === 'image' && (item.thumbnail_url || item.url) ? (
                                                                <img src={item.thumbnail_url || item.url} alt={item.nome} className="w-full h-full object-cover" crossOrigin="anonymous" />
                                                            ) : (
                                                                <div className="flex flex-col items-center gap-1 p-2">
                                                                    {item.tipo_arquivo === 'video' && <Film className="w-8 h-8 text-purple-400" />}
                                                                    {item.tipo_arquivo === 'audio' && <Music className="w-8 h-8 text-amber-400" />}
                                                                    {(item.tipo_arquivo === 'document' || !['image', 'video', 'audio'].includes(item.tipo_arquivo)) && <FileText className="w-8 h-8 text-slate-400" />}
                                                                    <span className="text-[10px] uppercase font-bold text-slate-400 tracking-wider">{item.tipo_arquivo}</span>
                                                                </div>
                                                            )}

                                                            {isSelected && (
                                                                <div className="absolute inset-0 bg-primary/20 flex items-start justify-end p-2">
                                                                    <div className="h-5 w-5 bg-white rounded-full flex items-center justify-center shadow-sm">
                                                                        <Check className="w-3.5 h-3.5 text-primary" />
                                                                    </div>
                                                                </div>
                                                            )}
                                                        </div>
                                                        <div className="p-2 border-t">
                                                            <p className="text-[11px] font-bold text-slate-700 truncate">{item.nome}</p>
                                                            <span className="text-[9px] text-muted-foreground uppercase font-semibold">{item.tipo_arquivo}</span>
                                                        </div>
                                                    </div>
                                                );
                                            })}
                                        </div>
                                    ) : (
                                        /* ===== LEVEL 2: List View ===== */
                                        <div className="space-y-2 animate-in fade-in slide-in-from-right-2 duration-200">
                                            {filteredSupportMedia.map((item) => {
                                                const isSelected = selectedMedia.some(f => f.id === item.id);
                                                const isPrivate = isPrivateBucket(item.url);
                                                return (
                                                    <div
                                                        key={item.id}
                                                        onClick={() => !isPrivate && toggleMediaSelection(item)}
                                                        className={cn(
                                                            "group flex items-center justify-between p-3 rounded-xl border bg-white transition-all",
                                                            isPrivate
                                                                ? "opacity-50 cursor-not-allowed"
                                                                : "cursor-pointer hover:shadow-sm",
                                                            isSelected ? "border-primary bg-primary/5 ring-1 ring-primary" : "border-slate-200"
                                                        )}
                                                    >
                                                        <div className="flex items-center gap-4 min-w-0">
                                                            <div className="h-12 w-12 bg-slate-100 rounded-lg flex items-center justify-center shrink-0 overflow-hidden">
                                                                {isPrivate ? (
                                                                    <Loader2 className="w-5 h-5 text-slate-300 animate-spin" />
                                                                ) : item.tipo_arquivo === 'image' && (item.thumbnail_url || item.url) ? (
                                                                    <img src={item.thumbnail_url || item.url} alt={item.nome} className="w-full h-full object-cover" crossOrigin="anonymous" />
                                                                ) : (
                                                                    <>
                                                                        {item.tipo_arquivo === 'video' && <Film className="w-5 h-5 text-purple-400" />}
                                                                        {item.tipo_arquivo === 'audio' && <Music className="w-5 h-5 text-amber-400" />}
                                                                        {(item.tipo_arquivo === 'document' || !['image', 'video', 'audio'].includes(item.tipo_arquivo)) && <FileText className="w-5 h-5 text-slate-400" />}
                                                                    </>
                                                                )}
                                                            </div>
                                                            <div className="min-w-0">
                                                                <p className="text-sm font-bold text-slate-800 truncate">{item.nome}</p>
                                                                <div className="flex items-center gap-2 mt-0.5">
                                                                    <Badge variant="outline" className="text-[9px] h-4 uppercase">{item.tipo_arquivo}</Badge>
                                                                    {isPrivate && <span className="text-[9px] text-amber-500 font-medium">Link privado</span>}
                                                                    {item.created_at && <span className="text-[10px] text-slate-400">{new Date(item.created_at).toLocaleDateString('pt-BR')}</span>}
                                                                </div>
                                                            </div>
                                                        </div>
                                                        {!isPrivate && <Checkbox checked={isSelected} className="rounded-full shadow-none border-slate-300" />}
                                                    </div>
                                                );
                                            })}
                                        </div>
                                    )}
                                </div>

                                {/* Footer */}
                                <div className="p-4 border-t bg-white flex items-center justify-between shrink-0">
                                    <div className="text-xs font-medium text-slate-500">
                                        {selectedSupportFolder
                                            ? selectedMedia.length === 0
                                                ? `${filteredSupportMedia.length} arquivo(s) disponível(is)`
                                                : `${selectedMedia.length} selecionado(s)`
                                            : `${filteredSupportFolders.length} pasta(s)`
                                        }
                                    </div>
                                    <div className="flex gap-2">
                                        <Button variant="outline" size="sm" onClick={resetMediaModal}>Cancelar</Button>
                                        {selectedSupportFolder && (
                                            <Button
                                                size="sm"
                                                disabled={selectedMedia.length === 0}
                                                onClick={handleConfirmMediaSend}
                                                className="bg-primary hover:bg-primary/90 font-bold"
                                            >
                                                Continuar ({selectedMedia.length})
                                            </Button>
                                        )}
                                    </div>
                                </div>
                            </>
                        ) : (
                            /* ===== CONFIRMATION STEP ===== */
                            <>
                                <DialogHeader className="p-6 pb-2 border-b">
                                    <DialogTitle className="text-xl font-bold font-space flex items-center gap-2">
                                        <AlertTriangle className="w-5 h-5 text-amber-500" />
                                        Confirmar Envio
                                    </DialogTitle>
                                    <DialogDescription className="text-xs mt-1">
                                        Deseja enviar {selectedMedia.length} arquivo(s) para o paciente <strong>{contact?.name}</strong>?
                                    </DialogDescription>
                                </DialogHeader>

                                <div className="flex-1 overflow-y-auto p-6 space-y-4">
                                    <div className="bg-slate-50 rounded-xl p-4 border space-y-2">
                                        {selectedMedia.map(file => (
                                            <div key={file.id} className="flex items-center justify-between bg-white p-2 rounded-lg border border-slate-100 shadow-sm">
                                                <div className="flex items-center gap-3 overflow-hidden">
                                                    <div className="h-8 w-8 bg-slate-50 rounded flex items-center justify-center shrink-0">
                                                        {file.tipo_arquivo === 'image' && <Image className="w-4 h-4 text-emerald-500" />}
                                                        {file.tipo_arquivo === 'video' && <Film className="w-4 h-4 text-blue-500" />}
                                                        {file.tipo_arquivo === 'audio' && <Music className="w-4 h-4 text-amber-500" />}
                                                        {file.tipo_arquivo === 'document' && <FileText className="w-4 h-4 text-slate-500" />}
                                                    </div>
                                                    <span className="text-[13px] font-medium text-slate-700 truncate">{file.nome}</span>
                                                </div>
                                                <Badge variant="outline" className="text-[9px] uppercase tracking-tighter opacity-70">
                                                    {file.tipo_arquivo}
                                                </Badge>
                                            </div>
                                        ))}
                                    </div>
                                    <div className="text-center py-2">
                                        <p className="text-xs text-muted-foreground italic">
                                            As mídias serão enviadas individualmente via Evolution API.
                                        </p>
                                    </div>
                                </div>

                                <div className="p-4 border-t bg-white flex items-center justify-end gap-3">
                                    <Button variant="ghost" size="sm" onClick={() => setIsConfirmingMedia(false)} disabled={sendingMedia}>
                                        Voltar para seleção
                                    </Button>
                                    <Button
                                        size="sm"
                                        onClick={() => handleSendMedia(selectedMedia, true)}
                                        disabled={sendingMedia}
                                        className="bg-emerald-600 hover:bg-emerald-700 text-white font-bold h-10 px-8"
                                    >
                                        {sendingMedia ? (
                                            <>
                                                <Loader2 className="w-4 h-4 mr-2 animate-spin" />
                                                Enviando...
                                            </>
                                        ) : (
                                            <>
                                                <Send className="w-4 h-4 mr-2" />
                                                Confirmar Envio
                                            </>
                                        )}
                                    </Button>
                                </div>
                            </>
                        )}
                    </DialogContent>
                </Dialog>
            </div>
        </div>
    );
}
