import { useState, useEffect, useRef } from "react";
import { useNavigate } from "react-router-dom";
import {
    DndContext,
    DragOverlay,
    closestCorners,
    KeyboardSensor,
    PointerSensor,
    useSensor,
    useSensors,
    type DragStartEvent,
    type DragEndEvent,
} from "@dnd-kit/core";
import { sortableKeyboardCoordinates } from "@dnd-kit/sortable";
import { useCrmStore, type Lead, type LeadStatus } from "@/store/useCrmStore";
import { Column } from "@/components/kanban/Column";
import { LeadCard } from "@/components/kanban/LeadCard";
import { AddLeadModal } from "@/components/modals/AddLeadModal";
import { createPortal } from "react-dom";
import { useAuth } from "@/contexts/AuthContext";
import { Button } from "@/components/ui/button";
import { Tooltip, TooltipContent, TooltipProvider, TooltipTrigger } from "@/components/ui/tooltip";
import { Dialog, DialogContent, DialogHeader, DialogTitle } from "@/components/ui/dialog";
import { Badge } from "@/components/ui/badge";
import { AlertTriangle, Users, Loader2, TrendingUp, Plus, Info } from "lucide-react";
import { startOfMonth, endOfMonth, isWithinInterval, startOfDay, endOfDay } from "date-fns";
import { FilterSelect } from "@/components/kanban/FilterSelect";
import { DateRangePicker } from "@/components/kanban/DateRangePicker";
import { cn } from "@/lib/utils";


export default function CrmBoard() {
    const { clinic } = useAuth();
    const { leads, columns, moveLead, fetchBoardData, loading } = useCrmStore();
    const navigate = useNavigate();
    const [activeLead, setActiveLead] = useState<Lead | null>(null);
    const [isAddModalOpen, setIsAddModalOpen] = useState(false);
    const [isConvModalOpen, setIsConvModalOpen] = useState(false);

    // --- FILTER STATES ---
    const [search, setSearch] = useState("");
    const [sourceFilter, setSourceFilter] = useState("all");
    const [proFilter, setProFilter] = useState("all");
    const [procFilter, setProcFilter] = useState("all");
    const [scoreFilter, setScoreFilter] = useState("all");
    const [dateRange, setDateRange] = useState({
        from: startOfMonth(new Date()),
        to: endOfMonth(new Date())
    });

    // --- DRAG TO SCROLL LOGIC ---
    const scrollContainerRef = useRef<HTMLDivElement>(null);
    const [isDown, setIsDown] = useState(false);
    const [isDragging, setIsDragging] = useState(false);
    const [startX, setStartX] = useState(0);
    const [scrollLeft, setScrollLeft] = useState(0);

    const onMouseDown = (e: React.MouseEvent) => {
        if (!scrollContainerRef.current) return;

        // Prevent drag-scroll if clicking on a draggable card or button
        if ((e.target as Element).closest('[data-lead-card], button, [role="button"]')) {
            return;
        }

        setIsDown(true);
        setStartX(e.pageX - scrollContainerRef.current.offsetLeft);
        setScrollLeft(scrollContainerRef.current.scrollLeft);
    };

    useEffect(() => {
        const onMouseUp = () => {
            setIsDown(false);
            setIsDragging(false);
        };
        const onMouseMove = (e: MouseEvent) => {
            if (!isDown || !scrollContainerRef.current) return;
            e.preventDefault();
            const x = e.pageX - scrollContainerRef.current.offsetLeft;
            const walk = (x - startX) * 1.5;

            if (Math.abs(walk) > 5 && !isDragging) {
                setIsDragging(true);
            }

            scrollContainerRef.current.scrollLeft = scrollLeft - walk;
        };

        if (isDown) {
            window.addEventListener('mouseup', onMouseUp);
            window.addEventListener('mousemove', onMouseMove);
        }
        return () => {
            window.removeEventListener('mouseup', onMouseUp);
            window.removeEventListener('mousemove', onMouseMove);
        };
    }, [isDown, startX, scrollLeft, isDragging]);

    // --- END FILTER STATES ---

    useEffect(() => {
        if (clinic?.id) {
            fetchBoardData(clinic.id);
        }
    }, [clinic?.id, fetchBoardData]);

    const sensors = useSensors(
        useSensor(PointerSensor, {
            activationConstraint: {
                distance: 5,
            },
        }),
        useSensor(KeyboardSensor, {
            coordinateGetter: sortableKeyboardCoordinates,
        })
    );

    const handleDragStart = (event: DragStartEvent) => {
        const { active } = event;
        const lead = active.data.current?.lead as Lead;
        setActiveLead(lead);
    };

    const handleDragEnd = (event: DragEndEvent) => {
        const { active, over } = event;

        setActiveLead(null);

        if (!over) return;

        const leadId = active.id as string;
        const newStatus = over.id as LeadStatus;

        if (active.data.current?.lead.status !== newStatus) {
            moveLead(leadId, newStatus);
        }
    };

    const handleLeadClick = (lead: Lead) => {
        navigate(`/crm/${lead.status}/${lead.id}`);
    };

    // --- FILTERING LOGIC ---
    const filteredLeads = leads.filter(lead => {
        const matchesSearch = lead.name.toLowerCase().includes(search.toLowerCase()) ||
            lead.phone.includes(search);
        const matchesSource = sourceFilter === "all" || lead.source === sourceFilter;
        const matchesPro = proFilter === "all" || lead.responsible_name === proFilter;
        const matchesProc = procFilter === "all" || lead.procedure_name === procFilter;

        let matchesScore = true;
        if (scoreFilter === "hot") matchesScore = (lead.sevenn_score || 0) >= 70;
        else if (scoreFilter === "warm") matchesScore = (lead.sevenn_score || 0) >= 40 && (lead.sevenn_score || 0) < 70;
        else if (scoreFilter === "cold") matchesScore = (lead.sevenn_score || 0) < 40;

        const leadDate = lead.created_at ? new Date(lead.created_at) : null;
        const matchesDate = !leadDate || isWithinInterval(leadDate, {
            start: startOfDay(dateRange.from),
            end: endOfDay(dateRange.to)
        });

        return matchesSearch && matchesSource && matchesPro && matchesProc && matchesScore && matchesDate;
    });

    // --- KPI CALCULATIONS ---
    const now = new Date();
    const todayStr = now.toISOString().split('T')[0];

    const oportunidadesCount = leads.length;
    const leadsHoje = leads.filter(l => l.created_at?.startsWith(todayStr)).length;

    const activeStatus = ['won', 'lost', 'archived'];
    const emAbertoLeads = leads.filter(l => !activeStatus.includes(l.status));
    const emAbertoCount = emAbertoLeads.length;

    const potencialVenda = emAbertoLeads.reduce((acc, l) => acc + (Number(l.estimated_value) || 0), 0);

    const criticalLeads = emAbertoLeads.filter(l => {
        const isEarlyStage = !['scheduled', 'won', 'lost'].includes(l.status);
        if (!isEarlyStage) return false;

        if (l.conversation_status === 'closed') return false;

        const lastInteraction = l.last_interaction_at ? new Date(l.last_interaction_at) : (l.created_at ? new Date(l.created_at) : null);
        if (!lastInteraction) return false;

        const hoursDiff = (now.getTime() - lastInteraction.getTime()) / (1000 * 60 * 60);
        return hoursDiff > 72; // +3 days
    });
    const criticalCount = criticalLeads.length;

    const wonLeads = leads.filter(l => l.status === 'won');
    const lostCount = leads.filter(l => l.status === 'lost').length;
    const closedCount = wonLeads.length + lostCount;
    const conversionRate = closedCount > 0 ? Math.round((wonLeads.length / closedCount) * 100) : 0;

    const faturamentoTotal = wonLeads.reduce((acc, l) => acc + (Number(l.estimated_value) || 0), 0);

    const convByPro = Array.from(new Set(leads.map(l => l.responsible_name || 'Não Atribuído'))).map(pro => {
        const proLeads = leads.filter(l => l.responsible_name === pro || (!l.responsible_name && pro === 'Não Atribuído'));
        const won = proLeads.filter(l => l.status === 'won').length;
        const total = proLeads.filter(l => ['won', 'lost'].includes(l.status)).length;
        return { name: pro, rate: total > 0 ? Math.round((won / total) * 100) : 0, count: won };
    }).sort((a, b) => b.rate - a.rate);

    const convBySource = Array.from(new Set(leads.map(l => l.source || 'Outro'))).map(src => {
        const srcLeads = leads.filter(l => (l.source || 'Outro') === src);
        const won = srcLeads.filter(l => l.status === 'won').length;
        const total = srcLeads.filter(l => ['won', 'lost'].includes(l.status)).length;
        return { name: src, rate: total > 0 ? Math.round((won / total) * 100) : 0, count: won };
    }).sort((a, b) => b.rate - a.rate);

    if (loading) {
        return (
            <div className="flex-1 flex items-center justify-center">
                <Loader2 className="h-8 w-8 animate-spin text-primary" />
            </div>
        );
    }

    return (
        <div className="h-full flex flex-col space-y-1.5 select-none">
            {/* Polished Pipeline Header */}
            {/* Premium Enterprise Pipeline Header - Compact Mode */}
            <div className="px-4 pt-1">
                <div className="bg-white/90 dark:bg-card/90 backdrop-blur-2xl border border-gray-100 dark:border-border/50 shadow-sm rounded-xl py-2 px-4 flex items-center justify-between gap-4 transition-all">
                    <div className="flex items-center gap-6">
                        <div className="shrink-0 flex items-center gap-3">
                            <h1 className="text-lg font-bold tracking-tight text-gray-900 dark:text-zinc-50">Pipeline</h1>
                            <div className="h-6 w-px bg-gray-200 dark:bg-zinc-700" />
                        </div>

                        <div className="flex items-center gap-6">
                            {/* Card 1: Oportunidades */}
                            <TooltipProvider delayDuration={100}>
                                <Tooltip>
                                    <TooltipTrigger asChild>
                                        <div className="flex flex-col items-center text-center cursor-help group">
                                            <div className="flex items-center gap-1 mb-0.5">
                                                <span className="text-[9px] font-medium uppercase tracking-wider text-gray-400 group-hover:text-blue-500 transition-colors">Oportunidades</span>
                                                <Info className="w-2.5 h-2.5 text-gray-300 group-hover:text-blue-400 transition-colors" />
                                            </div>
                                            <div className="flex items-center justify-center gap-2">
                                                <span className="text-lg font-bold tracking-tight text-gray-900 dark:text-zinc-50 leading-none">{oportunidadesCount}</span>
                                                {leadsHoje > 0 && (
                                                    <span className="text-[9px] font-bold bg-blue-50 text-blue-700 px-1.5 rounded-full border border-blue-100 dark:bg-blue-900/20 dark:text-blue-300 dark:border-blue-900/30">
                                                        +{leadsHoje}
                                                    </span>
                                                )}
                                            </div>
                                        </div>
                                    </TooltipTrigger>
                                    <TooltipContent side="top" className="text-[11px] font-medium bg-white dark:bg-zinc-800 text-zinc-600 dark:text-zinc-300 border-zinc-200 dark:border-zinc-700 shadow-xl">
                                        Volume total de leads captados no período selecionado.
                                    </TooltipContent>
                                </Tooltip>
                            </TooltipProvider>

                            {/* Card 2: Em Aberto */}
                            <TooltipProvider delayDuration={100}>
                                <Tooltip>
                                    <TooltipTrigger asChild>
                                        <div className="flex flex-col items-center text-center border-l border-gray-100 dark:border-zinc-800 pl-6 cursor-help group">
                                            <div className="flex items-center gap-1 mb-0.5">
                                                <span className="text-[9px] font-medium uppercase tracking-wider text-gray-400 group-hover:text-amber-500 transition-colors">Em Aberto</span>
                                                <Info className="w-2.5 h-2.5 text-gray-300 group-hover:text-amber-400 transition-colors" />
                                            </div>
                                            <span className="text-lg font-bold tracking-tight text-gray-900 dark:text-zinc-50 leading-none">{emAbertoCount}</span>
                                        </div>
                                    </TooltipTrigger>
                                    <TooltipContent side="top" className="text-[11px] font-medium bg-white dark:bg-zinc-800 text-zinc-600 dark:text-zinc-300 border-zinc-200 dark:border-zinc-700 shadow-xl">
                                        Leads que estão em fases ativas (exclui ganhos/descartes).
                                    </TooltipContent>
                                </Tooltip>
                            </TooltipProvider>

                            {/* Card 4: Críticos */}
                            <TooltipProvider delayDuration={100}>
                                <Tooltip>
                                    <TooltipTrigger asChild>
                                        <div className="flex flex-col items-center text-center border-l border-gray-100 dark:border-zinc-800 pl-6 cursor-help group">
                                            <div className="flex items-center gap-1.5 mb-0.5">
                                                <span className={cn("text-[9px] font-medium uppercase tracking-wider transition-colors", criticalCount > 0 ? "text-rose-600" : "text-gray-400")}>Críticos</span>
                                                <Info className={cn("w-2.5 h-2.5 transition-colors", criticalCount > 0 ? "text-rose-300" : "text-gray-300 group-hover:text-rose-400")} />
                                                {criticalCount > 0 && <AlertTriangle className="w-3 h-3 text-rose-500 animate-pulse" />}
                                            </div>
                                            <span className={cn("text-lg font-bold tracking-tight leading-none transition-colors", criticalCount > 0 ? "text-rose-600" : "text-gray-300")}>
                                                {criticalCount}
                                            </span>
                                        </div>
                                    </TooltipTrigger>
                                    <TooltipContent side="top" className={cn("font-medium border shadow-xl text-[11px]", criticalCount > 0 ? "bg-rose-50 text-rose-700 border-rose-100" : "bg-emerald-50 text-emerald-700 border-emerald-100")}>
                                        {criticalCount > 0 ? "Atenção necessária: Leads em fases iniciais sem contato há mais de 72 horas." : "Tudo sob controle."}
                                    </TooltipContent>
                                </Tooltip>
                            </TooltipProvider>

                            {/* Card 3: Potencial */}
                            <TooltipProvider delayDuration={100}>
                                <Tooltip>
                                    <TooltipTrigger asChild>
                                        <div className="flex flex-col items-center text-center border-l border-gray-100 dark:border-zinc-800 pl-6 cursor-help group">
                                            <div className="flex items-center gap-1 mb-0.5">
                                                <span className="text-[9px] font-medium uppercase tracking-wider text-gray-400 group-hover:text-blue-600 transition-colors">Potencial</span>
                                                <Info className="w-2.5 h-2.5 text-gray-300 group-hover:text-blue-400 transition-colors" />
                                            </div>
                                            <span className="text-lg font-bold tracking-tight text-blue-600 dark:text-blue-400 leading-none">
                                                {new Intl.NumberFormat('pt-BR', { style: 'currency', currency: 'BRL', maximumFractionDigits: 0 }).format(potencialVenda)}
                                            </span>
                                        </div>
                                    </TooltipTrigger>
                                    <TooltipContent side="top" className="text-[11px] font-medium bg-white dark:bg-zinc-800 text-zinc-600 dark:text-zinc-300 border-zinc-200 dark:border-zinc-700 shadow-xl">
                                        Expectativa financeira baseada nos leads em aberto.
                                    </TooltipContent>
                                </Tooltip>
                            </TooltipProvider>

                            {/* Card 5: Taxa de Conversão */}
                            <TooltipProvider delayDuration={100}>
                                <Tooltip>
                                    <TooltipTrigger asChild>
                                        <div
                                            onClick={() => setIsConvModalOpen(true)}
                                            className="flex flex-col items-center text-center border-l border-gray-100 dark:border-zinc-800 pl-6 cursor-pointer group"
                                        >
                                            <div className="flex items-center gap-1 mb-0.5">
                                                <span className="text-[9px] font-medium uppercase tracking-wider text-gray-400 group-hover:text-purple-600 transition-colors">
                                                    Conversão
                                                </span>
                                                <Info className="w-2.5 h-2.5 text-gray-300 group-hover:text-purple-400 transition-colors" />
                                            </div>
                                            <span className="text-lg font-bold tracking-tight text-gray-900 dark:text-zinc-50 leading-none group-hover:text-purple-600 transition-colors">{conversionRate}%</span>
                                        </div>
                                    </TooltipTrigger>
                                    <TooltipContent side="top" className="text-[11px] font-medium bg-white dark:bg-zinc-800 text-zinc-600 dark:text-zinc-300 border-zinc-200 dark:border-zinc-700 shadow-xl">
                                        Eficiência do funil baseada em leads ganhos vs encerrados.
                                    </TooltipContent>
                                </Tooltip>
                            </TooltipProvider>

                            {/* Card 6: Faturamento */}
                            <TooltipProvider delayDuration={100}>
                                <Tooltip>
                                    <TooltipTrigger asChild>
                                        <div className="flex flex-col items-center text-center border-l border-gray-100 dark:border-zinc-800 pl-6 cursor-help group">
                                            <div className="flex items-center gap-1 mb-0.5">
                                                <span className="text-[9px] font-medium uppercase tracking-wider text-gray-400 group-hover:text-emerald-600 transition-colors">Faturamento</span>
                                                <Info className="w-2.5 h-2.5 text-gray-300 group-hover:text-emerald-400 transition-colors" />
                                            </div>
                                            <span className="text-lg font-bold tracking-tight text-emerald-600 dark:text-emerald-400 leading-none">
                                                {new Intl.NumberFormat('pt-BR', { style: 'currency', currency: 'BRL', maximumFractionDigits: 0 }).format(faturamentoTotal)}
                                            </span>
                                        </div>
                                    </TooltipTrigger>
                                    <TooltipContent side="top" className="text-[11px] font-medium bg-white dark:bg-zinc-800 text-zinc-600 dark:text-zinc-300 border-zinc-200 dark:border-zinc-700 shadow-xl">
                                        Receita total gerada pelos leads marcados como Ganhos.
                                    </TooltipContent>
                                </Tooltip>
                            </TooltipProvider>
                        </div>
                    </div>

                    <Button
                        onClick={() => setIsAddModalOpen(true)}
                        variant="default"
                        size="sm"
                        className="gap-2 shadow-sm font-semibold h-9 px-4 rounded-lg bg-gray-900 text-white hover:bg-gray-800 dark:bg-zinc-100 dark:text-zinc-900 shrink-0"
                    >
                        <Plus className="h-4 w-4" /> Novo Lead
                    </Button>
                </div>

                {/* Filters Section */}
                <div className="grid grid-cols-1 md:grid-cols-7 gap-2 bg-muted/40 p-1.5 rounded-xl border border-border/40 mt-1.5">
                    <DateRangePicker value={dateRange} onChange={setDateRange} />

                    <div className="col-span-2 relative">
                        <Users className="absolute left-2.5 top-1/2 -translate-y-1/2 w-3.5 h-3.5 text-muted-foreground" />
                        <input
                            className="w-full bg-background border rounded-lg h-8 pl-8 text-[11px] focus:ring-1 focus:ring-primary outline-none transition-all"
                            placeholder="Buscar por nome ou celular..."
                            value={search}
                            onChange={(e) => setSearch(e.target.value)}
                        />
                    </div>

                    <FilterSelect
                        value={sourceFilter}
                        onChange={setSourceFilter}
                        placeholder="TODAS FONTES"
                        options={[
                            { value: "all", label: "TODAS FONTES" },
                            { value: "Instagram", label: "INSTAGRAM" },
                            { value: "Google", label: "GOOGLE" },
                            { value: "Indicação", label: "INDICAÇÃO" },
                            { value: "Passante", label: "PASSANTE" },
                        ]}
                    />

                    <FilterSelect
                        value={proFilter}
                        onChange={setProFilter}
                        placeholder="PROFISSIONAL"
                        options={[
                            { value: "all", label: "PROFISSIONAL" },
                            { value: "Dr. João", label: "DR. JOÃO" },
                            { value: "Dra. Ana", label: "DRA. ANA" },
                        ]}
                    />

                    <FilterSelect
                        value={procFilter}
                        onChange={setProcFilter}
                        placeholder="PROCEDIMENTO"
                        options={[
                            { value: "all", label: "PROCEDIMENTO" },
                            { value: "Harmonização", label: "HARMONIZAÇÃO" },
                            { value: "Botox", label: "BOTOX" },
                            { value: "Peeling", label: "PEELING" },
                        ]}
                    />

                    <FilterSelect
                        value={scoreFilter}
                        onChange={setScoreFilter}
                        placeholder="QUALQUER SCORE"
                        options={[
                            { value: "all", label: "QUALQUER SCORE" },
                            { value: "hot", label: "QUENTE (70+)" },
                            { value: "warm", label: "MORNO (40-70)" },
                            { value: "cold", label: "FRIO (0-40)" },
                        ]}
                    />
                </div>
            </div>

            <DndContext
                sensors={sensors}
                collisionDetection={closestCorners}
                onDragStart={handleDragStart}
                onDragEnd={handleDragEnd}
            >
                <div
                    ref={scrollContainerRef}
                    onMouseDown={onMouseDown}
                    className={cn(
                        "flex-1 overflow-x-auto overflow-y-hidden pt-1 scrollbar-hide cursor-grab select-none active:cursor-grabbing",
                        isDragging ? "cursor-grabbing" : ""
                    )}
                >
                    <div className="flex h-full gap-3 px-4 pb-2 min-w-max">
                        {columns.map((col) => (
                            <Column
                                key={col.id}
                                id={col.id}
                                title={col.title}
                                color={col.color}
                                leads={filteredLeads.filter((l) => l.status === col.id)}
                                onLeadClick={handleLeadClick}
                            />
                        ))}
                    </div>
                </div>

                {createPortal(
                    <DragOverlay>
                        {activeLead ? (
                            <LeadCard lead={activeLead} onClick={() => { }} />
                        ) : null}
                    </DragOverlay>,
                    document.body
                )}
            </DndContext>

            <AddLeadModal
                isOpen={isAddModalOpen}
                onClose={() => setIsAddModalOpen(false)}
            />

            {/* Modal de Detalhes da Conversão */}
            <Dialog open={isConvModalOpen} onOpenChange={setIsConvModalOpen}>
                <DialogContent className="sm:max-w-xl">
                    <DialogHeader>
                        <DialogTitle className="text-xl font-black flex items-center gap-2">
                            <TrendingUp className="w-6 h-6 text-purple-500" />
                            Análise de Conversão
                        </DialogTitle>
                    </DialogHeader>
                    <div className="grid grid-cols-2 gap-8 py-6">
                        <div className="space-y-4">
                            <h4 className="text-[10px] font-black uppercase text-muted-foreground border-b pb-2 tracking-widest">Por Profissional</h4>
                            {convByPro.map(pro => (
                                <div key={pro.name} className="flex items-center justify-between group">
                                    <div className="flex flex-col">
                                        <span className="text-xs font-bold leading-none">{pro.name}</span>
                                        <span className="text-[9px] text-muted-foreground mt-1">{pro.count} vendas concluídas</span>
                                    </div>
                                    <Badge variant="secondary" className="font-black text-[10px] group-hover:bg-purple-100 group-hover:text-purple-600 transition-colors">
                                        {pro.rate}%
                                    </Badge>
                                </div>
                            ))}
                        </div>
                        <div className="space-y-4">
                            <h4 className="text-[10px] font-black uppercase text-muted-foreground border-b pb-2 tracking-widest">Por Origem</h4>
                            {convBySource.map(src => (
                                <div key={src.name} className="flex items-center justify-between group">
                                    <div className="flex flex-col">
                                        <span className="text-xs font-bold leading-none">{src.name}</span>
                                        <span className="text-[9px] text-muted-foreground mt-1">{src.count} vendas concluídas</span>
                                    </div>
                                    <Badge variant="secondary" className="font-black text-[10px] group-hover:bg-blue-100 group-hover:text-blue-600 transition-colors">
                                        {src.rate}%
                                    </Badge>
                                </div>
                            ))}
                        </div>
                    </div>
                </DialogContent>
            </Dialog>
        </div>
    );
}
