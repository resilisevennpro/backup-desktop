-- 3. CRIAR políticas RLS corretas para APPOINTMENTS

-- Política de SELECT (leitura)
CREATE POLICY "clinic_members_select_appointments"
ON appointments
FOR SELECT
USING (
    clinic_id IN (
        SELECT clinic_id 
        FROM profiles 
        WHERE id = auth.uid()
    )
);

-- Política de INSERT (criação)
CREATE POLICY "clinic_members_insert_appointments"
ON appointments
FOR INSERT
WITH CHECK (
    clinic_id IN (
        SELECT clinic_id 
        FROM profiles 
        WHERE id = auth.uid()
    )
);

-- Política de UPDATE (atualização)
CREATE POLICY "clinic_members_update_appointments"
ON appointments
FOR UPDATE
USING (
    clinic_id IN (
        SELECT clinic_id 
        FROM profiles 
        WHERE id = auth.uid()
    )
)
WITH CHECK (
    clinic_id IN (
        SELECT clinic_id 
        FROM profiles 
        WHERE id = auth.uid()
    )
);

-- Política de DELETE (exclusão)
CREATE POLICY "clinic_members_delete_appointments"
ON appointments
FOR DELETE
USING (
    clinic_id IN (
        SELECT clinic_id 
        FROM profiles 
        WHERE id = auth.uid()
    )
);
