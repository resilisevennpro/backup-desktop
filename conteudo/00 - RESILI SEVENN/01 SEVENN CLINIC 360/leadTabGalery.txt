// Tipos
interface GalleryImage {
    id: string;
    url: string;
    thumbnail: string;
    category: 'paciente' | 'clinica' | 'editadas' | 'posts';
    created_at: string; // From database
    author: string;
    description?: string;
    lead_id?: string;
    clinic_id?: string;
}



const CATEGORIES = [
    { id: 'todas', label: 'TODAS', icon: null },
    { id: 'paciente', label: 'PACIENTE', icon: WhatsAppIcon },
    { id: 'clinica', label: 'CLÍNICA', icon: ClinicIcon },
    { id: 'editadas', label: 'EDITADAS', icon: Pencil },
    { id: 'posts', label: 'POSTS', icon: ExternalLink },
];

interface LeadGalleryTabProps {
    leadId?: string;
    onSendToChat?: (imageUrl: string) => void;
}

export function LeadGalleryTab({ leadId, onSendToChat }: LeadGalleryTabProps) {
    const { clinic, profile } = useAuth();
    const [images, setImages] = useState<GalleryImage[]>([]);
    const [loading, setLoading] = useState(true);
    const [isUploading, setIsUploading] = useState(false);
    const fileInputRef = useRef<HTMLInputElement>(null);
    // 1. Fetch Images from Database
    useEffect(() => {
        if (!leadId) return;

        const fetchImages = async () => {
            setLoading(true);
            try {
                const { data, error } = await supabase
                    .from('lead_gallery')
                    .select('*')
                    .eq('lead_id', leadId)
                    .order('created_at', { ascending: false });

                if (error) throw error;
                setImages(data || []);
            } catch (error) {
                console.error("Erro ao carregar galeria:", error);
                toast.error("Não foi possível carregar as imagens da galeria.");
            } finally {
                setLoading(false);
            }
        };

        fetchImages();
    }, [leadId]);

    // 2. Upload Logic
    const handleFileUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
        const file = e.target.files?.[0];
        if (!file || !leadId || !clinic?.id) return;

        setIsUploading(true);
        const fileExt = file.name.split('.').pop();
        const fileName = `${leadId}/${Date.now()}.${fileExt}`;
        const filePath = `gallery/${fileName}`;

        try {
            // A. Upload to Supabase Storage
            const { error: uploadError } = await supabase.storage
                .from('lead-images')
                .upload(filePath, file);

            if (uploadError) throw uploadError;

            // B. Get Public URL
            const { data: { publicUrl } } = supabase.storage
                .from('lead-images')
                .getPublicUrl(filePath);

            // C. Save Reference in lead_gallery Table
            const newImageData = {
                clinic_id: clinic.id,
                lead_id: leadId,
                url: publicUrl,
                thumbnail: publicUrl, // Using publicUrl as thumbnail for now
                category: 'clinica', // Default for clinic uploads
                author: profile?.full_name || 'Clínica',
                description: 'Upload via sistema'
            };

            const { data: inserted, error: dbError } = await supabase
                .from('lead_gallery')
                .insert([newImageData])
                .select()
                .single();

            if (dbError) throw dbError;

            // D. Update Local State
            setImages(prev => [inserted, ...prev]);
            toast.success("Upload realizado com sucesso!");
        } catch (error: any) {
            console.error("Erro no upload:", error);
            toast.error(`Falha no upload: ${error.message || 'Erro desconhecido'}`);
        } finally {
            setIsUploading(false);
            if (fileInputRef.current) fileInputRef.current.value = "";
        }
    };

    const deleteImage = async (img: GalleryImage) => {
        try {
            // Delete from Database
            const { error: dbError } = await supabase
                .from('lead_gallery')
                .delete()
                .eq('id', img.id);

            if (dbError) throw dbError;

            // Try to delete from Storage (extract path from URL)
            const urlParts = img.url.split('lead-images/');
            if (urlParts.length > 1) {
                const path = urlParts[1];
                await supabase.storage.from('lead-images').remove([path]);
            }

            setImages(prev => prev.filter(i => i.id !== img.id));
            toast.success("Imagem removida!");
        } catch (error) {
            console.error("Erro ao deletar:", error);
            toast.error("Erro ao remover imagem.");
        } finally {
            setImageToDelete(null);
        }
    };

    const [activeCategory, setActiveCategory] = useState('todas');
    const [selectedImages, setSelectedImages] = useState<string[]>([]);
    const [viewImage, setViewImage] = useState<GalleryImage | null>(null);
    const [isEditorOpen, setIsEditorOpen] = useState(false);
    const [isCompareOpen, setIsCompareOpen] = useState(false);
    const [editingImage, setEditingImage] = useState<GalleryImage | null>(null);
    const [imageToDelete, setImageToDelete] = useState<GalleryImage | null>(null);

    // Editor State
    const canvasRef = useRef<HTMLCanvasElement>(null);
    const [drawColor, setDrawColor] = useState('#ef4444');
    const [drawTool, setDrawTool] = useState<'arrow' | 'circle' | 'text'>('arrow');
    const [isDrawing, setIsDrawing] = useState(false);
    const [startPos, setStartPos] = useState({ x: 0, y: 0 });

    // Filter images by category
    const filteredImages = activeCategory === 'todas'
        ? images
        : images.filter(img => img.category === activeCategory);

    // Group by date
    const groupedImages = filteredImages.reduce((groups, img) => {
        const date = img.created_at ? img.created_at.split('T')[0] : 'Desconhecido';
        if (!groups[date]) groups[date] = [];
        groups[date].push(img);
        return groups;
    }, {} as Record<string, GalleryImage[]>);

    const sortedDates = Object.keys(groupedImages).sort((a, b) => new Date(b).getTime() - new Date(a).getTime());

    // Selection handlers
    const toggleSelectImage = (id: string) => {
        setSelectedImages(prev =>
            prev.includes(id) ? prev.filter(i => i !== id) : [...prev, id]
        );
    };

    const clearSelection = () => setSelectedImages([]);

    // Open editor
    const openEditor = (img: GalleryImage) => {
        setEditingImage(img);
        setIsEditorOpen(true);
    };

    // Open compare modal
    const openCompareModal = () => {
        if (selectedImages.length === 2) {
            setIsCompareOpen(true);
        }
    };

    // Canvas drawing functions
    const handleCanvasMouseDown = (e: React.MouseEvent<HTMLCanvasElement>) => {
        const canvas = canvasRef.current;
        if (!canvas) return;
        const rect = canvas.getBoundingClientRect();
        setStartPos({
            x: e.clientX - rect.left,
            y: e.clientY - rect.top
        });
        setIsDrawing(true);
    };

    const handleCanvasMouseUp = (e: React.MouseEvent<HTMLCanvasElement>) => {
        if (!isDrawing) return;
        const canvas = canvasRef.current;
        const ctx = canvas?.getContext('2d');
        if (!canvas || !ctx) return;

        const rect = canvas.getBoundingClientRect();
        const endX = e.clientX - rect.left;
        const endY = e.clientY - rect.top;

        ctx.strokeStyle = drawColor;
        ctx.lineWidth = 3;
        ctx.lineCap = 'round';

        if (drawTool === 'circle') {
            const radius = Math.sqrt(Math.pow(endX - startPos.x, 2) + Math.pow(endY - startPos.y, 2));
            ctx.beginPath();
            ctx.arc(startPos.x, startPos.y, radius, 0, 2 * Math.PI);
            ctx.stroke();
        } else if (drawTool === 'arrow') {
            // Draw line
            ctx.beginPath();
            ctx.moveTo(startPos.x, startPos.y);
            ctx.lineTo(endX, endY);
            ctx.stroke();
            // Draw arrowhead
            const angle = Math.atan2(endY - startPos.y, endX - startPos.x);
            const headLen = 15;
            ctx.beginPath();
            ctx.moveTo(endX, endY);
            ctx.lineTo(endX - headLen * Math.cos(angle - Math.PI / 6), endY - headLen * Math.sin(angle - Math.PI / 6));
            ctx.moveTo(endX, endY);
            ctx.lineTo(endX - headLen * Math.cos(angle + Math.PI / 6), endY - headLen * Math.sin(angle + Math.PI / 6));
            ctx.stroke();
        } else if (drawTool === 'text') {
            const text = prompt('Digite o rótulo:');
            if (text) {
                ctx.font = 'bold 16px Inter, sans-serif';
                ctx.fillStyle = drawColor;
                ctx.fillText(text, startPos.x, startPos.y);
            }
        }

        setIsDrawing(false);
    };

    // Initialize canvas with image
    const [canvasLoaded, setCanvasLoaded] = useState(false);

    useEffect(() => {
        if (isEditorOpen && editingImage && canvasRef.current) {
            setCanvasLoaded(false);
            const canvas = canvasRef.current;
            const ctx = canvas.getContext('2d');
            if (!ctx) return;

            // TIMEOUT DE SEGURANÇA: Se não carregar em 8 segundos, força a saída da tela de loading
            const timer = setTimeout(() => {
                if (!canvasLoaded) {
                    console.warn("Timeout atingido no editor. Forçando renderização.");
                    setCanvasLoaded(true);
                }
            }, 8000);

            const loadImage = async () => {
                try {
                    let src = editingImage.url;
                    // Bypass Cache logic
                    if (src.startsWith('http')) {
                        const separator = src.includes('?') ? '&' : '?';
                        src = src + separator + 't=' + new Date().getTime();
                    }

                    console.log("Iniciando carregamento da imagem via fetch:", src);

                    const response = await fetch(src, {
                        mode: 'cors',
                        cache: 'no-store'
                    });

                    if (!response.ok) throw new Error(`Fetch error: ${response.status} ${response.statusText}`);

                    const blob = await response.blob();
                    const objectUrl = URL.createObjectURL(blob);

                    const img = new Image();
                    img.onload = () => {
                        clearTimeout(timer);
                        console.log("Imagem carregada (onload fired)");

                        const maxWidth = 1000;
                        const maxHeight = 800;
                        let width = img.width;
                        let height = img.height;

                        if (width > maxWidth) {
                            height = (maxWidth / width) * height;
                            width = maxWidth;
                        }
                        if (height > maxHeight) {
                            width = (maxHeight / height) * width;
                            height = maxHeight;
                        }

                        canvas.width = width;
                        canvas.height = height;
                        ctx.clearRect(0, 0, width, height);
                        ctx.drawImage(img, 0, 0, width, height);

                        URL.revokeObjectURL(objectUrl);
                        setCanvasLoaded(true);
                    };

                    img.onerror = (e) => {
                        console.error("Erro no img.onerror dentro do fetch logic:", e);
                        URL.revokeObjectURL(objectUrl);
                        throw new Error("Falha ao renderizar blob de imagem");
                    };

                    img.src = objectUrl;

                } catch (error) {
                    clearTimeout(timer);
                    console.error("Erro crítico ao carregar imagem para editor:", error);
                    toast.error("Erro ao carregar imagem para edição. Verifique o console.");
                    setCanvasLoaded(true);
                    setIsEditorOpen(false);
                }
            };

            loadImage();

            return () => clearTimeout(timer);


        }
    }, [isEditorOpen, editingImage]);

    // Save edited image
    const saveEditedImage = async () => {
        const canvas = canvasRef.current;
        if (!canvas || !leadId || !clinic?.id) return;

        try {
            const dataUrl = canvas.toDataURL('image/png');

            // Convert dataURL to Blob for upload
            const response = await fetch(dataUrl);
            const blob = await response.blob();

            const filePath = `gallery/${leadId}/edited_${Date.now()}.png`;

            // 1. Upload to Supabase Storage
            const { error: uploadError } = await supabase.storage
                .from('lead-images')
                .upload(filePath, blob);

            if (uploadError) throw uploadError;

            // 2. Get Public URL
            const { data: { publicUrl } } = supabase.storage
                .from('lead-images')
                .getPublicUrl(filePath);

            // 3. Save to Database
            const newImageData = {
                clinic_id: clinic.id,
                lead_id: leadId,
                url: publicUrl,
                thumbnail: publicUrl,
                category: 'editadas',
                author: profile?.full_name || 'Profissional',
                description: 'Imagem com marcações'
            };

            const { data: inserted, error: dbError } = await supabase
                .from('lead_gallery')
                .insert([newImageData])
                .select()
                .single();

            if (dbError) throw dbError;

            setImages(prev => [inserted, ...prev]);
            setIsEditorOpen(false);
            setEditingImage(null);
            toast.success("Edição salva com sucesso!");
        } catch (error) {
            console.error("Erro ao salvar edição:", error);
            toast.error("Não foi possível salvar a imagem editada.");
        }
    };

    // Generate before/after
    const generateBeforeAfter = (format: 'feed' | 'stories') => {
        if (selectedImages.length !== 2 || !clinic?.id) return;

        const [beforeId, afterId] = selectedImages;
        const beforeImg = images.find(i => i.id === beforeId);
        const afterImg = images.find(i => i.id === afterId);

        if (!beforeImg || !afterImg) return;

        // Create canvas for composition
        const canvas = document.createElement('canvas');
        const ctx = canvas.getContext('2d');
        if (!ctx) return;

        const width = format === 'feed' ? 1080 : 1080;
        const height = format === 'feed' ? 1350 : 1920;

        canvas.width = width;
        canvas.height = height;

        // Background
        ctx.fillStyle = '#ffffff';
        ctx.fillRect(0, 0, width, height);

        const loadImage = (src: string): Promise<HTMLImageElement> => {
            return new Promise((resolve, reject) => {
                const img = new Image();
                img.crossOrigin = 'anonymous';
                img.onload = () => resolve(img);
                img.onerror = reject;
                // Cache bypass optimization for canvas export
                if (src.startsWith('http')) {
                    const separator = src.includes('?') ? '&' : '?';
                    img.src = src + separator + 't=' + new Date().getTime();
                } else {
                    img.src = src;
                }
            });
        };

        Promise.all([loadImage(beforeImg.url), loadImage(afterImg.url)]).then(async ([before, after]) => {
            const imgHeight = format === 'feed' ? 550 : 800;
            const imgWidth = width / 2 - 30;
            const startY = format === 'feed' ? 100 : 200;

            // Draw images side by side
            ctx.drawImage(before, 20, startY, imgWidth, imgHeight);
            ctx.drawImage(after, width / 2 + 10, startY, imgWidth, imgHeight);

            // Labels
            ctx.font = 'bold 32px Inter, sans-serif';
            ctx.fillStyle = '#1e293b';
            ctx.textAlign = 'center';
            ctx.fillText('ANTES', width / 4, startY + imgHeight + 50);
            ctx.fillText('DEPOIS', (width * 3) / 4, startY + imgHeight + 50);

            // Watermark
            ctx.font = 'bold 24px Inter, sans-serif';
            ctx.fillStyle = '#59A52C';
            ctx.fillText('SEVENN', width / 2, height - 50);

            // Download
            const link = document.createElement('a');
            link.download = `antes-depois-${format}-${Date.now()}.png`;
            link.href = canvas.toDataURL('image/png');
            link.click();

            // Save to posts category
            const newPostData = {
                clinic_id: clinic.id,
                lead_id: leadId,
                url: canvas.toDataURL('image/png'),
                thumbnail: canvas.toDataURL('image/png'),
                category: 'posts',
                author: 'Sistema',
                description: `Antes e Depois - ${format === 'feed' ? 'Feed' : 'Stories'}`
            };

            // For simplicity in a mock-to-real migration, we'll try to just update state if DB table isn't ready for large blobs,
            // but the correct way is to upload to storage first.
            // However, the user specifically asked for "Upload Real", so I'll stick to storage if possible.
            // To keep it robust, I'll just update state for this complex generated post for now or implement storage if requested.
            // Let's implement storage for consistency.

            try {
                const blob = await (await fetch(newPostData.url)).blob();
                const path = `gallery/${leadId}/post_${Date.now()}.png`;
                await supabase.storage.from('lead-images').upload(path, blob);
                const { data: { publicUrl } } = supabase.storage.from('lead-images').getPublicUrl(path);

                const { data: inserted } = await supabase.from('lead_gallery').insert([{
                    ...newPostData,
                    url: publicUrl,
                    thumbnail: publicUrl
                }]).select().single();

                if (inserted) setImages(prev => [inserted, ...prev]);
            } catch (e) {
                console.error("Erro ao salvar post:", e);
                // Fallback to state if DB fails for some reason
                // setImages(prev => [{...newPostData, id: Date.now().toString()} as GalleryImage, ...prev]);
            }

            setIsCompareOpen(false);
            clearSelection();
        });
    };

    const formatDate = (dateStr: string) => {
        if (!dateStr || dateStr === 'Desconhecido') return 'DATA DESCONHECIDA';
        const date = new Date(dateStr + 'T12:00:00');
        return date.toLocaleDateString('pt-BR', { day: '2-digit', month: 'long', year: 'numeric' }).toUpperCase();
    };

    const getCategoryBadge = (category: string) => {
        switch (category) {
            case 'paciente': return { icon: WhatsAppIcon, color: 'bg-green-500' };
            case 'clinica': return { icon: ClinicIcon, color: 'bg-blue-500' };
            case 'editadas': return { icon: Pencil, color: 'bg-orange-500' };
            case 'posts': return { icon: ExternalLink, color: 'bg-purple-500' };
            default: return { icon: null, color: 'bg-gray-500' };
        }
    };

    return (
        <div className="flex flex-col h-full">
            {/* Toolbar */}
            <div className="flex items-center justify-between px-6 py-3 bg-white border-b sticky top-0 z-10">
                {/* Category Tabs */}
                <div className="flex gap-1 bg-slate-100 p-1 rounded-lg">
                    {CATEGORIES.map(cat => (
                        <button
                            key={cat.id}
                            onClick={() => setActiveCategory(cat.id)}
                            className={cn(
                                "px-3 py-1.5 text-[10px] font-bold uppercase tracking-wider rounded-md transition-all flex items-center gap-1.5",
                                activeCategory === cat.id
                                    ? "bg-white text-slate-800 shadow-sm"
                                    : "text-slate-500 hover:text-slate-700"
                            )}
                        >
                            {cat.icon && <cat.icon />}
                            {cat.label}
                        </button>
                    ))}
                </div>

                {/* Actions */}
                <div className="flex items-center gap-2">
                    {selectedImages.length > 0 && (
                        <Badge variant="secondary" className="bg-primary/10 text-primary border-none gap-1.5 px-3 py-1 h-8">
                            <Layers className="w-3 h-3" />
                            Selecionado ({selectedImages.length}/2)
                        </Badge>
                    )}

                    {selectedImages.length === 2 && (
                        <Button
                            onClick={openCompareModal}
                            className="bg-[#59A52C] hover:bg-[#4d8f26] text-white font-bold text-xs gap-2 h-8"
                        >
                            Gerar Comparativo
                        </Button>
                    )}

                    <input
                        type="file"
                        className="hidden"
                        ref={fileInputRef}
                        accept="image/*"
                        onChange={handleFileUpload}
                    />
                    <Button
                        variant="outline"
                        className="h-8 text-xs font-bold gap-2 border-slate-200"
                        onClick={() => fileInputRef.current?.click()}
                        disabled={isUploading}
                    >
                        {isUploading ? (
                            <Loader2 className="w-3.5 h-3.5 animate-spin" />
                        ) : (
                            <Upload className="w-3.5 h-3.5" />
                        )}
                        {isUploading ? "Enviando..." : "Upload"}
                    </Button>
                </div>
            </div>

            {/* Gallery Grid */}
            <div className="flex-1 overflow-y-auto p-6 bg-slate-50/50">
                {sortedDates.map(date => (
                    <div key={date} className="mb-8">
                        {/* Date Header */}
                        <div className="flex items-center gap-2 mb-4">
                            <Calendar className="w-4 h-4 text-slate-400" />
                            <span className="text-xs font-bold text-slate-500 uppercase tracking-wider">
                                {formatDate(date)}
                            </span>
                        </div>

                        {/* Images Grid */}
                        <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-5 gap-4">
                            {groupedImages[date].map(img => {
                                const badge = getCategoryBadge(img.category);
                                const isSelected = selectedImages.includes(img.id);

                                return (
                                    <div
                                        key={img.id}
                                        className={cn(
                                            "relative aspect-square rounded-lg overflow-hidden bg-white border-2 transition-all cursor-pointer group shadow-sm hover:shadow-md",
                                            isSelected ? "border-primary ring-2 ring-primary/20" : "border-transparent hover:border-slate-200"
                                        )}
                                        onClick={() => toggleSelectImage(img.id)}
                                    >
                                        {/* Image */}
                                        <img
                                            src={img.thumbnail}
                                            alt={img.description}
                                            crossOrigin="anonymous"
                                            className="w-full h-full object-cover transition-transform duration-300 group-hover:scale-105"
                                        />

                                        {/* Category Badge */}
                                        <div className={cn("absolute top-2 left-2 w-6 h-6 rounded-full flex items-center justify-center text-white shadow-md", badge.color)}>
                                            {badge.icon && <badge.icon />}
                                        </div>

                                        {/* Selection Indicator */}
                                        {isSelected && (
                                            <div className="absolute top-2 right-2 w-6 h-6 rounded-full bg-primary flex items-center justify-center text-white shadow-md">
                                                <Check className="w-3.5 h-3.5" />
                                            </div>
                                        )}

                                        {/* Hover Actions */}
                                        <div className="absolute inset-0 bg-black/50 opacity-0 group-hover:opacity-100 transition-opacity flex items-center justify-center gap-2">
                                            <Button
                                                size="icon"
                                                variant="secondary"
                                                className="h-8 w-8 rounded-full bg-white/90 hover:bg-white"
                                                onClick={(e) => { e.stopPropagation(); setViewImage(img); }}
                                            >
                                                <ZoomIn className="w-4 h-4 text-slate-700" />
                                            </Button>
                                            <Button
                                                size="icon"
                                                variant="secondary"
                                                className="h-8 w-8 rounded-full bg-white/90 hover:bg-white"
                                                onClick={(e) => { e.stopPropagation(); openEditor(img); }}
                                            >
                                                <Pencil className="w-4 h-4 text-slate-700" />
                                            </Button>
                                            <Button
                                                size="icon"
                                                variant="secondary"
                                                className="h-8 w-8 rounded-full bg-white/90 hover:bg-white"
                                                onClick={(e) => {
                                                    e.stopPropagation();
                                                    setImageToDelete(img);
                                                }}
                                            >
                                                <Trash2 className="w-4 h-4 text-red-500" />
                                            </Button>
                                        </div>
                                    </div>
                                );
                            })}
                        </div>
                    </div>
                ))}

                {loading ? (
                    <div className="flex flex-col items-center justify-center py-20">
                        <Loader2 className="w-8 h-8 text-primary animate-spin mb-4" />
                        <span className="text-sm font-medium text-slate-500">Buscando imagens da galeria...</span>
                    </div>
                ) : filteredImages.length === 0 && (
                    <div className="flex flex-col items-center justify-center py-20 text-center">
                        <div className="w-16 h-16 rounded-full bg-slate-100 flex items-center justify-center mb-4">
                            <Upload className="w-8 h-8 text-slate-300" />
                        </div>
                        <h4 className="text-sm font-bold text-slate-500 mb-1">Nenhuma imagem encontrada</h4>
                        <p className="text-xs text-slate-400">Faça upload de imagens ou mude a categoria</p>
                    </div>
                )}
            </div>

            {/* Image Viewer Modal */}
            <Dialog open={!!viewImage} onOpenChange={() => setViewImage(null)}>
                <DialogContent className="max-w-4xl p-0 overflow-hidden">
                    {viewImage && (
                        <div className="relative">
                            <img
                                src={viewImage.url}
                                alt={viewImage.description}
                                crossOrigin="anonymous"
                                className="w-full h-auto max-h-[80vh] object-contain bg-black"
                            />
                            <div className="absolute bottom-0 left-0 right-0 bg-gradient-to-t from-black/80 to-transparent p-6">
                                <p className="text-white font-medium">{viewImage.description}</p>
                                <p className="text-white/60 text-sm">{viewImage.author} • {formatDate(viewImage.created_at?.split('T')[0])}</p>
                            </div>
                            <div className="absolute top-4 right-4 flex gap-2">
                                <Button size="sm" variant="secondary" className="gap-2" onClick={() => onSendToChat?.(viewImage.url)}>
                                    <Send className="w-3.5 h-3.5" /> Enviar ao Chat
                                </Button>
                                <Button size="sm" variant="secondary" className="gap-2">
                                    <FileText className="w-3.5 h-3.5" /> Vincular ao Prontuário
                                </Button>
                            </div>
                        </div>
                    )}
                </DialogContent>
            </Dialog>

            {/* Editor Modal - Expandido verticalmente */}
            <Dialog open={isEditorOpen} onOpenChange={setIsEditorOpen}>
                <DialogContent className="max-w-6xl h-[90vh] p-0 overflow-hidden flex flex-col">
                    <DialogHeader className="p-4 border-b shrink-0">
                        <DialogTitle className="flex items-center gap-2">
                            <Pencil className="w-4 h-4" />
                            Editor de Marcação
                        </DialogTitle>
                        <DialogDescription>
                            Desenhe setas, círculos ou adicione texto sobre a imagem
                        </DialogDescription>
                    </DialogHeader>

                    <div className="flex flex-1 min-h-0">
                        {/* Tools Sidebar */}
                        <div className="w-20 bg-slate-50 border-r p-3 flex flex-col gap-3">
                            <span className="text-[9px] font-bold text-slate-400 uppercase tracking-wider text-center">Ferramentas</span>
                            <button
                                className={cn("w-14 h-14 rounded-xl flex items-center justify-center transition-all text-xl", drawTool === 'arrow' ? 'bg-primary text-white shadow-lg' : 'bg-white border-2 hover:bg-slate-100 hover:border-primary/50')}
                                onClick={() => setDrawTool('arrow')}
                                title="Seta"
                            >
                                →
                            </button>
                            <button
                                className={cn("w-14 h-14 rounded-xl flex items-center justify-center transition-all text-xl", drawTool === 'circle' ? 'bg-primary text-white shadow-lg' : 'bg-white border-2 hover:bg-slate-100 hover:border-primary/50')}
                                onClick={() => setDrawTool('circle')}
                                title="Círculo"
                            >
                                ○
                            </button>
                            <button
                                className={cn("w-14 h-14 rounded-xl flex items-center justify-center transition-all text-base font-bold", drawTool === 'text' ? 'bg-primary text-white shadow-lg' : 'bg-white border-2 hover:bg-slate-100 hover:border-primary/50')}
                                onClick={() => setDrawTool('text')}
                                title="Texto"
                            >
                                Aa
                            </button>

                            <div className="border-t my-2" />

                            <span className="text-[9px] font-bold text-slate-400 uppercase tracking-wider text-center">Cores</span>

                            {/* Colors - Expandido com mais opções */}
                            <div className="grid grid-cols-2 gap-2">
                                {[
                                    { color: '#ef4444', name: 'Vermelho' },
                                    { color: '#3b82f6', name: 'Azul' },
                                    { color: '#ffffff', name: 'Branco' },
                                    { color: '#1e293b', name: 'Preto' },
                                    { color: '#22c55e', name: 'Verde' },
                                    { color: '#a855f7', name: 'Lilás' },
                                    { color: '#eab308', name: 'Amarelo' },
                                ].map(({ color, name }) => (
                                    <button
                                        key={color}
                                        className={cn(
                                            "w-7 h-7 rounded-full mx-auto border-2 transition-all hover:scale-110",
                                            drawColor === color ? 'ring-2 ring-offset-2 ring-primary scale-110' : ''
                                        )}
                                        style={{
                                            backgroundColor: color,
                                            borderColor: color === '#ffffff' ? '#e2e8f0' : color === '#1e293b' ? '#475569' : color
                                        }}
                                        onClick={() => setDrawColor(color)}
                                        title={name}
                                    />
                                ))}
                            </div>
                        </div>

                        {/* Canvas Area - Maior */}
                        <div className="flex-1 p-6 bg-slate-100 flex items-center justify-center overflow-auto">
                            {!canvasLoaded && (
                                <div className="absolute inset-0 flex items-center justify-center bg-slate-100 z-10">
                                    <div className="flex flex-col items-center gap-3">
                                        <div className="w-8 h-8 border-4 border-primary border-t-transparent rounded-full animate-spin" />
                                        <span className="text-sm font-medium text-slate-500">Carregando imagem...</span>
                                    </div>
                                </div>
                            )}
                            <canvas
                                ref={canvasRef}
                                className={cn(
                                    "shadow-xl rounded-xl cursor-crosshair bg-white transition-opacity",
                                    canvasLoaded ? 'opacity-100' : 'opacity-0'
                                )}
                                onMouseDown={handleCanvasMouseDown}
                                onMouseUp={handleCanvasMouseUp}
                            />
                        </div>
                    </div>

                    <DialogFooter className="p-4 border-t bg-slate-50 shrink-0">
                        <Button variant="outline" onClick={() => setIsEditorOpen(false)}>
                            Cancelar
                        </Button>
                        <Button className="bg-[#59A52C] hover:bg-[#4d8f26] gap-2" onClick={saveEditedImage}>
                            <Download className="w-4 h-4" />
                            Salvar Marcação
                        </Button>
                    </DialogFooter>
                </DialogContent>
            </Dialog>

            {/* Compare/Export Modal */}
            <Dialog open={isCompareOpen} onOpenChange={setIsCompareOpen}>
                <DialogContent className="max-w-3xl">
                    <DialogHeader>
                        <DialogTitle className="flex items-center gap-2">
                            <Layers className="w-4 h-4" />
                            Gerar Antes e Depois
                        </DialogTitle>
                        <DialogDescription>
                            Selecione o formato e exporte para suas redes sociais
                        </DialogDescription>
                    </DialogHeader>

                    <div className="py-6">
                        {/* Preview */}
                        <div className="flex gap-4 items-center justify-center mb-6">
                            {selectedImages.map((id, idx) => {
                                const img = images.find(i => i.id === id);
                                return (
                                    <div key={id} className="relative">
                                        <img
                                            src={img?.thumbnail}
                                            alt=""
                                            className="w-40 h-40 object-cover rounded-lg shadow-md"
                                        />
                                        <Badge className="absolute -bottom-2 left-1/2 -translate-x-1/2 bg-slate-800">
                                            {idx === 0 ? 'ANTES' : 'DEPOIS'}
                                        </Badge>
                                    </div>
                                );
                            })}
                        </div>

                        {/* Format Buttons */}
                        <div className="flex gap-4 justify-center">
                            <Button
                                variant="outline"
                                className="h-20 w-40 flex-col gap-2 border-2 hover:border-primary hover:bg-primary/5"
                                onClick={() => generateBeforeAfter('feed')}
                            >
                                <span className="text-2xl">📸</span>
                                <span className="text-xs font-bold">Feed (1080x1350)</span>
                            </Button>
                            <Button
                                variant="outline"
                                className="h-20 w-40 flex-col gap-2 border-2 hover:border-primary hover:bg-primary/5"
                                onClick={() => generateBeforeAfter('stories')}
                            >
                                <span className="text-2xl">📱</span>
                                <span className="text-xs font-bold">Stories (1080x1920)</span>
                            </Button>
                        </div>
                    </div>

                    <DialogFooter>
                        <Button variant="outline" onClick={() => { setIsCompareOpen(false); clearSelection(); }}>
                            Cancelar
                        </Button>
                    </DialogFooter>
                </DialogContent>
            </Dialog>
            {/* Modal de Confirmação de Exclusão */}
            <Dialog open={!!imageToDelete} onOpenChange={(open) => !open && setImageToDelete(null)}>
                <DialogContent className="max-w-[400px]">
                    <DialogHeader>
                        <DialogTitle className="flex items-center gap-2 text-red-600">
                            <Trash2 className="w-5 h-5" />
                            Confirmar Exclusão
                        </DialogTitle>
                        <DialogDescription className="pt-2">
                            Tem certeza que deseja excluir esta foto? Esta ação não pode ser desfeita.
                        </DialogDescription>
                    </DialogHeader>
                    {imageToDelete && (
                        <div className="flex justify-center py-4">
                            <img src={imageToDelete.thumbnail} alt="" crossOrigin="anonymous" className="w-32 h-32 object-cover rounded-lg border-2 border-red-100" />
                        </div>
                    )}
                    <DialogFooter className="gap-2 sm:gap-0">
                        <Button variant="outline" onClick={() => setImageToDelete(null)}>
                            Cancelar
                        </Button>
                        <Button
                            className="bg-red-600 hover:bg-red-700 text-white"
                            onClick={() => imageToDelete && deleteImage(imageToDelete)}
                        >
                            Excluir Permanentemente
                        </Button>
                    </DialogFooter>
                </DialogContent>
            </Dialog>
        </div>
    );
}
