        } else if (periodFilter === 'yesterday') {
            const yesterdayStart = startOfDay(subDays(now, 1));
            const yesterdayEnd = endOfDay(subDays(now, 1));
            result = result.filter(p => {
                const createdDate = new Date(p.createdAt);
                return createdDate >= yesterdayStart && createdDate <= yesterdayEnd;
            });
        } else if (periodFilter === 'thisWeek') {
            const weekStart = startOfWeek(now, { weekStartsOn: 0 }); // Sunday
            result = result.filter(p => {
                const createdDate = new Date(p.createdAt);
                return createdDate >= weekStart;
            });
        } else if (periodFilter === 'lastWeek') {
            const lastWeekStart = startOfWeek(subDays(now, 7), { weekStartsOn: 0 });
            const lastWeekEnd = endOfWeek(subDays(now, 7), { weekStartsOn: 0 });
            result = result.filter(p => {
                const createdDate = new Date(p.createdAt);
                return createdDate >= lastWeekStart && createdDate <= lastWeekEnd;
            });
        } else if (periodFilter === 'last14days') {
            const fourteenDaysAgo = subDays(now, 14);
            result = result.filter(p => {
                const createdDate = new Date(p.createdAt);
                return createdDate >= fourteenDaysAgo;
            });
        } else if (periodFilter === 'thisMonth') {
            const startOfThisMonth = startOfMonth(now);
            result = result.filter(p => {
                const createdDate = new Date(p.createdAt);
                return createdDate >= startOfThisMonth;
            });
        } else if (periodFilter === 'lastMonth') {
            const lastMonthStart = startOfMonth(subMonths(now, 1));
            const lastMonthEnd = endOfMonth(subMonths(now, 1));
            result = result.filter(p => {
                const createdDate = new Date(p.createdAt);
                return createdDate >= lastMonthStart && createdDate <= lastMonthEnd;
            });
        } else if (periodFilter === 'thisYear') {
            const yearStart = startOfYear(now);
            result = result.filter(p => {
                const createdDate = new Date(p.createdAt);
                return createdDate >= yearStart;
            });
        } else if (periodFilter === '365days') {
            const oneYearAgo = subMonths(now, 12);
            result = result.filter(p => {
                const createdDate = new Date(p.createdAt);
                return createdDate >= oneYearAgo;
            });
        } else if (periodFilter === 'custom') {
            // Custom date range
            result = result.filter(p => {
                const createdDate = new Date(p.createdAt);
                return createdDate >= startOfDay(dateRange.from) && createdDate <= endOfDay(dateRange.to);
            });
        }
        // If 'all', no date filtering

        return result;
    }, [activeTab, searchTerm, procedureFilter, periodFilter, birthdayMonthFilter, patients, fiveMonthsAgo, twelveMonthsAgo, now, dateRange]);

        const closedLeadsWithCycle = sourceData.filter(l => typeof l.sales_cycle_days === 'number');

            : "0.0";

        const recurrencePending = sourceData.filter(p => {
            if (!p.lastVisitAt) return false;
            const monthsDiff = differenceInMonths(now, new Date(p.lastVisitAt));
            return monthsDiff >= 4 && monthsDiff <= 6;
        });

        return {
            baseAtiva: activePatients.length,
            ltvMedio: avgLtv,
            ticketMedioTotal: totalLtv,
            recorrenciaPendente: recurrencePending.length,
            avgSalesCycle: avgCycle
        };

                                                    {tierConfig.label}
                                                </Badge>
                                                {pendingSignatureLeadIds.has(patient.id) && (
                                                    <Badge
                                                        variant="outline"
                                                        className="text-[10px] font-bold px-2 py-0.5 bg-amber-50 border-amber-200 text-amber-700"
                                                    >
                                                        <FileText className="w-3 h-3 mr-1" />
                                                        Aguardando Assinatura
                                                    </Badge>
                                                )}
                                            </div>
                                        </TableCell>
                                        <TableCell>
                                            <div className="flex flex-col">
                                                <span className="text-sm text-slate-700 font-medium">
                                                    {format(new Date(patient.createdAt), "dd MMM yyyy", { locale: ptBR })}
                                                </span>
                                                <span className="text-xs text-slate-400">
                                                    {formatDistanceToNow(new Date(patient.createdAt), { addSuffix: true, locale: ptBR })}
                                                </span>
                                            </div>
                                        </TableCell>
                                        <TableCell>
                                            {patient.lastProcedure ? (
                                                <div className="flex flex-col">
                                                    <span className="text-sm text-slate-700 font-medium">
                                                        {patient.lastProcedure.name}
                                                    </span>
                                                    <span className="text-xs text-slate-400">
                                                        {format(new Date(patient.lastProcedure.date), "dd MMM yyyy", { locale: ptBR })}
                                                    </span>
                                                </div>
                                            ) : (
                                                <span className="text-xs text-slate-400">—</span>
                                            )}
                                        </TableCell>
                                        <TableCell className="text-center">
                                            <span className="text-sm font-bold text-slate-800 font-space">
                                                {formatCurrency(patient.ltv)}
                                            </span>
                                        </TableCell>
                                        <TableCell className="text-center">
                                            {(patient.outstandingBalance || 0) > 0 ? (
                                                <span className="text-sm font-bold text-red-600 font-space">
                                                    {formatCurrency(patient.outstandingBalance!)}
                                                </span>
                                            ) : (
                                                <span className="text-xs text-slate-400">—</span>
                                            )}
                                        </TableCell>
                                        <TableCell>
                                            {patient.nextAppointment ? (
                                                <div
                                                    className="flex flex-col cursor-pointer group"
                                                    onClick={(e) => {
                                                        e.stopPropagation();
                                                        if (patient.nextAppointmentId) {
                                                            navigate('/calendar', {
                                                                state: {
                                                                    openAppointmentId: patient.nextAppointmentId
                                                                }
                                                            });
                                                        }
                                                    }}
                                                >
                                                    <span className="text-xs font-bold text-emerald-600 group-hover:text-emerald-700 transition-colors flex items-center gap-1">
                                                        <CalendarDays className="w-3 h-3" />
                                                        {format(new Date(patient.nextAppointment), "dd MMM yyyy", { locale: ptBR })}
                                                    </span>
                                                    <span className="text-[10px] text-slate-400 font-medium group-hover:text-slate-500">
                                                        Faltam {differenceInDays(new Date(patient.nextAppointment), new Date())} dias
                                                    </span>
                                                </div>
                                            ) : (
                                                <Badge variant="outline" className="bg-amber-50 text-amber-600 border-amber-200 hover:bg-amber-100 font-normal shadow-sm">
                                                    Sem Agendamento
                                                </Badge>
                                            )}
                                        </TableCell>
                                        <TableCell onClick={e => e.stopPropagation()}>
                                            <DropdownMenu>
                                                <DropdownMenuTrigger asChild>
                                                    <Button variant="ghost" size="icon" className="h-8 w-8">
                                                        <MoreHorizontal className="h-4 w-4 text-slate-400" />
                                                    </Button>
                                                </DropdownMenuTrigger>
                                                <DropdownMenuContent align="end" className="w-48">
                                                    <DropdownMenuItem asChild>
                                                        <a
                                                            href={`/crm/${patient.stage || 'Venda Realizada'}/${patient.id}?tab=prontuario`}
                                                            onClick={(e) => {
                                                                e.preventDefault();
                                                                const stage = patient.stage || 'Venda Realizada';
                                                                navigate(`/crm/${stage}/${patient.id}?tab=prontuario`);
                                                            }}
                                                            className="flex items-center gap-2 text-sm cursor-pointer w-full"
                                                        >
                                                            <FileText className="h-4 w-4" />
                                                            Ver Prontuário
                                                        </a>
                                                    </DropdownMenuItem>
                                                    <DropdownMenuItem asChild>
                                                        <a
                                                            href="/calendar"
                                                            onClick={(e) => {
                                                                e.preventDefault();
                                                                navigate('/calendar', {
                                                                    state: {
                                                                        openAddModal: true,
                                                                        leadId: patient.id,
                                                                        leadName: patient.name
                                                                    }
                                                                });
                                                            }}
                                                            className="flex items-center gap-2 text-sm cursor-pointer w-full"
                                                        >
                                                            <CalendarDays className="h-4 w-4" />
                                                            Agendar Retorno
                                                        </a>
                                                    </DropdownMenuItem>
                                                </DropdownMenuContent>
                                            </DropdownMenu>
                                        </TableCell>
                                    </TableRow>
                                );
                            })}

                            {paginatedPatients.length === 0 && (
                                <TableRow>
                                    <TableCell colSpan={8} className="h-32 text-center">
                                        <p className="text-sm text-slate-400">Nenhum paciente encontrado com os filtros selecionados.</p>
                                    </TableCell>
                                </TableRow>
                            )}
                        </TableBody>
                    </Table>

                    {/* Pagination Footer */}
                    {filteredPatients.length > ITEMS_PER_PAGE && (
                        <div className="flex items-center justify-between px-4 py-3 border-t border-slate-200 bg-slate-50/50">
                            <p className="text-sm text-slate-500">
                                Exibindo <span className="font-semibold text-slate-700">{startItem}</span> a{" "}
                                <span className="font-semibold text-slate-700">{endItem}</span> de{" "}
                                <span className="font-semibold text-slate-700">{filteredPatients.length}</span> pacientes
                            </p>
                            <div className="flex items-center gap-2">
                                <Button
                                    variant="outline"
                                    size="sm"
                                    onClick={() => setCurrentPage(prev => Math.max(prev - 1, 1))}
                                    disabled={currentPage === 1}
                                    className="h-8 px-3 text-xs gap-1"
                                >
                                    <ChevronLeft className="h-4 w-4" />
                                    Anterior
                                </Button>
                                <span className="text-sm text-slate-600 px-2">
                                    Página {currentPage} de {totalPages}
                                </span>
                                <Button
                                    variant="outline"
                                    size="sm"
                                    onClick={() => setCurrentPage(prev => Math.min(prev + 1, totalPages))}
                                    disabled={currentPage === totalPages}
                                    className="h-8 px-3 text-xs gap-1"
                                >
                                    Próximo
                                    <ChevronRight className="h-4 w-4" />
                                </Button>
                            </div>
                        </div>
                    )}
                </div>
            </div>

            {/* Bulk Action Bar */}
            {selectedPatients.size >= 2 && (
                <div className="fixed bottom-6 left-1/2 -translate-x-1/2 z-50">
                    <div className="bg-slate-900 text-white px-6 py-3 rounded-full shadow-2xl flex items-center gap-4 animate-in slide-in-from-bottom-4 duration-300">
                        <div className="flex items-center gap-2">
                            <Check className="w-4 h-4 text-emerald-400" />
                            <span className="text-sm font-medium">
                                {selectedPatients.size} pacientes selecionados
                            </span>
                        </div>
                        <div className="w-px h-5 bg-slate-700" />
                        <Button
                            size="sm"
                            className="gap-2 bg-emerald-600 hover:bg-emerald-700 text-white h-8 px-4 text-xs font-semibold"
                            onClick={() => {
                                alert(`Enviando campanha para ${selectedPatients.size} pacientes...`);
                            }}
                        >
                            <Send className="h-3.5 w-3.5" />
                            Enviar Campanha de Retorno
                        </Button>
                        <Button
                            size="sm"
                            variant="ghost"
                            className="text-slate-400 hover:text-white h-8 px-3 text-xs"
                            onClick={() => setSelectedPatients(new Set())}
                        >
                            Cancelar
                        </Button>
                    </div>
                </div>
            )}
        </div>
    );
}
