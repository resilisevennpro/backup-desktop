import { useState, useEffect, useRef } from 'react';
import { useParams } from 'react-router-dom';
import { supabase } from '@/lib/supabase';
import { Button } from '@/components/ui/button';
import { Card, CardContent, CardHeader, CardTitle, CardDescription } from '@/components/ui/card';
import { Check, Eraser, AlertCircle, FileText, Lock, Clock } from 'lucide-react';
import { toast, Toaster } from 'sonner';

export default function SignDocumentPage() {
    const { token } = useParams();
    const [document, setDocument] = useState<any>(null);
    const [loading, setLoading] = useState(true);
    const [error, setError] = useState<string | null>(null);
    const [success, setSuccess] = useState(false);

    // Canvas State
    const canvasRef = useRef<HTMLCanvasElement>(null);
    const [isDrawing, setIsDrawing] = useState(false);
    const [hasSignature, setHasSignature] = useState(false);

    useEffect(() => {
        const fetchDocument = async () => {
            if (!token) {
                setError('Link inválido.');
                setLoading(false);
                return;
            }

            try {
                const { data, error } = await supabase
                    .from('clinic_documents')
                    .select('*')
                    .eq('public_token', token)
                    .single();

                if (error || !data) {
                    setError('Documento não encontrado.');
                    setLoading(false);
                    return;
                }

                if (data.status === 'signed') {
                    setError('Este documento já foi assinado.');
                    setLoading(false);
                    return;
                }

                if (data.expires_at && new Date(data.expires_at) < new Date()) {
                    setError('Este link de assinatura expirou. Solicite um novo.');
                    setLoading(false);
                    return;
                }

                setDocument(data);

            } catch (err) {
                console.error(err);
                setError('Erro ao carregar documento.');
            } finally {
                setLoading(false);
            }
        };

        fetchDocument();
    }, [token]);

    // Canvas Logic
    useEffect(() => {
        if (!loading && !error && !success) {
            // Init canvas
            setTimeout(() => {
                const canvas = canvasRef.current;
                if (canvas) {
                    const ratio = Math.max(window.devicePixelRatio || 1, 1);
                    canvas.width = canvas.offsetWidth * ratio;
                    canvas.height = canvas.offsetHeight * ratio;
                    const ctx = canvas.getContext('2d');
                    if (ctx) ctx.scale(ratio, ratio);
                }
            }, 500);
        }
    }, [loading, error, success]);

    const getTouchPos = (canvasDom: HTMLCanvasElement, touchEvent: React.TouchEvent) => {
        const rect = canvasDom.getBoundingClientRect();
        return {
            x: touchEvent.touches[0].clientX - rect.left,
            y: touchEvent.touches[0].clientY - rect.top
        };
    };

    const getMousePos = (canvasDom: HTMLCanvasElement, mouseEvent: React.MouseEvent) => {
        const rect = canvasDom.getBoundingClientRect();
        return {
            x: mouseEvent.clientX - rect.left,
            y: mouseEvent.clientY - rect.top
        };
    };

    const startDraw = (e: React.MouseEvent | React.TouchEvent) => {
        setIsDrawing(true);
        const canvas = canvasRef.current;
        if (!canvas) return;
        const ctx = canvas.getContext('2d');
        if (!ctx) return;

        ctx.strokeStyle = "#000";
        ctx.lineWidth = 2;
        ctx.lineJoin = "round";
        ctx.lineCap = "round";

        const pos = 'touches' in e ? getTouchPos(canvas, e as React.TouchEvent) : getMousePos(canvas, e as React.MouseEvent);
        ctx.beginPath();
        ctx.moveTo(pos.x, pos.y);
    };

    const draw = (e: React.MouseEvent | React.TouchEvent) => {
        if (!isDrawing) return;
        const canvas = canvasRef.current;
        if (!canvas) return;
        const ctx = canvas.getContext('2d');
        if (!ctx) return;

        e.preventDefault();
        const pos = 'touches' in e ? getTouchPos(canvas, e as React.TouchEvent) : getMousePos(canvas, e as React.MouseEvent);
        ctx.lineTo(pos.x, pos.y);
        ctx.stroke();
        if (!hasSignature) setHasSignature(true);
    };

    const endDraw = () => {
        setIsDrawing(false);
    };

    const clearSignature = () => {
        const canvas = canvasRef.current;
        if (canvas) {
            const ctx = canvas.getContext('2d');
            ctx?.clearRect(0, 0, canvas.width, canvas.height);
            setHasSignature(false);
        }
    };

    const handleSubmit = async () => {
        if (!hasSignature || !document || !canvasRef.current) return;

        try {
            setLoading(true);
            const canvas = canvasRef.current;
            const blob = await new Promise<Blob | null>(resolve => canvas.toBlob(resolve, 'image/png'));

            if (!blob) throw new Error('Falha na imagem');

            const fileName = `signatures/${document.id}_${Date.now()}.png`;

            // Upload
            const { error: uploadError } = await supabase.storage
                .from('lead-images')
                .upload(fileName, blob);

            if (uploadError) throw uploadError;

            // Get Public URL
            const { data: { publicUrl } } = supabase.storage
                .from('lead-images')
                .getPublicUrl(fileName);

            const now = new Date().toISOString();

            // Update Doc
            const { error: dbError } = await supabase
                .from('clinic_documents')
                .update({
                    status: 'signed',
                    signature_url: publicUrl,
                    signed_at: now
                })
                .eq('id', document.id);

            if (dbError) throw dbError;

            setSuccess(true);
            setLoading(false);

        } catch (error) {
            console.error(error);
            toast.error('Erro ao salvar assinatura. Tente novamente.');
            setLoading(false);
        }
    };

    if (loading && !document && !error) {
        return (
            <div className="min-h-screen flex items-center justify-center bg-slate-50">
                <div className="animate-spin rounded-full h-12 w-12 border-b-2 border-primary"></div>
            </div>
        );
    }

    if (error) {
        return (
            <div className="min-h-screen flex items-center justify-center bg-slate-50 p-4">
                <Card className="w-full max-w-md border-destructive/20 bg-destructive/5 text-center">
                    <CardHeader className="flex flex-col items-center gap-2">
                        <AlertCircle className="w-12 h-12 text-destructive" />
                        <CardTitle className="text-destructive">Acesso Indisponível</CardTitle>
                        <CardDescription className="text-slate-600 font-medium">{error}</CardDescription>
                    </CardHeader>
                    <CardContent>
                        <p className="text-sm text-slate-500">Entre em contato com a clínica se acreditar que isso é um erro.</p>
                    </CardContent>
                </Card>
            </div>
        );
    }

    if (success) {
        return (
            <div className="min-h-screen flex items-center justify-center bg-slate-50 p-4">
                <Card className="w-full max-w-md text-center border-emerald-200 bg-emerald-50">
                    <CardHeader className="flex flex-col items-center gap-2">
                        <div className="bg-emerald-100 p-3 rounded-full">
                            <Check className="w-8 h-8 text-emerald-600" />
                        </div>
                        <CardTitle className="text-emerald-800">Assinado com Sucesso!</CardTitle>
                        <CardDescription className="text-emerald-700">
                            Agradecemos sua colaboração. O documento foi registrado em nosso sistema.
                        </CardDescription>
                    </CardHeader>
                    <CardContent>
                        <p className="text-sm text-emerald-600/80">Você já pode fechar esta página.</p>
                    </CardContent>
                </Card>
            </div>
        );
    }

    return (
        <div className="min-h-screen bg-slate-100 py-8 px-4 md:px-8">
            <Toaster position="top-center" />
            <div className="max-w-3xl mx-auto space-y-6">

                {/* Header */}
                <div className="bg-white rounded-xl shadow-sm p-6 border border-slate-200">
                    <div className="flex items-center gap-3 mb-2">
                        <div className="bg-primary/10 p-2 rounded-lg">
                            <FileText className="w-6 h-6 text-primary" />
                        </div>
                        <div>
                            <h1 className="text-xl font-bold text-slate-900">{document?.title}</h1>
                            <p className="text-sm text-slate-500 flex items-center gap-1">
                                <Clock className="w-3 h-3" /> Criado em {new Date(document?.created_at).toLocaleDateString()}
                            </p>
                        </div>
                    </div>
                </div>

                {/* Document Content */}
                <div className="flex justify-center">
                    <div
                        className="prose max-w-none w-full p-6 bg-white rounded-lg border shadow-sm text-slate-900"
                        dangerouslySetInnerHTML={{ __html: document?.content || '' }}
                    />
                </div>

                {/* Signature Block */}
                <Card className="border-2 border-primary/20 shadow-md">
                    <CardHeader className="bg-primary/5 pb-4">
                        <CardTitle className="text-primary flex items-center gap-2">
                            <Lock className="w-5 h-5" /> Assinatura Digital
                        </CardTitle>
                        <CardDescription>
                            Por favor, assine no quadro abaixo para concordar com o documento.
                        </CardDescription>
                    </CardHeader>
                    <CardContent className="pt-6">
                        <div className="border-2 border-dashed border-slate-300 rounded-xl bg-white overflow-hidden touch-none relative">
                            <canvas
                                ref={canvasRef}
                                className="w-full h-48 cursor-crosshair touch-none"
                                onMouseDown={startDraw}
                                onMouseMove={draw}
                                onMouseUp={endDraw}
                                onMouseLeave={endDraw}
                                onTouchStart={startDraw}
                                onTouchMove={draw}
                                onTouchEnd={endDraw}
                            />
                            {!hasSignature && (
                                <div className="absolute inset-0 flex items-center justify-center pointer-events-none opacity-20">
                                    <span className="text-2xl font-handwriting text-slate-400">Assine aqui</span>
                                </div>
                            )}
                        </div>

                        <div className="flex items-center justify-between mt-4">
                            <Button variant="ghost" size="sm" onClick={clearSignature} className="text-destructive gap-1 hover:text-destructive hover:bg-destructive/10">
                                <Eraser className="w-4 h-4" /> Limpar
                            </Button>
                            <Button
                                onClick={handleSubmit}
                                disabled={!hasSignature || loading}
                                className="bg-primary hover:bg-primary/90 text-white gap-2 px-8"
                            >
                                {loading && <div className="animate-spin rounded-full h-4 w-4 border-b-2 border-white"></div>}
                                Finalizar Assinatura
                            </Button>
                        </div>
                    </CardContent>
                </Card>

            </div>
        </div>
    );
}
