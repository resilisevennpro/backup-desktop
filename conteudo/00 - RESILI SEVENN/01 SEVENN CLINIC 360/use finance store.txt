import { create } from 'zustand';
import { supabase } from '@/lib/supabase';
import type { Transaction } from '@/components/finance/types';
import { addMonths, startOfMonth, differenceInCalendarMonths, format } from 'date-fns';
import { ptBR } from 'date-fns/locale';
import { toast } from "sonner";

// --- INTERFACES NOVAS (BI) ---
export interface FinanceMetrics {
    totalRevenue: number;
    totalProfit: number;
    totalCommissions: number;
    totalMaterials: number;
    totalFixedCosts: number; // Taxas + Marketing
    profitMargin: number;
    revenueGoal: number;
    marginGoal: number;
    totalCardFees: number;
    totalOperatingCosts: number;
    totalTaxes: number;
    marketingInvested: number;
}

export interface ProcedureProfit {
    name: string;
    revenue: number;
    cost: number;
    profit: number;
    margin: number;
}

export interface TopPatient {
    id: string;
    name: string;
    totalSpent: number;
}

// --- INTERFACES DE PROJEÇÃO E ANTECIPAÇÃO ---
export interface FinancialAnticipation {
    id: string;
    mes_referencia: string;
    valor_bruto_retirado: number;
    valor_taxa: number;
    valor_liquido_recebido: number;
    created_at?: string;
}

export interface FinancialProjection {
    monthlyData: {
        name: string; // Ex: 'Fev'
        fullDate: string; // ISO Date do dia 1 do mês
        value: number; // Valor Líquido Previsto
        originalValue: number; // Valor sem antecipação
    }[];
    quarterly: { q1: number; q2: number; q3: number; q4: number };
    kpis: { totalGuaranteed: number; peakMonth: string; monthlyAverage: number };
}

// --- STATE COMPLETO (CRUD + BI) ---
interface FinanceState {
    // Dados de Lançamentos (CRUD Antigo mantido)
    transactions: Transaction[];

    // Dados de BI (Novos)
    metrics: FinanceMetrics;
    profitByProcedure: ProcedureProfit[];
    topPatients: TopPatient[];
    expensesBreakdown: { commissions: number; materials: number; fixed: number };

    // Projeção Financeira & Antecipações
    financialProjection: FinancialProjection;
    anticipations: FinancialAnticipation[];

    // Estado Geral
    loading: boolean;

    // Ações de Lançamentos (Mantidas)
    fetchTransactions: (clinicId: string) => Promise<void>;
    addTransaction: (transaction: Omit<Transaction, 'id'> & { clinic_id: string, lead_id?: string }) => Promise<void>;
    updateTransaction: (id: string, updates: Partial<Transaction>) => Promise<void>;
    deleteTransaction: (id: string) => Promise<void>;

    // Ação de BI (Nova)
    fetchFinanceData: (clinicId: string, range: string, start?: Date, end?: Date, filterProfessionalId?: string, filterProcedure?: string) => Promise<void>;

    // Ação de Antecipação
    registerAnticipation: (clinicId: string, data: { targetMonth: Date; grossAmount: number; feeAmount: number; netAmount: number }) => Promise<void>;

    // Ação de Despesa
    addExpense: (clinicId: string, data: { description: string; amount: number; category: string; date: Date; status: 'paid' | 'pending' }) => Promise<void>;
}

export const useFinanceStore = create<FinanceState>((set, get) => ({
    // Inicialização
    transactions: [],
    metrics: { totalRevenue: 0, totalProfit: 0, totalCommissions: 0, totalMaterials: 0, totalFixedCosts: 0, profitMargin: 0, revenueGoal: 0, marginGoal: 20, totalCardFees: 0, totalOperatingCosts: 0, totalTaxes: 0, marketingInvested: 0 },
    profitByProcedure: [],
    topPatients: [],
    expensesBreakdown: { commissions: 0, materials: 0, fixed: 0 },
    financialProjection: {
        monthlyData: [],
        quarterly: { q1: 0, q2: 0, q3: 0, q4: 0 },
        kpis: { totalGuaranteed: 0, peakMonth: '-', monthlyAverage: 0 },
    },
    anticipations: [],
    loading: false,

    // --- FUNÇÕES CRUD (MANTIDAS DO ORIGINAL) ---

    fetchTransactions: async (clinicId: string) => {
        set({ loading: true });
        try {
            const { data, error } = await supabase
                .from('financial_transactions')
                .select(`
                    *,
                    leads (name, procedure, procedure_id, status),
                    professionals:professional_id ( id, name, commission_percentage )
                `)
                .eq('clinic_id', clinicId)
                // Ordena primeiro pela data da transação, depois pela data de criação (para desempatar o mesmo dia)
                .order('transaction_date', { ascending: false })
                .order('created_at', { ascending: false });

            if (error) throw error;

            const formattedTransactions: Transaction[] = data.map(t => {
                // 1. CORREÇÃO DE DATA (Lógica antes do retorno)
                // Se a data vier simples "YYYY-MM-DD", adicionamos T12:00:00
                const rawDate = t.transaction_date || "";
                const safeDate = rawDate.includes('T') ? rawDate : `${rawDate}T12:00:00`;

                // 2. EXTRAÇÃO DE DADOS DO PROFISSIONAL
                const profData = Array.isArray(t.professionals) ? t.professionals[0] : t.professionals;
                const profName = profData?.name || t.metadata?.professional_name || 'Não Atribuído';
                // Pega a porcentagem vinda da query nova
                const profCommission = profData?.commission_percentage || 0;

                // 3. NORMALIZAÇÃO VISUAL DO PAGAMENTO (Novo!)
                // Traduz o código do banco (ex: "debit") para o Visual (ex: "Débito")
                let finalPaymentMethod = t.payment_method || '';
                if (finalPaymentMethod) {
                    const pmLower = finalPaymentMethod.toLowerCase();

                    // Se encontrar "credit" ou "crédito", transforma na chave 'credit' (Ambar)
                    if (pmLower.includes('credit') || pmLower.includes('crédito') || pmLower.includes('credito')) {
                        finalPaymentMethod = 'credit';
                    }
                    // Se encontrar "debit" ou "débito", transforma na chave 'debit' (Azul)
                    else if (pmLower === 'debit' || pmLower.includes('debit') || pmLower.includes('débito')) {
                        finalPaymentMethod = 'debit';
                    }
                    // Se encontrar "pix", transforma na chave 'pix' (Verde Água)
                    else if (pmLower.includes('pix')) {
                        finalPaymentMethod = 'pix';
                    }
                    // Se encontrar "money", transforma na chave 'money' (Verde Escuro)
                    else if (pmLower === 'money' || pmLower.includes('dinheiro') || pmLower.includes('cash')) {
                        finalPaymentMethod = 'money';
                    }
                    // --- OUTROS MÉTODOS ---
                    else if (pmLower.includes('boleto')) {
                        finalPaymentMethod = 'boleto'; // (Geralmente Roxo ou Cinza Escuro)
                    }
                    else if (pmLower.includes('partnership') || pmLower.includes('parceria')) {
                        finalPaymentMethod = 'partnership'; // (Pode ser Rosa ou Laranja)
                    }
                    else if (pmLower.includes('other') || pmLower.includes('outro')) {
                        finalPaymentMethod = 'other'; // Cinza
                    }
                }
                // 4. RETORNO DO OBJETO
                return {
                    id: t.id,
                    clinic_id: t.clinic_id,
                    lead_id: t.lead_id,
                    description: t.description,
                    amount: t.amount,
                    type: t.type,
                    category: t.category,
                    transaction_date: safeDate, // Data corrigida para visualização
                    status: t.status,
                    patient: t.leads?.name || t.metadata?.patient_name || 'Cliente Avulso',

                    professional: profName,
                    professional_id: t.professional_id,
                    commission_rate: profCommission, // <--- O Campo que faltava para a aba Comissões funcionar!
                    lead_status: t.leads?.status, // status do lead para o link

                    procedure: t.leads?.procedure || t.metadata?.procedure_name || t.description,
                    payment_method: finalPaymentMethod, // <--- Alteração pra correção
                    installments: t.installments,
                    notes: t.notes,
                    created_at: t.created_at,
                    is_edited: t.is_edited
                };
            });

            set({ transactions: formattedTransactions, loading: false });
        } catch (error) {
            console.error('Erro ao buscar transações:', error);
            set({ loading: false });
        }
    },

    addTransaction: async (transaction) => {
        const { clinic_id, lead_id, ...data } = transaction;
        try {
            const dbPayload = {
                clinic_id,
                lead_id: lead_id || null,
                description: data.description,
                amount: data.amount,
                type: data.type,
                category: data.category,
                transaction_date: data.transaction_date,
                status: data.status,
                notes: data.notes,
                payment_method: data.payment_method, // Garantindo que salve o método
                metadata: {
                    patient_name: data.patient,
                    procedure_name: data.procedure,
                    payment_method: data.payment_method,
                    professional_id: data.professional
                }
            };
            const { data: inserted, error } = await supabase
                .from('financial_transactions')
                .insert([dbPayload])
                .select()
                .single();
            if (error) throw error;

            const newTransaction: Transaction = { id: inserted.id, ...data };
            set(state => ({ transactions: [newTransaction, ...state.transactions] }));
        } catch (error) {
            console.error('Erro ao adicionar transação:', error);
            alert('Erro ao salvar transação.');
        }
    },

    addExpense: async (clinicId, data) => {
        set({ loading: true });
        try {
            const { error } = await supabase.from('financial_transactions').insert([{
                clinic_id: clinicId,
                description: data.description,
                amount: data.amount,
                category: data.category,
                type: 'expense',
                payment_method: 'money',
                status: data.status,
                transaction_date: data.date.toISOString(),
                installments: 1,
                created_at: new Date().toISOString()
            }]);

            if (error) throw error;

            toast.success('Despesa registrada com sucesso');
            await get().fetchTransactions(clinicId);
            // Tenta atualizar BI se possível (range padrão 'this_month')
            await get().fetchFinanceData(clinicId, 'this_month');
        } catch (error) {
            console.error('Erro ao adicionar despesa:', error);
            toast.error('Erro ao adicionar despesa');
        } finally {
            set({ loading: false });
        }
    },

    updateTransaction: async (id, updates) => {
        try {
            // CORREÇÃO CRÍTICA: Agora aceitamos todos os campos, não apenas o status.
            const dbUpdates: any = {};

            // Mapeia explicitamente o que pode ser atualizado
            if (updates.status) dbUpdates.status = updates.status;
            if (updates.payment_method) dbUpdates.payment_method = updates.payment_method; // <--- O culpado era a falta disso!
            if (updates.installments) dbUpdates.installments = updates.installments;
            if (updates.notes) dbUpdates.notes = updates.notes;
            if (updates.payment_date) dbUpdates.payment_date = updates.payment_date;

            // Se virou pago, marca a data exata do pagamento
            if (updates.status === 'paid') {
                dbUpdates.paid_at = updates.payment_date || new Date().toISOString();
            }

            const { error } = await supabase.from('financial_transactions').update(dbUpdates).eq('id', id);
            if (error) throw error;

            // Atualiza o estado local imediatamente (para a tela piscar azul na hora)
            set(state => ({
                transactions: state.transactions.map(t => t.id === id ? { ...t, ...updates } : t)
            }));
        } catch (error) {
            console.error('Erro ao atualizar transação:', error);
            // toast.error('Erro ao atualizar transação.');
        }
    },

    deleteTransaction: async (id: string) => {
        try {
            const { error } = await supabase.from('financial_transactions').delete().eq('id', id);
            if (error) throw error;
            set(state => ({ transactions: state.transactions.filter(t => t.id !== id) }));
        } catch (error) {
            console.error('Erro ao deletar transação:', error);
        }
    },

    // --- NOVA FUNÇÃO INTELIGENTE (BI & DASHBOARD FINANCEIRO) ---

    fetchFinanceData: async (clinicId, range, start, end, filterProfessionalId, filterProcedure) => {
        set({ loading: true });
        try {
            const startDate = start ? start.toISOString() : new Date(new Date().getFullYear(), new Date().getMonth(), 1).toISOString();
            const endDate = end ? end.toISOString() : new Date().toISOString();

            // 1. Configurações Globais (Marketing & Metas)
            const { data: clinicConfig } = await supabase
                .from('clinics')
                .select('meta_faturamento, verba_marketing, meta_margem_liquida, imposto_aliquota')
                .eq('id', clinicId)
                .single();

            // 2. Taxas da Maquininha
            const { data: fees } = await supabase
                .from('config_taxas_pagamento')
                .select('metodo_pagamento, taxa_percentual')
                .eq('clinic_id', clinicId);
            const feeMap = new Map(fees?.map(f => [f.metodo_pagamento, Number(f.taxa_percentual)]) || []);

            // 3. Profissionais (Novos campos para Regra Híbrida)
            const { data: professionals } = await supabase
                .from('professionals')
                .select('id, name, commission_percentage, tipo_atuacao, descontar_taxas, descontar_material')
                .eq('clinic_id', clinicId);

            // 4. Serviços (Custo de Material)
            const { data: services } = await supabase
                .from('services')
                .select('name, custo_material')
                .eq('clinic_id', clinicId);

            // Criando mapa de Custo de Material por nome do procedimento
            const materialMap = new Map(services?.map(s => [s.name, Number(s.custo_material)]) || []);

            // 5. Query Inteligente de Transações
            let query = supabase
                .from('financial_transactions')
                .select(`
                    id, amount, description, payment_method, created_at, installments,
                    type, category, status, transaction_date, lead_id, metadata,
                    professional_id, commission_status, commission_paid_at,
                    leads!left(name, procedure, status)
                `)
                .eq('clinic_id', clinicId)
                .gte('transaction_date', startDate)
                .lte('transaction_date', endDate)
                .order('transaction_date', { ascending: false });

            if (filterProfessionalId) {
                query = query.eq('professional_id', filterProfessionalId);
            }

            const { data: rawTransactions } = await query;

            // 6. CÁLCULOS MATEMÁTICOS (Com a nova lógica Híbrida)
            let totalRevenue = 0;
            let totalCommissions = 0;
            let totalMaterials = 0;
            let totalCardFees = 0;

            const procedureStats: Record<string, { revenue: number; cost: number }> = {};
            const patientStats: Record<string, number> = {};

            // Mapeando para o formato Transaction e enriquecendo
            const enrichedTransactions = rawTransactions?.map((t: any) => {
                // Filtro Manual de Procedimento
                const procName = t.metadata?.procedure_name || t.leads?.procedure || t.description || 'Outros';

                const amount = Number(t.amount) || 0;

                // --- NOVA LÓGICA DE COMISSÃO LINHA A LINHA ---

                // A. Identificar o Profissional
                const professional = professionals?.find(p => p.id === t.professional_id);

                // B. Calcular Taxa de Cartão (Lógica de Parcelas)
                const method = t.payment_method || 'pix';
                let feePercent = 0;
                if (method === 'credito_parcelado' || method === 'credito') {
                    const installments = Number(t.installments) || 1;
                    if (installments === 1) feePercent = feeMap.get('credito_1x') || 0;
                    else if (installments >= 2 && installments <= 4) feePercent = feeMap.get('credito_2x_4x') || 0;
                    else if (installments >= 5 && installments <= 8) feePercent = feeMap.get('credito_5x_8x') || 0;
                    else if (installments >= 9) feePercent = feeMap.get('credito_9x_12x') || 0;
                } else {
                    feePercent = feeMap.get(method) || 0;
                }
                const transactionFeeVal = amount * (feePercent / 100);

                // C. Identificar Custo de Material
                const materialCostVal = materialMap.get(procName) || 0;

                // D. Determinar Base de Cálculo (O Pulo do Gato)
                let baseDeCalculo = amount;

                if (professional?.descontar_taxas) {
                    baseDeCalculo -= transactionFeeVal;
                }
                if (professional?.descontar_material) {
                    baseDeCalculo -= materialCostVal;
                }

                if (baseDeCalculo < 0) baseDeCalculo = 0;

                // E. Calcular Comissão Final
                const commissionRate = professional?.commission_percentage || 0;
                const commissionValue = baseDeCalculo * (commissionRate / 100);

                // Agregando se passar no filtro e for RECEITA PAGA (para os cards do topo)
                const passesFilter = (!filterProcedure || filterProcedure === 'Todos' || procName === filterProcedure);

                if (passesFilter && t.type === 'income' && t.status === 'paid') {
                    totalRevenue += amount;
                    totalCardFees += transactionFeeVal;
                    totalMaterials += materialCostVal;
                    totalCommissions += commissionValue;

                    // Extrato de Procedimentos e Pacientes
                    if (!procedureStats[procName]) procedureStats[procName] = { revenue: 0, cost: 0 };
                    procedureStats[procName].revenue += amount;
                    procedureStats[procName].cost += (commissionValue + materialCostVal + transactionFeeVal);

                    const patientName = t.metadata?.patient_name || t.leads?.name || 'Avulso';
                    if (patientName !== 'Avulso') {
                        patientStats[patientName] = (patientStats[patientName] || 0) + amount;
                    }
                }

                // Retorna objeto enriquecido
                return {
                    ...t,
                    amount, // Garante number
                    calculated_commission: commissionValue,
                    professional_name: professional?.name || 'Não Atribuído',
                    professional: professional?.name || 'Não Atribuído', // Compatibilidade
                    patient: t.leads?.name || 'Avulso',
                    procedure: procName,
                    // Garante outros campos da interface Transaction
                    clinic_id: clinicId,
                    lead_status: t.leads?.status,
                };
            }) || [];


            // 7. LÓGICA DA 'FATIA LARANJA' (TAXAS E CUSTOS)
            const isFiltered = !!filterProfessionalId || (!!filterProcedure && filterProcedure !== 'Todos');

            // 7b. BUSCAR DESPESAS REAIS DO PERÍODO
            const { data: expenses } = await supabase
                .from('financial_transactions')
                .select('amount, category')
                .eq('clinic_id', clinicId)
                .eq('type', 'expense')
                .in('status', ['paid'])
                .gte('transaction_date', startDate)
                .lte('transaction_date', endDate);

            // Filtrar despesas duplicadas (Comissões, Taxas, Impostos)
            const filteredExpenses = expenses?.filter(e => {
                const cat = e.category || '';
                // Ignora categorias duplicadas na lógica contábil
                return !['Comissões', 'Taxas Financeiras', 'Impostos'].includes(cat);
            });
            const totalOperatingCosts = filteredExpenses?.reduce((acc, curr) => acc + (Number(curr.amount) || 0), 0) || 0;

            const marketingInvested = isFiltered ? 0 : (Number(clinicConfig?.verba_marketing) || 0);

            // Cálculo de Impostos (Simples Nacional)
            const taxRate = Number(clinicConfig?.imposto_aliquota || 0);
            const totalTaxes = isFiltered ? 0 : (totalRevenue * (taxRate / 100));

            // Exibição de Custos Fixos (Soma Marketing + Taxas Maquininha + Despesas Reais + Impostos)
            const displayedFixedCosts = isFiltered ? totalCardFees : (totalCardFees + marketingInvested + totalOperatingCosts + totalTaxes);

            // 8. RESULTADOS FINAIS
            const totalProfit = totalRevenue - (totalCommissions + totalMaterials + displayedFixedCosts);

            // Ordenação
            const profitByProcedure = Object.keys(procedureStats).map(name => {
                const s = procedureStats[name];
                return {
                    name,
                    revenue: s.revenue,
                    cost: s.cost,
                    profit: s.revenue - s.cost,
                    margin: s.revenue > 0 ? ((s.revenue - s.cost) / s.revenue) * 100 : 0
                };
            }).sort((a, b) => b.profit - a.profit);

            const topPatients = Object.entries(patientStats)
                .map(([name, val]) => ({ id: name, name, totalSpent: val }))
                .sort((a, b) => b.totalSpent - a.totalSpent)
                .slice(0, 5);

            // ========== 9. PROJEÇÃO FINANCEIRA (24x LOOKBACK) ==========
            const now = new Date();
            const currentMonthStart = startOfMonth(now);
            const lookback24Start = addMonths(currentMonthStart, -24);

            // 9a. Buscar transações de crédito dos últimos 24 meses
            const { data: creditTransactions } = await supabase
                .from('financial_transactions')
                .select('amount, transaction_date, payment_method, installments')
                .eq('clinic_id', clinicId)
                .eq('type', 'income')
                .in('status', ['paid'])
                .or('payment_method.ilike.%credit%,payment_method.ilike.%crédito%,payment_method.ilike.%credito%')
                .gte('transaction_date', lookback24Start.toISOString());

            // 9b. Buscar antecipações
            const { data: anticipationsData } = await supabase
                .from('antecipacoes_financeiras')
                .select('*')
                .eq('clinic_id', clinicId);

            const anticipations: FinancialAnticipation[] = (anticipationsData || []).map((a: any) => ({
                id: a.id,
                mes_referencia: a.mes_referencia,
                valor_bruto_retirado: Number(a.valor_bruto_retirado),
                valor_taxa: Number(a.valor_taxa),
                valor_liquido_recebido: Number(a.valor_liquido_recebido),
                created_at: a.created_at,
            }));

            // 9c. Montar buckets de 12 meses (Mês Atual → Mês Atual + 11)
            const monthNames = ['Jan', 'Fev', 'Mar', 'Abr', 'Mai', 'Jun', 'Jul', 'Ago', 'Set', 'Out', 'Nov', 'Dez'];
            const projectionBuckets: number[] = new Array(12).fill(0);
            const projectionDates: Date[] = [];

            for (let i = 0; i < 12; i++) {
                projectionDates.push(startOfMonth(addMonths(now, i)));
            }

            // 9d. Distribuir parcelas nos buckets
            // Obter taxa média de crédito para cálculo líquido
            const creditFeePercent = feeMap.get('credito_2x_4x') || feeMap.get('credito_1x') || 0;

            creditTransactions?.forEach((ct: any) => {
                const saleDate = new Date(ct.transaction_date);
                const amount = Number(ct.amount) || 0;
                const installments = Number(ct.installments) || 1;
                const parcelaLiquida = (amount * (1 - creditFeePercent / 100)) / installments;

                for (let p = 0; p < installments; p++) {
                    // D+30 por parcela: 1ª parcela chega ~30 dias, 2ª ~60 dias, etc.
                    const receiveDate = addMonths(saleDate, p + 1);
                    const receiveMonthStart = startOfMonth(receiveDate);

                    // Verificar se cai dentro dos próximos 12 meses
                    const bucketIndex = differenceInCalendarMonths(receiveMonthStart, currentMonthStart);
                    if (bucketIndex >= 0 && bucketIndex < 12) {
                        projectionBuckets[bucketIndex] += parcelaLiquida;
                    }
                }
            });

            // 9e. Guardar valores originais (antes de antecipações)
            const originalBuckets = [...projectionBuckets];

            // 9f. Aplicar antecipações
            anticipations.forEach(ant => {
                const refDate = startOfMonth(new Date(ant.mes_referencia + 'T12:00:00'));
                const refIndex = differenceInCalendarMonths(refDate, currentMonthStart);
                if (refIndex >= 0 && refIndex < 12) {
                    projectionBuckets[refIndex] -= ant.valor_bruto_retirado;
                    if (projectionBuckets[refIndex] < 0) projectionBuckets[refIndex] = 0;
                }
                // Líquido recebido no mês atual (índice 0)
                projectionBuckets[0] += ant.valor_liquido_recebido;
            });

            // 9g. Montar monthlyData
            const monthlyData = projectionDates.map((date, i) => ({
                name: monthNames[date.getMonth()],
                fullDate: date.toISOString(),
                value: Math.round(projectionBuckets[i] * 100) / 100,
                originalValue: Math.round(originalBuckets[i] * 100) / 100,
            }));

            // 9h. KPIs
            const totalGuaranteed = monthlyData.reduce((sum, m) => sum + m.value, 0);
            const peakEntry = monthlyData.reduce((max, m) => m.value > max.value ? m : max, monthlyData[0]);
            const monthlyAverage = totalGuaranteed / 12;

            // 9i. Trimestres (baseado na posição real dos meses do ano)
            const quarterValues = { q1: 0, q2: 0, q3: 0, q4: 0 };
            monthlyData.forEach(m => {
                const month = new Date(m.fullDate).getMonth(); // 0-11
                if (month <= 2) quarterValues.q1 += m.value;
                else if (month <= 5) quarterValues.q2 += m.value;
                else if (month <= 8) quarterValues.q3 += m.value;
                else quarterValues.q4 += m.value;
            });

            const financialProjection: FinancialProjection = {
                monthlyData,
                quarterly: quarterValues,
                kpis: {
                    totalGuaranteed: Math.round(totalGuaranteed * 100) / 100,
                    peakMonth: peakEntry ? `${peakEntry.name}.` : '-',
                    monthlyAverage: Math.round(monthlyAverage * 100) / 100,
                },
            };

            set({
                metrics: {
                    totalRevenue,
                    totalProfit,
                    totalCommissions,
                    totalMaterials,
                    totalFixedCosts: displayedFixedCosts,
                    profitMargin: totalRevenue > 0 ? (totalProfit / totalRevenue) * 100 : 0,
                    revenueGoal: Number(clinicConfig?.meta_faturamento || 0),
                    marginGoal: Number(clinicConfig?.meta_margem_liquida || 20),
                    // Novos campos detalhados
                    totalCardFees,
                    totalOperatingCosts,
                    totalTaxes,
                    marketingInvested
                },
                profitByProcedure,
                topPatients,
                expensesBreakdown: {
                    commissions: totalCommissions,
                    materials: totalMaterials,
                    fixed: displayedFixedCosts
                },
                transactions: enrichedTransactions,
                financialProjection,
                anticipations,
                loading: false
            });

        } catch (error) {
            console.error("Erro no BI Financeiro:", error);
            set({ loading: false });
        }
    },

    // --- AÇÃO DE ANTECIPAÇÃO ---
    registerAnticipation: async (clinicId, data) => {
        try {
            // Passo 1: Insert na tabela antecipacoes_financeiras
            const mesRef = format(startOfMonth(data.targetMonth), 'yyyy-MM-dd');
            const { error: antError } = await supabase
                .from('antecipacoes_financeiras')
                .insert([{
                    clinic_id: clinicId,
                    mes_referencia: mesRef,
                    valor_bruto_retirado: data.grossAmount,
                    valor_taxa: data.feeAmount,
                    valor_liquido_recebido: data.netAmount,
                }]);
            if (antError) throw antError;

            // Passo 2: Insert na financial_transactions APENAS DA TAXA
            const { error: txError } = await supabase
                .from('financial_transactions')
                .insert([{
                    clinic_id: clinicId,
                    description: `Taxa Antecipação - ${format(data.targetMonth, 'MMM/yyyy', { locale: ptBR })}`,
                    amount: data.feeAmount,
                    type: 'expense',
                    category: 'Taxas Financeiras',
                    transaction_date: new Date().toISOString().split('T')[0],
                    status: 'paid',
                }]);
            if (txError) throw txError;

            // Passo 3: Recarregar dados
            // Precisamos de clinicId armazenado para recarregar. Vamos usar o state atual.
            await get().fetchFinanceData(clinicId, 'custom',
                startOfMonth(new Date()),
                new Date()
            );

        } catch (error) {
            console.error('Erro ao registrar antecipação:', error);
            throw error; // Re-throw para o componente tratar
        }
    }
}));
