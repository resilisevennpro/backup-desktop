"use client"

import React, { useState } from "react"
import { X, Calendar, User, Stethoscope, CreditCard, Banknote, FileText, Loader2 } from "lucide-react"

// MOCK: Lista de Profissionais e Procedimentos (Temporário para teste)
const PROFISSIONAIS = ["Dra. Ana", "Dr. Pedro", "Dra. Julia"]
const PROCEDIMENTOS = ["Toxina Botulínica", "Preenchimento Facial", "Bioestimulador", "Limpeza de Pele"]
const FORMAS_PAGAMENTO = ["Crédito", "Débito", "Pix", "Dinheiro"]

interface NewTransactionModalProps {
  isOpen: boolean
  onClose: () => void
}

export default function NewTransactionModal({ isOpen, onClose }: NewTransactionModalProps) {
  const [isLoading, setIsLoading] = useState(false)
  
  // Estado do Formulário
  const [formData, setFormData] = useState({
    data: new Date().toISOString().split('T')[0], // Hoje YYYY-MM-DD
    paciente: "",
    profissional: "",
    procedimento: "",
    valorVenda: "",
    formaPagamento: "Crédito",
    qtdParcelas: "1",
    observacoes: ""
  })

  // Se não estiver aberto, não renderiza nada
  if (!isOpen) return null

  // Atualiza os campos conforme o usuário digita
  const handleChange = (e: React.ChangeEvent<HTMLInputElement | HTMLSelectElement | HTMLTextAreaElement>) => {
    const { name, value } = e.target
    
    // Regra: Se mudar para PIX/Débito/Dinheiro, trava parcela em 1x
    if (name === "formaPagamento" && value !== "Crédito") {
        setFormData(prev => ({ ...prev, [name]: value, qtdParcelas: "1" }))
    } else {
        setFormData(prev => ({ ...prev, [name]: value }))
    }
  }

  // Envio para o n8n
  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault()
    setIsLoading(true)

    try {
      // 1. Prepara os dados
      const payload = {
        data: formData.data,
        paciente: formData.paciente,
        profissional: formData.profissional,
        procedimento: formData.procedimento,
        valor_venda: parseFloat(formData.valorVenda),
        forma_pagamento: formData.formaPagamento,
        qtd_parcelas: parseInt(formData.qtdParcelas),
        observacoes: formData.observacoes,
        criado_em: new Date().toISOString(),
        sheet_id: "13DTsONPjNXR-fy-JmjSxecEYZcpevjcJ_7VhUs7MuYM"
      }

      // 2. Dispara para o Webhook do n8n
        const response = await fetch("https://n8n-n8n.wtgf7u.easypanel.host/webhook/d964e3cb-a0d9-4e14-9474-1bf1e58173a8", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(payload),
      })

      /* const response = await fetch("SUA_URL_DO_N8N_AQUI", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(payload),
      })
      if (!response.ok) throw new Error("Erro")
      */

      alert("Venda registrada com sucesso! (Simulação)")
      onClose() // Fecha o modal
      
      // Limpa o formulário para a próxima vez
      setFormData({
        data: new Date().toISOString().split('T')[0],
        paciente: "",
        profissional: "",
        procedimento: "",
        valorVenda: "",
        formaPagamento: "Crédito",
        qtdParcelas: "1",
        observacoes: ""
      })

    } catch (error) {
      alert("Erro ao salvar.")
    } finally {
      setIsLoading(false)
    }
  }

  return (
    // Fundo Escuro (Z-Index alto para ficar acima de tudo)
    <div className="fixed inset-0 z-[9999] flex items-center justify-center bg-black/60 backdrop-blur-sm p-4">
      
      {/* O Card do Modal (CORREÇÃO AQUI) */}
      {/* max-h-[90vh] = Altura máxima de 90% da tela */}
      {/* overflow-y-auto = Permite rolar se for maior que a tela */}
      <div className="bg-white rounded-2xl shadow-2xl w-full max-w-2xl max-h-[90vh] overflow-y-auto animate-in zoom-in-95 duration-200 flex flex-col">
        
        {/* Cabeçalho (Fixo no topo do scroll visualmente, mas rola junto nesse caso simples) */}
        <div className="flex items-center justify-between px-6 py-4 border-b border-gray-100 bg-emerald-50/50 sticky top-0 z-10 backdrop-blur-md">
          <div>
            <h2 className="text-xl font-bold text-gray-800">Nova Venda</h2>
            <p className="text-xs text-gray-500">Preencha os dados.</p>
          </div>
          <button onClick={onClose} className="p-2 rounded-full hover:bg-gray-200 text-gray-500 transition-colors">
            <X className="w-5 h-5" />
          </button>
        </div>

        {/* Formulário */}
        <form onSubmit={handleSubmit} className="p-6 space-y-6">
          
          {/* ... (Todo o conteúdo dos inputs continua igual aqui dentro) ... */}
          {/* Vou repetir o começo pra você se localizar, mantenha seus inputs iguais */}
          
          <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
            <div className="space-y-1">
              <label className="text-xs font-semibold text-gray-600 flex items-center gap-1">
                <Calendar className="w-3.5 h-3.5" /> DATA
              </label>
              <input 
                type="date" 
                name="data"
                value={formData.data} 
                onChange={handleChange}
                className="w-full h-10 px-3 border border-gray-200 rounded-lg focus:ring-2 focus:ring-emerald-500 outline-none"
                required 
              />
            </div>
             {/* ... RESTO DOS SEUS INPUTS (Paciente, Profissional, etc) ... */}
             <div className="space-y-1">
              <label className="text-xs font-semibold text-gray-600 flex items-center gap-1">
                <User className="w-3.5 h-3.5" /> PACIENTE
              </label>
              <input 
                type="text" 
                name="paciente"
                placeholder="Ex: Maria Silva"
                value={formData.paciente} 
                onChange={handleChange}
                className="w-full h-10 px-3 border border-gray-200 rounded-lg focus:ring-2 focus:ring-emerald-500 outline-none"
                required 
              />
            </div>
          </div>

          <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
            <div className="space-y-1">
              <label className="text-xs font-semibold text-gray-600 flex items-center gap-1">
                <Stethoscope className="w-3.5 h-3.5" /> PROFISSIONAL
              </label>
              <select 
                name="profissional"
                value={formData.profissional} 
                onChange={handleChange}
                className="w-full h-10 px-3 border border-gray-200 rounded-lg focus:ring-2 focus:ring-emerald-500 outline-none bg-white"
                required
              >
                <option value="">Selecione...</option>
                {PROFISSIONAIS.map(prof => <option key={prof} value={prof}>{prof}</option>)}
              </select>
            </div>

            <div className="space-y-1">
              <label className="text-xs font-semibold text-gray-600 flex items-center gap-1">
                <FileText className="w-3.5 h-3.5" /> PROCEDIMENTO
              </label>
              <select 
                name="procedimento"
                value={formData.procedimento} 
                onChange={handleChange}
                className="w-full h-10 px-3 border border-gray-200 rounded-lg focus:ring-2 focus:ring-emerald-500 outline-none bg-white"
                required
              >
                <option value="">Selecione...</option>
                {PROCEDIMENTOS.map(proc => <option key={proc} value={proc}>{proc}</option>)}
              </select>
            </div>
          </div>

          <hr className="border-gray-100" />

          {/* Financeiro */}
          <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
            <div className="space-y-1">
              <label className="text-xs font-semibold text-gray-600 flex items-center gap-1">
                <Banknote className="w-3.5 h-3.5" /> VALOR (R$)
              </label>
              <input 
                type="number" 
                name="valorVenda"
                placeholder="0.00"
                step="0.01"
                value={formData.valorVenda} 
                onChange={handleChange}
                className="w-full h-10 px-3 border border-gray-200 rounded-lg focus:ring-2 focus:ring-emerald-500 outline-none font-medium"
                required 
              />
            </div>

            <div className="space-y-1">
              <label className="text-xs font-semibold text-gray-600 flex items-center gap-1">
                <CreditCard className="w-3.5 h-3.5" /> PAGAMENTO
              </label>
              <select 
                name="formaPagamento"
                value={formData.formaPagamento} 
                onChange={handleChange}
                className="w-full h-10 px-3 border border-gray-200 rounded-lg focus:ring-2 focus:ring-emerald-500 outline-none bg-white"
                required
              >
                {FORMAS_PAGAMENTO.map(forma => <option key={forma} value={forma}>{forma}</option>)}
              </select>
            </div>

            <div className={`space-y-1 transition-opacity duration-200 ${formData.formaPagamento === 'Crédito' ? 'opacity-100' : 'opacity-40 grayscale'}`}>
              <label className="text-xs font-semibold text-gray-600">PARCELAS</label>
              <select 
                name="qtdParcelas"
                value={formData.qtdParcelas} 
                onChange={handleChange}
                disabled={formData.formaPagamento !== 'Crédito'}
                className="w-full h-10 px-3 border border-gray-200 rounded-lg focus:ring-2 focus:ring-emerald-500 outline-none bg-white"
              >
                {[1,2,3,4,5,6,10,12].map(num => <option key={num} value={num}>{num}x</option>)}
              </select>
            </div>
          </div>

          <div className="space-y-1">
            <label className="text-xs font-semibold text-gray-600">OBSERVAÇÕES</label>
            <textarea 
              name="observacoes"
              value={formData.observacoes}
              onChange={handleChange}
              rows={2}
              className="w-full px-3 py-2 border border-gray-200 rounded-lg focus:ring-2 focus:ring-emerald-500 outline-none resize-none text-sm"
            />
          </div>

          {/* Rodapé Botões */}
          <div className="flex gap-3 pt-4 sticky bottom-0 bg-white pb-2 border-t border-gray-50">
            <button 
              type="button" 
              onClick={onClose}
              className="flex-1 h-11 border border-gray-200 text-gray-700 font-medium rounded-xl hover:bg-gray-50 transition-colors"
            >
              Cancelar
            </button>
            <button 
              type="submit" 
              disabled={isLoading}
              className="flex-1 h-11 bg-emerald-600 hover:bg-emerald-700 text-white font-semibold rounded-xl shadow-lg shadow-emerald-600/20 transition-all flex items-center justify-center gap-2"
            >
              {isLoading ? <Loader2 className="w-5 h-5 animate-spin" /> : "Confirmar Venda"}
            </button>
          </div>

        </form>
      </div>
    </div>
  )
}
