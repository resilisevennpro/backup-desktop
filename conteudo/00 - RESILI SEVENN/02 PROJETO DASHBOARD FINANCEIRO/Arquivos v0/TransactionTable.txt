"use client"

import React, { useState, useEffect } from "react"
// 1. ADICIONADO: Import do ícone 'Plus'
import { Edit2, Trash2, Search, Filter, Loader2, AlertCircle, Plus } from "lucide-react" 
import NewTransactionModal from "./NewTransactionModal"

export default function TransactionsTable() {
  const [searchTerm, setSearchTerm] = useState("")
  const [transactions, setTransactions] = useState<any[]>([])
  const [isLoading, setIsLoading] = useState(true)
  const [isDeleting, setIsDeleting] = useState(false)
  
  const [isModalOpen, setIsModalOpen] = useState(false)
  const [selectedTransaction, setSelectedTransaction] = useState<any>(null)

  const SHEET_ID = "13DTsONPjNXR-fy-JmjSxecEYZcpevjcJ_7VhUs7MuYM"
  const N8N_GET_URL = "https://n8n-n8n.wtgf7u.easypanel.host/webhook/bd622e25-707a-470e-b052-277df2e1812c"
  const N8N_DELETE_URL = "https://n8n-n8n.wtgf7u.easypanel.host/webhook/07ae495f-5f68-4a64-a007-09cfad35d577" 

  const parseDate = (dateString: any) => {
    if (!dateString) return new Date();
    if (typeof dateString === 'string' && dateString.includes('-')) return new Date(dateString);
    if (typeof dateString === 'string' && dateString.includes('/')) {
        const parts = dateString.split('/');
        return new Date(Number(parts[2]), Number(parts[1]) - 1, Number(parts[0]));
    }
    return new Date();
  }

  const fetchTransactions = async () => {
    try {
      setIsLoading(true)
      const response = await fetch(`${N8N_GET_URL}?sheet_id=${SHEET_ID}`)
      const data = await response.json()

      const validData = data.filter((item: any) => item.Paciente && item.Paciente !== "" && item.Valor_Venda)

      const mappedData = validData.map((item: any, index: number) => ({
        originalRowNumber: item.id_linha || 0,
        
        id: index,
        dataObj: parseDate(item.Data), 
        paciente: item.Paciente || "Sem nome",
        profissional: item.Profissional || "-",
        servico: item.Procedimento || "Procedimento",
        valor: typeof item.Valor_Venda === 'string' 
          ? parseFloat(item.Valor_Venda.replace("R$", "").replace(".", "").replace(",", ".").trim()) 
          : Number(item.Valor_Venda) || 0,
        status: "Concluído",
        isEdited: item.Status_Edicao && item.Status_Edicao.trim() === "EDITADO"
      }))

      setTransactions(mappedData.reverse())
    } catch (error) {
      console.error("Erro ao buscar vendas:", error)
    } finally {
      setIsLoading(false)
    }
  }

  useEffect(() => {
    fetchTransactions()
  }, [])

  const handleDelete = async (rowNumber: number, pacienteNome: string) => {
    if (!rowNumber || rowNumber === 0) {
        alert("Erro: Não foi possível identificar a linha. Verifique a coluna 'id_linha'.")
        return;
    }
    if (!confirm(`Tem certeza que deseja excluir a venda de ${pacienteNome}?`)) return;
    try {
      setIsDeleting(true)
      await fetch(N8N_DELETE_URL, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ sheet_id: SHEET_ID, row_number: rowNumber })
      })
      setTransactions(prev => prev.filter(t => t.originalRowNumber !== rowNumber))
      alert("Venda excluída com sucesso!")
    } catch (error) {
      alert("Erro ao excluir.")
    } finally {
      setIsDeleting(false)
    }
  }

  const handleEdit = (venda: any) => {
    setSelectedTransaction(venda) 
    setIsModalOpen(true) 
  }

  const filteredData = transactions.filter(t => 
    t.paciente && t.paciente.toLowerCase().includes(searchTerm.toLowerCase())
  )

  return (
    <div className="bg-white rounded-2xl shadow-sm border border-slate-100 overflow-hidden">
      
      <NewTransactionModal 
        isOpen={isModalOpen} 
        onClose={() => { setIsModalOpen(false); setSelectedTransaction(null); }} 
        editData={selectedTransaction} 
      />

      <div className="p-5 border-b border-slate-100 flex flex-col sm:flex-row gap-4 justify-between items-center bg-slate-50/50">
        <div>
          <h3 className="font-bold text-slate-800 text-lg">Últimos Lançamentos</h3>
          <p className="text-sm text-slate-500">{isLoading ? "Carregando..." : `${transactions.length} vendas encontradas.`}</p>
        </div>
        
        {/* 2. ADICIONADO: Grupo de Ações com o Botão Nova Venda */}
        <div className="flex gap-2 w-full sm:w-auto">
          
          {/* Botão Nova Venda */}
          <button 
            onClick={() => { setSelectedTransaction(null); setIsModalOpen(true); }}
            className="bg-emerald-600 hover:bg-emerald-700 text-white px-4 py-2 rounded-lg text-sm font-medium flex items-center gap-2 transition-colors shadow-sm shadow-emerald-600/20 whitespace-nowrap"
          >
            <Plus className="w-4 h-4" />
            <span className="hidden sm:inline">Nova Venda</span>
            {/* Ícone '+' visível no mobile quando texto esconde */}
            <span className="sm:hidden">Novo</span> 
          </button>

          <div className="relative w-full sm:w-64">
             <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-slate-400" />
            <input 
              type="text" 
              placeholder="Buscar paciente..." 
              className="w-full pl-10 pr-4 py-2 rounded-lg border border-slate-200 text-sm focus:ring-2 focus:ring-emerald-500 outline-none"
              value={searchTerm}
              onChange={(e) => setSearchTerm(e.target.value)}
            />
          </div>
          <button onClick={fetchTransactions} className="p-2 border border-slate-200 rounded-lg hover:bg-white text-slate-600">
            <Filter className="w-4 h-4" />
          </button>
        </div>
      </div>

      <div className="overflow-x-auto">
        <table className="w-full text-sm text-left">
          {/* 3. MOBILE: Escondemos o cabeçalho no mobile (hidden) e mostramos no desktop (md:table-header-group) */}
          <thead className="hidden md:table-header-group text-xs text-slate-500 uppercase bg-slate-50/80 border-b border-slate-100">
            <tr>
              <th className="px-6 py-4 font-semibold">Data</th>
              <th className="px-6 py-4 font-semibold">Paciente</th>
              <th className="px-6 py-4 font-semibold">Profissional</th>
              <th className="px-6 py-4 font-semibold">Serviço</th>
              <th className="px-6 py-4 font-semibold text-right">Valor</th>
              <th className="px-6 py-4 font-semibold text-center">Status</th>
              <th className="px-6 py-4 font-semibold text-center">Ações</th>
            </tr>
          </thead>
          
          <tbody className="divide-y divide-slate-100">
            {isLoading && (
              <tr><td colSpan={7} className="py-10 text-center"><Loader2 className="animate-spin inline mr-2"/>Carregando...</td></tr>
            )}
            
            {!isLoading && filteredData.map((item) => (
              <tr 
                key={item.id} 
                // 4. MOBILE: Transformamos a TR em Flex Column (Card) no mobile. No desktop volta a ser table-row.
                className={`
                    flex flex-col md:table-row 
                    bg-white border-b border-slate-100 
                    p-4 md:p-0 
                    gap-3 md:gap-0 
                    hover:bg-slate-50/80 transition-colors group 
                    ${isDeleting ? 'opacity-50 pointer-events-none' : ''}
                `}
              >
                {/* Célula DATA */}
                <td className="flex md:table-cell justify-between items-center px-0 py-0 md:px-6 md:py-4 text-slate-600 whitespace-nowrap">
                   {/* Rótulo visível apenas no mobile */}
                   <span className="md:hidden text-xs font-bold text-slate-400 uppercase">Data</span>
                   {item.dataObj.toLocaleDateString('pt-BR')}
                </td>

                {/* Célula PACIENTE */}
                <td className="flex md:table-cell justify-between items-center px-0 py-0 md:px-6 md:py-4 font-medium text-slate-800">
                    <span className="md:hidden text-xs font-bold text-slate-400 uppercase">Paciente</span>
                    <div className="flex items-center gap-2">
                        {item.paciente}
                        {item.isEdited && (
                            <span 
                                className="flex items-center gap-1 px-1.5 py-0.5 rounded text-[10px] font-bold bg-amber-100 text-amber-700 border border-amber-200"
                                title="Editado"
                            >
                                <AlertCircle className="w-3 h-3" /> <span className="hidden sm:inline">EDITADO</span>
                            </span>
                        )}
                    </div>
                </td>

                {/* Célula PROFISSIONAL */}
                <td className="flex md:table-cell justify-between items-center px-0 py-0 md:px-6 md:py-4 text-slate-500">
                    <span className="md:hidden text-xs font-bold text-slate-400 uppercase">Profissional</span>
                    <span className="bg-slate-100 text-slate-600 px-2 py-1 rounded text-xs border border-slate-200">{item.profissional}</span>
                </td>

                {/* Célula SERVIÇO */}
                <td className="flex md:table-cell justify-between items-center px-0 py-0 md:px-6 md:py-4 text-slate-600">
                    <span className="md:hidden text-xs font-bold text-slate-400 uppercase">Serviço</span>
                    {item.servico}
                </td>

                {/* Célula VALOR */}
                <td className="flex md:table-cell justify-between items-center px-0 py-0 md:px-6 md:py-4 text-right font-medium text-slate-800">
                  <span className="md:hidden text-xs font-bold text-slate-400 uppercase">Valor</span>
                  R$ {item.valor.toLocaleString('pt-BR', { minimumFractionDigits: 2 })}
                </td>

                {/* Célula STATUS */}
                <td className="flex md:table-cell justify-between items-center px-0 py-0 md:px-6 md:py-4 text-center">
                  <span className="md:hidden text-xs font-bold text-slate-400 uppercase">Status</span>
                  <span className="px-2 py-1 rounded-full text-[10px] font-bold uppercase tracking-wider bg-emerald-100 text-emerald-700">
                    {item.status}
                  </span>
                </td>

                {/* Célula AÇÕES */}
                <td className="flex md:table-cell justify-end md:justify-center items-center px-0 py-0 pt-2 md:pt-4 md:px-6 md:py-4 border-t md:border-t-0 border-slate-50 mt-2 md:mt-0">
                  <div className="flex items-center gap-2 opacity-100 md:opacity-0 group-hover:opacity-100 transition-opacity">
                    <button onClick={() => handleEdit(item)} className="p-1.5 text-blue-600 hover:bg-blue-50 rounded-md transition-colors border border-blue-100 md:border-transparent">
                        <Edit2 className="w-4 h-4" />
                    </button>
                    <button onClick={() => handleDelete(item.originalRowNumber, item.paciente)} className="p-1.5 text-red-600 hover:bg-red-50 rounded-md transition-colors border border-red-100 md:border-transparent">
                        <Trash2 className="w-4 h-4" />
                    </button>
                  </div>
                </td>
              </tr>
            ))}
          </tbody>
        </table>
      </div>
    </div>
  )
}
