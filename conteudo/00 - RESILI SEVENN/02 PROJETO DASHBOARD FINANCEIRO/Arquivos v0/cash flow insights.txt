"use client"

import { Card } from "@/components/ui/card"
import { TrendingUp, AlertCircle, Calendar, DollarSign, Target } from "lucide-react"
import { XAxis, YAxis, Tooltip, ResponsiveContainer, Area, AreaChart } from "recharts"

const miniChartData = [
  { month: "Jan", amount: 3200 },
  { month: "Fev", amount: 5000 },
  { month: "Mar", amount: 4100 },
  { month: "Abr", amount: 3800 },
  { month: "Mai", amount: 4500 },

  { month: "Set", amount: 1500 },
  { month: "Out", amount: 900 },
  { month: "Nov", amount: 400 },
  { month: "Dez", amount: 200 },
]

const CustomTooltip = ({ active, payload }: any) => {
  if (active && payload && payload.length) {
    return (
      <div className="bg-slate-900 text-white px-3 py-2 rounded-lg shadow-xl text-xs">
        <p className="font-semibold">{payload[0].payload.month}</p>
        <p>R$ {payload[0].value.toLocaleString("pt-BR")}</p>
      </div>
    )
  }
  return null
}
