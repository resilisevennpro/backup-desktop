                  {/* Mini Gráfico de Área */}

                      <AreaChart data={futureCashFlow}>
                        <defs>
                          <linearGradient id="miniGradient" x1="0" y1="0" x2="0" y2="1">

                  <div className="flex-1 space-y-3">

                      </div>
                    ))}
                  </div>
                </div>

              </div>
            </Card>

            {/* Despesas e Custos + Top 5 Pacientes */}
            <div className="grid grid-cols-1 lg:grid-cols-2 gap-6">
              {/* Despesas e Custos (Card 1) */}
              <Card className="p-6 lg:p-8 shadow-lg rounded-2xl bg-white border-slate-200">
                <div className="flex items-start justify-between mb-6">
                  <div>
                    <h2 className="text-xl lg:text-2xl font-bold text-slate-900 mb-1">Despesas e Custos</h2>

                {/* Gráfico de Distribuição de Custos */}
                <div className="flex flex-col">

                    <ResponsiveContainer width="100%" height="100%">

                        <Pie
                          data={costData}

                          cy="50%" /* Mantemos 50% (centro) pois o padding da div (pt-6) já resolve o corte */

                     <p className="text-sm font-bold text-amber-900">Resumo de Custos</p>

                       <span className="font-bold">
                         {/* Cálculo da % Total de Custos */}

                       </span> do faturamento são custos operacionais e comissões.

                  <div className="p-3 rounded-xl border border-slate-100 bg-slate-50">

                      <p className="text-sm font-bold text-slate-700">
                        {formatCurrencyBRL((apiData?.kpis.faturamentoBruto || 0) - (apiData?.kpis.lucroLiquido || 0))}
                      </p>
                   </div>
                   <div className="p-3 rounded-xl border border-emerald-100 bg-emerald-50">
                      <p className="text-xs text-emerald-600 mb-1">Lucro Real</p>
                      <p className="text-sm font-bold text-emerald-700">

                      </p>
                   </div>
                </div>

                {/* Lista de Comissões por Profissional (Limpa, sem blocos repetidos embaixo) */}
                {commissionsByProf.length > 0 && (
                  <div className="border-t border-slate-100 pt-4">
                    <h3 className="text-xs font-semibold text-slate-500 uppercase tracking-wider mb-3">
                      Detalhamento de Comissões
                    </h3>
                    <div className="space-y-2 max-h-40 overflow-y-auto pr-1">

                    </div>
                  </div>

              </Card>

              <Card className="shadow-lg rounded-2xl bg-white border-slate-200 overflow-hidden">
                
                {/* --- VERSÃO MOBILE (ACCORDION) --- */}
                <div className="lg:hidden">
                  <Accordion 
                    type="single" 
                    collapsible 
                    value={mobileTopPatientsOpen} 
                    onValueChange={setMobileTopPatientsOpen}

                    <AccordionItem value="item-1" className="border-none">
                      {/* Trigger com seta verde [&>svg]:text-[#59A52C] */}
                      <AccordionTrigger className="px-6 py-6 hover:no-underline hover:bg-slate-50/50 transition-colors [&>svg]:text-[#59A52C]">
                        <div className="flex items-center justify-between w-full pr-2">
                          <div className="text-left">
                            <h2 className="text-xl font-bold text-slate-900 mb-1">Top 5 Pacientes</h2>
                            <p className="text-slate-500 text-sm">Maiores valores faturados</p>
                          </div>
                          
                          {/* Texto Verde Dinâmico */}
                          <span className="text-xs font-bold text-[#59A52C] mr-2">
                            {mobileTopPatientsOpen === "item-1" ? "Fechar Top 5" : "Abrir Top 5"}
                          </span>
                        </div>
                      </AccordionTrigger>

                      <AccordionContent className="px-6 pb-6 pt-2">
                        {/* Renderiza a lista aqui dentro no mobile */}
                        <div className="space-y-3">
                           {renderPatientsList(topPatients)}
                        </div>
                      </AccordionContent>
                    </AccordionItem>
                  </Accordion>
                </div>

                {/* --- VERSÃO DESKTOP (FIXA) --- */}
                <div className="hidden lg:block p-8">
                  <div className="flex items-start justify-between mb-6">
                    <div>
                      <h2 className="text-2xl font-bold text-slate-900 mb-1">Top 5 Pacientes</h2>
                      <p className="text-slate-500 text-sm">Com maiores valores faturados no período selecionado</p>
                    </div>
                  </div>
                  <div className="space-y-3">
                    {/* Renderiza a lista sempre visível no desktop */}
                    {renderPatientsList(topPatients)}
                  </div>
                </div>

              </Card>
            </div>

            {/* Extrato detalhado */}
            <Card className="p-6 lg:p-8 shadow-lg rounded-2xl bg-white border-slate-200">
              <div className="flex items-start justify-between mb-6">
                <div>
                  <h2 className="text-xl lg:text-2xl font-bold text-slate-900 mb-1">
                    Extrato Detalhado de Vendas
                  </h2>
                  <p className="text-slate-500 text-sm">Pesquise rapidamente filtrando por "Paciente, Procedimento, Forma de Pagamento ou Data"</p>
                </div>
              </div>

              <Accordion 
                type="single" 
                collapsible 
                value={accordionValue}
                onValueChange={setAccordionValue}
              >
                <AccordionItem value="item-1" className="border-none">
                  <AccordionTrigger className="px-4 py-3 rounded-xl bg-[#59A52C]/30 hover:bg-[#59A52C]/70 transition-colors no-underline">

                      {accordionValue === "item-1" ? "Fechar Extrato" : "Abrir Extrato"}

                  </AccordionTrigger>

                  <AccordionContent className="pt-5">
                    {/* --- ÁREA DE FILTROS --- */}
                    <div className="flex flex-col lg:flex-row lg:items-center gap-3 mb-4">
                      {/* Select atualizado com Pagamento */}
                      <Select value={filterType} onValueChange={(v: any) => setFilterType(v)}>
                        <SelectTrigger className="w-full lg:w-56 bg-white border-slate-200">
                          <SelectValue placeholder="Tipo de filtro" />
                        </SelectTrigger>
                        <SelectContent>
                          <SelectItem value="paciente">Paciente</SelectItem>
                          <SelectItem value="procedimento">Procedimento</SelectItem>
                          <SelectItem value="pagamento">Forma de Pagamento</SelectItem>
                        </SelectContent>
                      </Select>

                      <div className="relative flex-1">
                        <Search className="w-4 h-4 text-slate-400 absolute left-3 top-1/2 -translate-y-1/2" />
                        <Input
                          value={searchQuery}
                          onChange={(e) => setSearchQuery(e.target.value)}
                          // Texto de ajuda atualizado
                          placeholder="Pesquise por Paciente, Procedimento, Pagamento..."
                          className="pl-9 bg-white border-slate-200"
                        />
                      </div>

                      <div className="flex items-center gap-2 w-full lg:w-auto">
                        <Input
                          value={dateFilter}
                          onChange={(e) => setDateFilter(e.target.value)}
                          placeholder="Data (ex: 20/01)"
                          className="bg-white border-slate-200"
                        />
                        <Button variant="outline" className="whitespace-nowrap" onClick={handleClearFilters}>
                          Limpar Filtros
                        </Button>
                      </div>
                    </div>

                    {/* --- TABELA / CARDS LIST --- */}
                    <div className="overflow-hidden border border-slate-200 rounded-xl">
                      {/* Cabeçalho (Só aparece no Desktop lg:grid) */}
                      <div className="hidden lg:grid grid-cols-5 bg-slate-50 px-4 py-3 text-xs font-semibold text-slate-500 uppercase tracking-wider">
                        <span>Data</span>
                        <span>Paciente</span>
                        <span>Procedimento</span>
                        <span className="text-right">Valor</span>
                        <span className="text-right">Forma Pagto</span>

                      <div className="divide-y divide-slate-200 bg-white">
                        {filteredSalesRows.length === 0 && (
                          <div className="px-4 py-6 text-sm text-slate-500 text-center">Nenhum registro encontrado.</div>
                        )}

                        {filteredSalesRows.map((row: SalesRow, idx: number) => (
                          <div
                            key={`${row.date}-${row.patient}-${idx}`}

                            className="flex flex-col gap-2 p-4 lg:grid lg:grid-cols-5 lg:px-4 lg:py-3 lg:gap-0 hover:bg-slate-50/50 transition-colors text-sm text-slate-700"

                               <span className="text-xs text-slate-500 lg:text-sm lg:text-slate-700 flex items-center">
                                 {formatDateBR(row.date)}
                               </span>
                               {/* No mobile o valor sobe para o topo direito para destaque */}
                               <span className="lg:hidden font-bold text-slate-900">
                                 {formatCurrencyBRL(row.value)}
                               </span>
                             </div>

                              {row.patient}
                            </div>

                            {/* Grupo 3: Procedimento */}
                            <div className="text-slate-600 lg:text-slate-700 lg:flex lg:items-center">
                              {row.procedure}
                            </div>

                            {/* Valor (Apenas Desktop - pois no mobile já mostramos no topo) */}
                            <div className="hidden lg:flex justify-end items-center font-semibold text-slate-900">
                              {formatCurrencyBRL(row.value)}
                            </div>

                            {/* Forma de Pagamento com Cores Dinâmicas (Com Parcelas) */}
                            <div className="flex lg:justify-end items-center mt-1 lg:mt-0">
                              <span

                                {row.payment}
                                {/* LÓGICA: Se for Crédito E tiver mais de 1 parcela, mostra (Nx) */}
                                {row.payment.toLowerCase().includes("crédito") && (row as any).qtdParcelas > 1 && (
                                  <span className="ml-1 opacity-90 font-bold">
                                    ({(row as any).qtdParcelas}x)
                                  </span>
                                )}
                              </span>
                            </div>
                          </div>
                        ))}
                      </div>
                    </div>
                  </AccordionContent>

            </Card>

            <div className="h-6" />
          </div>

      </main>
