// Helper para cores das Formas de Pagamento
function getPaymentBadgeColor(payment: string) {
  const p = payment.toLowerCase().trim()
  if (p.includes("pix")) return "bg-emerald-100 text-emerald-700"
  if (p.includes("dinheiro")) return "bg-emerald-600 text-white" // Novo: Verde Escuro
  if (p.includes("débito") || p.includes("debito")) return "bg-blue-100 text-blue-700"
  if (p.includes("crédito") || p.includes("credito")) return "bg-orange-100 text-orange-700"
  return "bg-slate-100 text-slate-700" // Padrão
}

function formatCurrencyBRL(value: number) {
  return value.toLocaleString("pt-BR", { style: "currency", currency: "BRL" })
}

function clamp(n: number, min: number, max: number) {
  return Math.max(min, Math.min(max, n))
}

// Helper para converter data YYYY-MM-DD para DD/MM/YYYY
function formatDateBR(isoDate: string) {
  if (!isoDate) return ""
  const [y, m, d] = isoDate.split("-")
  return `${d}/${m}/${y}`
}

// Helper para obter nome do mês
function getMonthName(date: Date) {
  return date.toLocaleString('pt-BR', { month: 'long' });
}

function CustomAreaTooltip({ active, payload, label }: any) {
  if (!active || !payload?.length) return null
  const value = payload?.[0]?.value ?? 0
  return (
    <div className="bg-white px-4 py-3 rounded-xl shadow-xl border border-slate-200">
      <p className="text-xs text-slate-500 font-medium mb-2">Dia {label}</p>
      <div className="flex items-center justify-between gap-6">
        <span className="text-xs text-slate-600">Receita:</span>
        <span className="text-sm font-bold text-slate-900">{formatCurrencyBRL(value)}</span>
      </div>
    </div>
  )
}

// Função auxiliar para somar meses a uma data
function addMonths(date: Date, months: number) {
  const d = new Date(date)
  d.setMonth(d.getMonth() + months)
  return d
}

        // REGRA DE CRÉDITO: D+30 (Mês seguinte)
        // Parcela 1 (i=0) -> Cai daqui a 1 mês
        const dataParcela = addMonths(dataVenda, i + 1)
        
        if (dataParcela > today) {
            const monthLabel = dataParcela.toLocaleDateString("pt-BR", { month: "short", year: "numeric" }).replace(" de ", "/")
            
            if (futureMap.has(monthLabel)) {
                futureMap.set(monthLabel, (futureMap.get(monthLabel) || 0) + valorParcela)
            }
        }
      }
    })

    // 4. Converter Map em Array ordenado para o Gráfico
    return monthOrder.map(monthLabel => ({
      month: monthLabel.split("/")[0], // Pega só o "fev." para o eixo X ficar limpo
      fullDate: monthLabel,
      amount: futureMap.get(monthLabel) || 0
    }))

      else if (["jul", "ago", "set"].some(x => m.includes(x))) quarters.Q3 += item.amount

                                <div className="grid grid-cols-1 lg:grid-cols-2 gap-6">
                                  {/* Despesas e Custos (Card 1) */}
                                  <Card className="p-6 lg:p-8 shadow-lg rounded-2xl bg-white border-slate-200">
                                    <div className="flex items-start justify-between mb-6">
                                      <div>
                                        <h2 className="text-xl lg:text-2xl font-bold text-slate-900 mb-1">Despesas e Custos</h2>

                                              data={costData}

                                    type="single"

                                        <div className="overflow-hidden border border-slate-200 rounded-xl">
                                          {/* Cabeçalho (Só aparece no Desktop lg:grid) */}

                                            <span>Data</span>

                                            <span>Procedimento</span>

                                            {filteredSalesRows.map((row: SalesRow, idx: number) => (
                                              <div
                                                key={`${row.date}-${row.patient}-${idx}`}
                                                // MOBILE: Flex Column (Card) | DESKTOP: Grid (Tabela)
                                                className="flex flex-col gap-2 p-4 lg:grid lg:grid-cols-5 lg:px-4 lg:py-3 lg:gap-0 hover:bg-slate-50/50 transition-colors text-sm text-slate-700"

                                                    {formatDateBR(row.date)}
                                                  </span>
                                                  {/* No mobile o valor sobe para o topo direito para destaque */}
                                                  <span className="lg:hidden font-bold text-slate-900">
                                                    {formatCurrencyBRL(row.value)}
                                                  </span>
                                                </div>

                                                {/* Grupo 2: Paciente (Destaque mobile) */}
                                                <div className="font-bold text-slate-900 lg:font-medium lg:flex lg:items-center">
                                                  {row.patient}
                                                </div>

                                                {/* Grupo 3: Procedimento */}
                                                <div className="text-slate-600 lg:text-slate-700 lg:flex lg:items-center">
                                                  {row.procedure}
                                                </div>

                                                {/* Valor (Apenas Desktop - pois no mobile já mostramos no topo) */}
                                                <div className="hidden lg:flex justify-end items-center font-semibold text-slate-900">
                                                  {formatCurrencyBRL(row.value)}
                                                </div>

                                                {/* Forma de Pagamento com Cores Dinâmicas (Com Parcelas) */}
                                                <div className="flex lg:justify-end items-center mt-1 lg:mt-0">
                                                  <span 
                                                    className={`inline-flex px-2.5 py-1 rounded-md text-xs font-semibold border ${getPaymentBadgeColor(row.payment)} border-transparent/10`}
                                                  >
                                                    {row.payment}
                                                    {/* LÓGICA: Se for Crédito E tiver mais de 1 parcela, mostra (Nx) */}
                                                    {row.payment.toLowerCase().includes("crédito") && (row as any).qtdParcelas > 1 && (
                                                      <span className="ml-1 opacity-90 font-bold">
                                                        ({(row as any).qtdParcelas}x)
                                                      </span>
                                                    )}
                                                  </span>
                                                </div>
                                              </div>
                                            ))}
                                          </div>
                                        </div>
                                      </AccordionContent>
                                    </AccordionItem>
                                  </Accordion>
                                </Card>           
                            {/* Footer spacing */}
                            <div className="h-6" />
                          </div>
                        </div>
                      ) : ( 
                    /* -------------------------------------------------- */
                    /* OPÇÃO 2: GESTÃO DE LANÇAMENTOS (NOVA TABELA)       */
                    /* -------------------------------------------------- */
                    <div className="space-y-6 animate-in fade-in slide-in-from-right-8 duration-300">
                      <div className="flex items-center justify-between">
                        <h2 className="text-2xl font-bold text-slate-800 flex items-center gap-2">
                          <ListChecks className="w-6 h-6 text-emerald-600" />
                          Gestão Financeira
                        </h2>
                      </div>
                      {/* Tabela de Lançamentos */}
                      <div className="mt-4">
                      <TransactionsTable />
                      </div>
                    </div>
              </div>
            </div>
          {/* MODAL GLOBAL */}
          <NewTransactionModal 
            isOpen={isNewTransactionModalOpen} 
            onClose={() => setIsNewTransactionModalOpen(false)} 
          />
      </main>
    </div>
  )
