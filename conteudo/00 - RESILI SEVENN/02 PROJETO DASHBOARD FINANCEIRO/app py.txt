@st.cache_data

    patients = ["Camila Silva", "Lucas Souza", "Mariana Lima", "Roberto Alves", "Fernanda Costa", 
                "Paulo Santos", "Juliana Rocha", "Bruno Dias", "Patricia Gomes", "Ricardo Melo"]
    procedures = {

    }

    payment_methods = ["Crédito 12x", "Débito", "Pix"]
    
    data = []
    today = datetime.now()
    # Spread dates across current month
    start_date = today.replace(day=1)
    
    for _ in range(30):
        proc_name, base_price = random.choice(list(procedures.items()))
        # Vary price slightly
        final_price = base_price * random.uniform(0.9, 1.1)

        data.append(item)
    
    df = pd.DataFrame(data)
    return df

RAW_DF = generate_data()

# --- BUSINESS LOGIC ---
def calculate_metrics(df):
    df_calc = df.copy()
    
    # 1. Taxa Maquininha
    def get_taxa(row):
        if row["Forma_Pagamento"] == "Crédito 12x": return 0.12
        if row["Forma_Pagamento"] == "Débito": return 0.01
        return 0.0 # Pix
    
    df_calc["Taxa_Perc"] = df_calc.apply(get_taxa, axis=1)
    df_calc["Taxa_Maquininha"] = df_calc["Valor_Venda"] * df_calc["Taxa_Perc"]
    
    # 2. Imposto (6%)
    df_calc["Imposto"] = df_calc["Valor_Venda"] * 0.06
    
    # 3. Comissão
    def get_commission(row):
        if "[Dona]" in row["Profissional"]:
            return 0.0
        elif "[Parceiro]" in row["Profissional"]:
            base = row["Valor_Venda"] - row["Custo_Material_Lab"] - row["Imposto"]
            return base * 0.30
        return 0.0
        
    df_calc["Comissão"] = df_calc.apply(get_commission, axis=1)
    
    # 4. Lucro Líquido Real
    df_calc["LUCRO_LIQUIDO_REAL"] = df_calc["Valor_Venda"] - (
        df_calc["Custo_Material_Lab"] + 
        df_calc["Taxa_Maquininha"] + 
        df_calc["Imposto"] + 
        df_calc["Comissão"]
    )
    
    # 5. Data Recebimento (Simplificado)
    # Pix = Same date, Debit = Same date (approx), Credit 12x = Split in 12 months
    # For chart purposes, we will treat 'Recebimento' as projected cash flow
    # This logic will be handled in the chart preparation to avoid exploding rows here
    
    return df_calc

DF_WORK = calculate_metrics(RAW_DF)

    # Logic to explode payments
    projection_data = []
    
    for idx, row in DF_FILTERED.iterrows():
        val_liq = row["LUCRO_LIQUIDO_REAL"] # Assuming we track net flow, or should be sales? Usually Cash Flow tracks total inflow. Let's use Sales - Tax/Fees deducted at source if relevant, but let's stick to simple SALES inflow for validation.
        # However, "Lucro Liquido" is better for "Real Money in Pocket". Let's use Net Profit for "Safe Cash Flow".

    df_flow = pd.DataFrame(projection_data)
