"use server"

import { Redis } from "@upstash/redis"

const redis = Redis.fromEnv()

export async function processSmileImage(formData: FormData) {
  try {
    console.log("[v0] Starting image processing with Stability AI...")

    const accessCode = formData.get("accessCode") as string

    if (!accessCode || !accessCode.trim()) {
      return {
        success: false,
        error: "Código de acesso é obrigatório",
      }
    }

    // Check credits in Redis
    const credits = await redis.get(`credits:${accessCode.trim()}`)

    if (credits === null || credits === undefined) {
      return {
        success: false,
        error: "Código de acesso inválido. Verifique com o administrador.",
      }
    }

    const creditsNumber = Number(credits)

    if (creditsNumber <= 0) {
      return {
        success: false,
        error: "Saldo de créditos esgotado. Entre em contato com o administrador para recarregar.",
      }
    }

    console.log(`[v0] Access code ${accessCode} validated. Credits available: ${creditsNumber}`)

    const apiKey = process.env.STABILITY_API_KEY

    if (!apiKey) {
      return {
        success: false,
        error: "Configuration Error: STABILITY_API_KEY is missing.",
      }
    }

    const imageFile = formData.get("image") as File

    if (!imageFile) {
      return {
        success: false,
        error: "No image file provided",
      }
    }

    const apiFormData = new FormData()
    apiFormData.append("image", imageFile)

    apiFormData.append("search_prompt", "teeth, visible teeth, upper teeth, lower teeth, dentition")

    const userPrompt = `DENTAL TRANSFORMATION: EXTREME ULTRA-WHITE BL1 ZIRCONIA VENEERS., keeping both upper and lower teeth visible.The goal is a Hollywood smile: intense whiteness, high opacity, and perfect alignment.MANDATORY REQUIREMENTS: 1. PRESERVE VISIBILITY: Keep upper and lower teeth visible, as they were in the original photo. Do not conceal or cover the lower teeth. 2. COLOR & LIGHTING(BL1 SHADE - CRUCIAL):* Pure white, snow white (BL1) Hex Code #FFFFFF - monochromatic; *Lighting:** COLD STUDIO LIGHTING. Blue undertone. NO yellow warm feedback.IGNORE the warm lighting of the face.* NO gradients, the tooth should have exactly the same brilliant white at the gum line and from gum tip to gum tip;* Removes ALL yellowish/warm tones near the gum line;* ​​100% opacity - zero translucency; * Teeth should appear solid and dense, blocking all background light. No grayish edges; Effect:** Self-emissive glowing white. The teeth must be brighter than anything else in the photo. 3. DENTAL ANATOMY:* Create deep, distinct shadow lines between each tooth (sharp separation); * Don't let them look like a single fused block. Each tooth is an individual unit;* Volumetric appearance with vestibular curves. Teeth should appear thick and voluminous (buccal volume), curving slightly outward to capture light; * High-gloss wet polishing with sharp, precise light reflections in the center of the teeth (surface). 4. STRUCTURE: * Perfect alignment and leveling; * Corrects any size discrepancies; * Sharp contrast line at the gum-tooth junction;* Gingival Contrast:** Creates a sharp, high-contrast line where the white tooth meets the pink gum. No blurring at the margin. Restriction: Keep the skin, lips, or any element of the image 100% original (natural gum contours preserved). You can only modify the internal dental structure (the teeth).    5. APPLICATION IN BOTH ARCHES:  * Transforms the upper teeth (maxillary arch);    * Transforms the lower teeth (mandibular arch), if visible.* Maintains the proper occlusal relationship.STRICT RESTRICTION: Modify only the structure and color of the teeth. Keep the skin, lips, features, and background 100% unchanged. * Make the white absolute, covering all yellow undertones completely.`

    apiFormData.append("prompt", userPrompt)

    apiFormData.append(
      "negative_prompt",
      // Coisas que deixam o dente amarelo/realista (PROIBIDO):
      "natural teeth, organic texture, realistic enamel, warm lighting, yellow tones, ivory, cream color, beige, off-white, shadow on teeth, translucent tips, transparent edges, " +
      // Defeitos estruturais (PROIBIDO):
      "missing lower teeth, fused teeth, monoblock, continuous white bar, no separation, flat look, cavities, rotting, metal, braces, blur, distorted, extra teeth, gum covering teeth, receding gums, noise, grainy, lips, changed lips"
    )

    apiFormData.append("output_format", "png") // Garante qualidade máxima (sem artefatos de JPG)
    
    // Grow Mask (Opcional): Expande a máscara em 3 pixels para "fundir" melhor a borda do dente com a gengiva
    apiFormData.append("grow_mask", "0")


    console.log("[v0] Sending request to Stability AI (Search and Replace)...")

    const response = await fetch("https://api.stability.ai/v2beta/stable-image/edit/search-and-replace", {
      method: "POST",
      headers: {
        Authorization: `Bearer ${apiKey}`,
        Accept: "application/json",
      },
      body: apiFormData,
    })

    if (!response.ok) {
      const errorText = await response.text()
      console.error("[v0] Stability API Error:", errorText)
      throw new Error(`Stability API error: ${response.status} - ${errorText}`)
    }

    const data = await response.json()

    if (data.image) {
      console.log("[v0] Successfully processed image")

      await redis.decr(`credits:${accessCode.trim()}`)
      console.log(`[v0] Credit deducted for ${accessCode}. Remaining: ${creditsNumber - 1}`)

      return {
        success: true,
        processedImageUrl: `data:image/png;base64,${data.image}`,
      }
    }

    return {
      success: false,
      error: "Failed to generate image. Stability AI did not return a valid image.",
    }
  } catch (error) {
    console.error("[v0] Error processing image:", error)
    return {
      success: false,
      error: `Processing failed: ${error instanceof Error ? error.message : "Unknown error"}`,
    }
  }
}
