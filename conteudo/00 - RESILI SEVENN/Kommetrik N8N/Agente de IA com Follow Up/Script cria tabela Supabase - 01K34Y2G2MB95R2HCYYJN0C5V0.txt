BEGIN;

-- Tabela
CREATE TABLE IF NOT EXISTS public.followup_lead_table (
  id                   BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

  followup_flow_name   TEXT NOT NULL CHECK (length(btrim(followup_flow_name)) > 0),
  agent                TEXT NOT NULL CHECK (length(btrim(agent)) > 0),
  generated_by         TEXT NOT NULL CHECK (length(btrim(generated_by)) > 0),
  user_name            TEXT NOT NULL CHECK (length(btrim(user_name)) > 0),
  entity_id            TEXT NOT NULL CHECK (length(btrim(entity_id)) > 0),

  last_interaction_at  TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  created_at           TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at           TIMESTAMPTZ NOT NULL DEFAULT NOW(),

  followup_count       INT  NOT NULL DEFAULT 0,

  contact_id           TEXT NOT NULL CHECK (length(btrim(contact_id)) > 0),
  chat_id              TEXT NOT NULL CHECK (length(btrim(chat_id)) > 0),
  element_type         TEXT NOT NULL CHECK (length(btrim(element_type)) > 0),
  author_id            TEXT NOT NULL CHECK (length(btrim(author_id)) > 0),
  talk_id              TEXT NOT NULL CHECK (length(btrim(talk_id)) > 0),
  workflow_origin_id   TEXT NOT NULL CHECK (length(btrim(workflow_origin_id)) > 0),
  workflow_name        TEXT NOT NULL CHECK (length(btrim(workflow_name)) > 0),

  kommo_account_id     BIGINT NOT NULL,

  tabela_memoria       TEXT NOT NULL CHECK (length(btrim(tabela_memoria)) > 0),

  CONSTRAINT followup_lead_table_entity_id_key UNIQUE (entity_id)
);

-- Comentários (tabela)
COMMENT ON TABLE public.followup_lead_table IS 'Tabela de controle de leads para cadências de follow-up.';

-- Comentários (colunas)
COMMENT ON COLUMN public.followup_lead_table.id                  IS 'Identificador único (PK).';
COMMENT ON COLUMN public.followup_lead_table.followup_flow_name  IS 'Nome do fluxo/cadência de follow-up aplicado.';
COMMENT ON COLUMN public.followup_lead_table.agent               IS 'Nome do agente responsável.';
COMMENT ON COLUMN public.followup_lead_table.generated_by        IS 'Origem geradora do registro (ex.: sistema, worker, IA).';
COMMENT ON COLUMN public.followup_lead_table.user_name           IS 'Nome do usuário/lead no canal.';
COMMENT ON COLUMN public.followup_lead_table.entity_id           IS 'Identificador único da entidade do lead, lead_id (único na tabela).';
COMMENT ON COLUMN public.followup_lead_table.last_interaction_at IS 'Timestamp da última interação conhecida com o lead.';
COMMENT ON COLUMN public.followup_lead_table.created_at          IS 'Timestamp de criação do registro.';
COMMENT ON COLUMN public.followup_lead_table.updated_at          IS 'Timestamp da última atualização do registro.';
COMMENT ON COLUMN public.followup_lead_table.followup_count      IS 'Quantidade de mensagens de follow-up já enviadas.';
COMMENT ON COLUMN public.followup_lead_table.contact_id          IS 'ID do contato no CRM/plataforma.';
COMMENT ON COLUMN public.followup_lead_table.chat_id             IS 'ID da conversa/canal.';
COMMENT ON COLUMN public.followup_lead_table.element_type        IS 'Tipo do elemento/evento associado ao registro.';
COMMENT ON COLUMN public.followup_lead_table.author_id           IS 'ID do autor da última mensagem/interação.';
COMMENT ON COLUMN public.followup_lead_table.talk_id             IS 'ID do “talk”/thread/conversa no provedor.';
COMMENT ON COLUMN public.followup_lead_table.workflow_origin_id  IS 'ID do workflow de origem.';
COMMENT ON COLUMN public.followup_lead_table.workflow_name       IS 'Nome do workflow de origem.';
COMMENT ON COLUMN public.followup_lead_table.kommo_account_id    IS 'ID da conta Kommo (ou equivalente) relacionada.';
COMMENT ON COLUMN public.followup_lead_table.tabela_memoria      IS 'Nome da tabela de memória associada ao lead.';

-- Índices compostos (com variação de ordenação em last_interaction_at)
CREATE INDEX IF NOT EXISTS idx_flt_comp_asc
  ON public.followup_lead_table (followup_count, followup_flow_name, last_interaction_at ASC);
COMMENT ON INDEX idx_flt_comp_asc IS 'Índice composto: followup_count, followup_flow_name, last_interaction_at ASC.';

CREATE INDEX IF NOT EXISTS idx_flt_comp_desc
  ON public.followup_lead_table (followup_count, followup_flow_name, last_interaction_at DESC);
COMMENT ON INDEX idx_flt_comp_desc IS 'Índice composto: followup_count, followup_flow_name, last_interaction_at DESC.';

-- Índices individuais
CREATE INDEX IF NOT EXISTS idx_flt_followup_count
  ON public.followup_lead_table (followup_count);
COMMENT ON INDEX idx_flt_followup_count IS 'Índice individual em followup_count.';

CREATE INDEX IF NOT EXISTS idx_flt_followup_flow_name
  ON public.followup_lead_table (followup_flow_name);
COMMENT ON INDEX idx_flt_followup_flow_name IS 'Índice individual em followup_flow_name.';

CREATE INDEX IF NOT EXISTS idx_flt_last_interaction_at_asc
  ON public.followup_lead_table (last_interaction_at ASC);
COMMENT ON INDEX idx_flt_last_interaction_at_asc IS 'Índice individual em last_interaction_at ASC.';

CREATE INDEX IF NOT EXISTS idx_flt_last_interaction_at_desc
  ON public.followup_lead_table (last_interaction_at DESC);
COMMENT ON INDEX idx_flt_last_interaction_at_desc IS 'Índice individual em last_interaction_at DESC.';

COMMIT;

BEGIN;

-- Function: atualiza automaticamente o updated_at em updates
CREATE OR REPLACE FUNCTION public.tg_set_updated_at()
RETURNS trigger
LANGUAGE plpgsql
AS $$
BEGIN
  NEW.updated_at := NOW();
  RETURN NEW;
END;
$$;

COMMENT ON FUNCTION public.tg_set_updated_at()
IS 'Trigger function que seta updated_at = now() em updates.';

-- Trigger: dispara antes de qualquer UPDATE
DROP TRIGGER IF EXISTS trg_followup_lead_table_set_updated_at ON public.followup_lead_table;

CREATE TRIGGER trg_followup_lead_table_set_updated_at
BEFORE UPDATE ON public.followup_lead_table
FOR EACH ROW
WHEN (OLD IS DISTINCT FROM NEW)
EXECUTE FUNCTION public.tg_set_updated_at();

COMMENT ON TRIGGER trg_followup_lead_table_set_updated_at ON public.followup_lead_table
IS 'Atualiza a coluna updated_at automaticamente antes de UPDATE.';

COMMIT;
