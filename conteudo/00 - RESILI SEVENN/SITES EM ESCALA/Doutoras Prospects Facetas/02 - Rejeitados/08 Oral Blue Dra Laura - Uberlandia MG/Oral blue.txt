                    if (carousel.scrollLeft >= maxScroll - 10) {
                        carousel.scrollTo({ left: 0, behavior: 'smooth' });
                    } else {
                        carousel.scrollBy({ left: cardWidth, behavior: 'smooth' });
                    }
                }, 5000);
            }

            function stopAutoScroll() {
                clearInterval(scrollInterval);
            }

            if (carousel) {
                carousel.addEventListener('scroll', updateActiveDot);
                carousel.addEventListener('touchstart', stopAutoScroll, {passive: true});
                carousel.addEventListener('mousedown', stopAutoScroll);
                startAutoScroll();
            }

            dots.forEach((dot, index) => {
                dot.addEventListener('click', () => {
                    stopAutoScroll(); 
                    if(carousel) {
                        const cardWidth = carousel.firstElementChild.offsetWidth + 24;
                        carousel.scrollTo({
                            left: cardWidth * index,
                            behavior: 'smooth'
                        });
                    }
                });
            });
        });

        // 6. Efeito de Digitação Infinita para H2
        const typingHeadings = document.querySelectorAll('h2');
        typingHeadings.forEach(heading => {
            const originalText = heading.innerText;
            heading.setAttribute('data-text', originalText);
            heading.innerText = ''; 
            
            heading.classList.add('typing-effect');

            const observer = new IntersectionObserver((entries) => {
                entries.forEach(entry => {
                    if (entry.isIntersecting) {
                        if (!heading.hasAttribute('data-typing')) {
                            startTyping(heading, originalText);
                            heading.setAttribute('data-typing', 'true'); 
                        }
                        observer.unobserve(entry.target); 
                    }
                });
            }, { threshold: 0.5 }); 

            observer.observe(heading);
        });

        function startTyping(element, text) {
            let i = 0;
            element.innerText = ''; 

            function type() {
                if (i < text.length) {
                    element.innerHTML += text.charAt(i);
                    i++;
                    setTimeout(type, 80); 
                } else {
                    setTimeout(() => {
                        element.innerText = ''; 
                        i = 0; 
                        startTyping(element, text); 
                    }, 2000);
                }
            }
            
            type();
        }
    </script>
