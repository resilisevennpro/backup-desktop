// Helper para corrigir o bug de data -1 dia (UTC fix)
const stringToDate = (dateString: string) => {
  if (!dateString) return new Date() // Proteção contra crash
  const [year, month, day] = dateString.split('-').map(Number)
  return new Date(year, month - 1, day, 12, 0, 0)
}

// Helper: Calcula a data de vencimento da fatura baseada na compra
const getInvoiceDueDate = (purchaseDate: Date, closingDay: number, dueDay: number) => {
  const pDate = new Date(purchaseDate)
  // Define o ano/mês base
  let targetYear = pDate.getFullYear()
  let targetMonth = pDate.getMonth() // 0 = Jan

  // Se comprou no dia do fechamento ou depois, joga pro mês seguinte
  if (pDate.getDate() >= closingDay) {
    targetMonth++
  }

  // Se o dia do vencimento for menor que o fechamento (ex: fecha 30, vence 05), 
  // significa que vence no mês seguinte ao fechamento
  if (dueDay < closingDay) {
    targetMonth++
  }

  // Ajusta virada de ano
  const dueDate = new Date(targetYear, targetMonth, dueDay, 12, 0, 0)
  return dueDate
}

                  value={searchTerm}
                  onChange={(e) => {
                    setSearchTerm(e.target.value)
                    setVisibleCount(15)
                  }}
                  className="w-full pl-10 pr-4 py-2 rounded-lg bg-zinc-800/50 text-white text-sm placeholder-zinc-500 focus:outline-none focus:ring-2 focus:ring-purple-500/50 transition-all border border-zinc-800"
                />
              </div>

              {/* Filtros de Tipo */}
              <div className="flex gap-2">
                <button
                  onClick={() => {
                    setTransactionFilter('all')
                    setVisibleCount(15)
                  }}
                  className={cn(
                    "px-4 py-2 rounded-lg text-sm font-medium whitespace-nowrap transition-all",
                    transactionFilter === 'all'
                      ? "bg-purple-500/20 text-purple-400 border border-purple-500/50"
                      : "bg-zinc-800/50 text-zinc-400 border border-zinc-800 hover:bg-zinc-800"
                  )}
                >
                  Todos
                </button>
                <button
                  onClick={() => {
                    setTransactionFilter('income')
                    setVisibleCount(15)
                  }}
                  className={cn(
                    "px-4 py-2 rounded-lg text-sm font-medium whitespace-nowrap transition-all",
                    transactionFilter === 'income'
                      ? "bg-green-500/20 text-green-400 border border-green-500/50"
                      : "bg-zinc-800/50 text-zinc-400 border border-zinc-800 hover:bg-zinc-800"
                  )}
                >
                  Receitas
                </button>
                <button
                  onClick={() => {
                    setTransactionFilter('expense')
                    setVisibleCount(15)
                  }}
                  className={cn(
                    "px-4 py-2 rounded-lg text-sm font-medium whitespace-nowrap transition-all",
                    transactionFilter === 'expense'
                      ? "bg-red-500/20 text-red-400 border border-red-500/50"
                      : "bg-zinc-800/50 text-zinc-400 border border-zinc-800 hover:bg-zinc-800"
                  )}
                >
                  Despesas
                </button>
              </div>
              {/* Filtros de Status (Novo) */}
              <div className="flex gap-2 p-1 bg-zinc-900 rounded-lg border border-zinc-800 self-start sm:self-auto">
                 <button
                  onClick={() => setStatusFilter('all')}
                  className={cn("px-3 py-1.5 text-xs font-medium rounded-md transition-all", statusFilter === 'all' ? "bg-zinc-700 text-white" : "text-zinc-400 hover:text-white")}
                >
                  Tudo
                </button>
                <button
                  onClick={() => setStatusFilter('paid')}
                  className={cn("px-3 py-1.5 text-xs font-medium rounded-md transition-all flex items-center gap-1", statusFilter === 'paid' ? "bg-emerald-500/20 text-emerald-400" : "text-zinc-400 hover:text-emerald-400")}
                >
                  <CheckCircle2 className="w-3 h-3" /> Realizados
                </button>
                <button
                  onClick={() => setStatusFilter('pending')}
                  className={cn("px-3 py-1.5 text-xs font-medium rounded-md transition-all flex items-center gap-1", statusFilter === 'pending' ? "bg-orange-500/20 text-orange-400" : "text-zinc-400 hover:text-orange-400")}
                >
                  <Clock className="w-3 h-3" /> Agendados
                </button>
              </div>
            </div>

            {/* Botão Limpar Filtro de Categoria */}
            {selectedCategoryDetails && (
              <button
                onClick={clearCategoryFilter}
                className="text-sm text-purple-400 hover:text-purple-300 transition-colors flex items-center gap-2 mb-3"
              >
                <X className="w-3 h-3" />
                Limpar filtro de categoria ({getCategoryLabel(selectedCategoryDetails, chartType)})
              </button>
            )}

            {/* Contador de Transações */}
            {listFilteredTransactions.length > 0 && (
              <p className="text-xs text-zinc-500">
                Exibindo {visibleTransactions.length} de {listFilteredTransactions.length} transações
              </p>
            )}
          </div>

          {listFilteredTransactions.length === 0 ? (
            <div className="p-8 md:p-12 text-center">
              <div className="inline-block p-4 rounded-full bg-zinc-800/50 mb-4">
                <Wallet className="w-8 h-8 md:w-12 md:h-12 text-zinc-600" />
              </div>
              <p className="text-zinc-500 text-sm md:text-lg">
                {searchTerm || transactionFilter !== 'all' || selectedCategoryDetails
                  ? 'Nenhuma transação encontrada.' 
                  : 'Nenhuma transação no período.'}
              </p>
            </div>
          ) : (
            <>
              <div className="divide-y divide-zinc-800/50">
                {visibleTransactions.map((transaction) => {
                  const IconComponent = getCategoryIcon(transaction.category || 'other', transaction.type)
                  const isPending = transaction.status === 'pending'
                  
                  return (
                    <div
                      key={transaction.id}
                      className="p-4 hover:bg-zinc-800/30 transition-colors duration-200 border-b border-zinc-800/50 last:border-0"
                    >
                      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3 sm:gap-4">
                        
                        {/* Bloco Esquerdo: Ícone + Descrição + Data */}
                        <div className="flex items-start gap-3 w-full sm:flex-1 min-w-0">
                          <div className={cn(
                            "p-2 rounded-lg flex-shrink-0 mt-1 sm:mt-0",
                            transaction.type === 'income'
                              ? 'bg-green-500/10 border border-green-500/30'
                              : 'bg-red-500/10 border border-red-500/30'
                          )}>
                            <IconComponent className={cn(
                              "w-5 h-5",
                              transaction.type === 'income' ? 'text-green-400' : 'text-red-400'
                            )} />
                          </div>
                          
                          <div className="min-w-0 flex-1">
                            {/* 🔥 Correção: break-words permite quebra de linha em nomes longos */}
                            <p className="font-medium text-white text-base leading-snug break-words pr-2">
                               {transaction.description}
                            </p>

                            {/* TAG DO CARTÃO (NOVO) */}
                            {transaction.payment_method === 'credit' && transaction.card_id && (
                              <div className="mt-1">
                                <span 
                                  className="inline-flex items-center gap-1 px-1.5 py-0.5 rounded text-[10px] font-bold text-white uppercase tracking-wider shadow-sm"
                                  style={{ 
                                    backgroundColor: cards.find(c => c.id === transaction.card_id)?.color || '#52525b' 
                                  }}
                                >
                                  <CreditCard className="w-3 h-3" />
                                  {cards.find(c => c.id === transaction.card_id)?.name}
                                </span>
                              </div>
                            )}

                            <div className="flex flex-wrap items-center gap-2 text-xs md:text-sm text-zinc-400 mt-1">
                              <span className="flex items-center gap-1">
                                <Calendar className="w-3 h-3" />
                                {formatDate(transaction.date)}
                              </span>
                              <span className="text-zinc-600">•</span>
                              <span className="flex items-center gap-1">
                                {getCategoryLabel(transaction.category || 'other', transaction.type)}
                              </span>
                              {isPending && (
                                <span className="flex items-center gap-1 text-orange-400 bg-orange-400/10 px-1.5 py-0.5 rounded ml-1">
                                  <Clock className="w-3 h-3" /> Pendente
                                </span>
                              )}
                            </div>
                          </div>
                        </div>

                        {/* Bloco Direito: Valor + Botões (Vai para baixo no mobile) */}
                        <div className="flex items-center justify-between sm:justify-end gap-4 w-full sm:w-auto mt-1 sm:mt-0 pl-[3.25rem] sm:pl-0">
                          <p className={cn(
                            "font-bold text-lg",
                            transaction.type === 'income'
                              ? 'text-green-400'
                              : 'text-red-400'
                          )}>
                            {transaction.type === 'income' ? '+' : '-'} {formatCurrency(transaction.amount)}
                          </p>
                          
                          <div className="flex items-center gap-1">
                            <button
                              onClick={() => handleEdit(transaction)}
                              className="p-2 text-zinc-500 hover:text-blue-400 hover:bg-blue-500/10 rounded-lg transition-all"
                            >
                              <Pencil className="w-5 h-5" />
                            </button>
                            <button
                              onClick={() => handleDelete(transaction.id, transaction.group_id)}
                              className="p-2 text-zinc-500 hover:text-red-400 hover:bg-red-500/10 rounded-lg transition-all"
                            >
                              <Trash2 className="w-5 h-5" />
                            </button>
                          </div>
                        </div>

                      </div>
                    </div>
                  )
                })}
              </div>

              {/* Botões de Paginação */}
              {listFilteredTransactions.length > 15 && (
                <div className="p-4 border-t border-zinc-800/50 flex gap-3 justify-center">
                  {visibleCount < listFilteredTransactions.length && (
                    <button
                      onClick={() => setVisibleCount(prev => prev + 15)}
                      className="px-6 py-2 text-sm text-zinc-400 hover:text-white transition-colors"
                    >
                      Ver mais transações
                    </button>
                  )}
                  {visibleCount > 15 && (
                    <button
                      onClick={() => setVisibleCount(15)}
                      className="px-6 py-2 text-sm text-zinc-400 hover:text-white transition-colors"
                    >
                      Ver menos
                    </button>
                  )}
                </div>
              )}
            </>
          )}
        </div>
      </main>

      {/* Modal de Adicionar Transação */}
      {showModal && (
        <div 
          className="fixed inset-0 bg-black/80 backdrop-blur-sm flex items-center justify-center z-50 p-4"
          onClick={() => setShowModal(false)}
        >
          <div 
            className="bg-zinc-900 rounded-xl border border-zinc-800/50 shadow-2xl shadow-black/50 max-w-md w-full p-6 max-h-[90vh] overflow-y-auto relative"
            onClick={(e) => e.stopPropagation()}
          >
            <button
              type="button"
              onClick={() => setShowModal(false)}
              className="absolute top-4 right-4 p-2 text-zinc-400 hover:text-white hover:bg-zinc-800 rounded-lg transition-all"
            >
              <X className="w-5 h-5" />
            </button>

            <h3 className="text-xl md:text-2xl font-bold text-white mb-6 bg-gradient-to-r from-green-400 to-purple-400 bg-clip-text text-transparent pr-10">
            {editingTransaction ? 'Editar Transação' : 'Nova Transação'}
            </h3>
            
            {/* Reutilizando seu formulário existente */}
            <form onSubmit={handleSubmit} className="space-y-4">
               <div>
                <label className="block text-sm font-medium text-zinc-300 mb-2">Tipo</label>
                <select
                  value={formData.type}
                  onChange={(e) => {
                    const newType = e.target.value as 'income' | 'expense'
                    setFormData({ ...formData, type: newType, category: '' })
                  }}
                  className="w-full rounded-lg bg-zinc-900 px-4 py-3 text-white focus:outline-none focus:ring-2 focus:ring-purple-500/50 transition-all border border-zinc-800"
                >
                  <option value="income" className="bg-zinc-900">Receita</option>
                  <option value="expense" className="bg-zinc-900">Despesa</option>
                </select>
              </div>

              <div>
                <label className="block text-sm font-medium text-zinc-300 mb-2">Categoria</label>
                <div className="relative" ref={dropdownRef}>
                  <button
                    type="button"
                    onClick={() => setCategoryDropdownOpen(!categoryDropdownOpen)}
                    className="w-full rounded-lg bg-zinc-900 px-4 py-3 text-white focus:outline-none focus:ring-2 focus:ring-purple-500/50 transition-all border border-zinc-800 flex items-center justify-between"
                  >
                    {selectedCategory ? (
                      <div className="flex items-center gap-3">
                        <selectedCategory.icon className="w-5 h-5" style={{ color: selectedCategory.color }} />
                        <span>{selectedCategory.label}</span>
                      </div>
                    ) : (
                      <span className="text-zinc-500">Selecione uma categoria</span>
                    )}
                    <ChevronDown className={cn("w-5 h-5 text-zinc-400 transition-transform", categoryDropdownOpen && "rotate-180")} />
                  </button>

                  {/* Dropdown de Categorias */}
                  {categoryDropdownOpen && (
                    <div className="absolute top-full left-0 right-0 mt-2 bg-zinc-900 border border-zinc-800 rounded-lg shadow-xl z-50 max-h-60 overflow-y-auto">
                      <div className="p-1 grid grid-cols-1 gap-1">
                        
                        {/* Lista as categorias existentes */}
                        {currentCategories.map((cat) => (
                          <button
                            key={cat.value}
                            type="button"
                            onClick={() => {
                              setFormData({ ...formData, category: cat.value })
                              setCategoryDropdownOpen(false)
                            }}
                            className="flex items-center gap-3 w-full p-2 hover:bg-zinc-800 rounded-md transition-colors"
                          >
                            <div className="w-8 h-8 rounded-full flex items-center justify-center bg-zinc-800" style={{ color: cat.color }}>
                              <cat.icon className="w-4 h-4" />
                            </div>
                            <span className="text-sm text-zinc-300">{cat.label}</span>
                            {formData.category === cat.value && (
                              <Check className="w-4 h-4 text-purple-500 ml-auto" />
                            )}
                          </button>
                        ))}

                        {/* 👇 BOTÃO DE ADICIONAR NOVA CATEGORIA */}
                        <button
                          type="button"
                          onClick={() => {
                            const newCatName = window.prompt(`Nova categoria de ${formData.type === 'income' ? 'Receita' : 'Despesa'}:`)
                            if (newCatName && newCatName.trim()) {
                              addCategory(newCatName, formData.type as 'income' | 'expense')
                            }
                          }}
                          className="flex items-center gap-3 w-full p-2 hover:bg-zinc-800/80 border-t border-zinc-800 mt-1 rounded-md transition-colors text-zinc-400 hover:text-white"
                        >
                          <div className="w-8 h-8 rounded-full flex items-center justify-center bg-zinc-800 border border-dashed border-zinc-600">
                            <Plus className="w-4 h-4" />
                          </div>
                          <span className="text-sm italic">Criar "{formData.type === 'income' ? 'Restituição' : 'Manutenção'}..."</span>
                        </button>

                      </div>
                    </div>
                  )}
                </div>
              </div>

              <div>
                <label className="block text-sm font-medium text-zinc-300 mb-2">Descrição</label>
                <input
                  type="text"
                  required
                  value={formData.description}
                  onChange={(e) => setFormData({ ...formData, description: e.target.value })}
                  className="w-full rounded-lg bg-zinc-900 px-4 py-3 text-white placeholder-zinc-500 focus:outline-none focus:ring-2 focus:ring-purple-500/50 transition-all border border-zinc-800"
                  placeholder="Ex: Salário, Aluguel, etc."
                />
              </div>

              <div>
                <label className="block text-sm font-medium text-zinc-300 mb-2">Valor</label>
                <input
                  type="number"
                  step="0.01"
                  required
                  value={formData.amount}
                  onChange={(e) => setFormData({ ...formData, amount: e.target.value })}
                  className="w-full rounded-lg bg-zinc-900 px-4 py-3 text-white placeholder-zinc-500 focus:outline-none focus:ring-2 focus:ring-purple-500/50 transition-all border border-zinc-800"
                  placeholder="0.00"
                />
              </div>

              {/* Seleção de Método de Pagamento */}
              <div className="mt-4">
                <label className="block text-sm font-medium text-zinc-300 mb-2">Método de Pagamento</label>
                
                {/* Botões Débito vs Crédito */}
                <div className="flex gap-2 p-1 bg-zinc-900 rounded-lg border border-zinc-800 mb-3">
                  <button
                    type="button"
                    onClick={() => { setPaymentMethod('debit'); setSelectedCardId(''); }}
                    className={cn("flex-1 py-2 text-sm font-medium rounded-md transition-all", paymentMethod === 'debit' ? "bg-zinc-700 text-white shadow" : "text-zinc-400 hover:text-zinc-200")}
                  >
                    💵 Débito / Pix
                  </button>
                  <button
                    type="button"
                    onClick={() => setPaymentMethod('credit')}
                    className={cn("flex-1 py-2 text-sm font-medium rounded-md transition-all", paymentMethod === 'credit' ? "bg-purple-600 text-white shadow" : "text-zinc-400 hover:text-zinc-200")}
                  >
                    💳 Crédito
                  </button>
                </div>

                {/* Se for Crédito, mostra a lista de cartões */}
                {paymentMethod === 'credit' && (
                  <div className="animate-in fade-in slide-in-from-top-2 p-3 bg-zinc-800/30 rounded-lg border border-zinc-800">
                    <label className="block text-xs font-medium text-zinc-400 mb-2">Selecione o Cartão</label>
                    <div className="grid grid-cols-2 gap-2">
                      {cards.map(card => (
                        <button
                          key={card.id}
                          type="button"
                          onClick={() => setSelectedCardId(card.id)}
                          className={cn("p-2 text-left rounded-lg border transition-all", selectedCardId === card.id ? "bg-purple-500/20 border-purple-500 text-white" : "bg-zinc-900 border-zinc-800 text-zinc-400 hover:border-zinc-700")}
                        >
                          <span className="block text-sm font-bold">{card.name}</span>
                          <span className="text-xs opacity-70">Fecha dia {card.closing_day}</span>
                        </button>
                      ))}
                      
                      {/* Botão Atalho para Configurar Cartões */}
                      <button
                         type="button"
                         onClick={() => { setShowModal(false); setShowCardSettings(true); }}
                         className="p-2 text-center rounded-lg border border-dashed border-zinc-700 text-zinc-500 hover:text-zinc-300 hover:border-zinc-500 flex items-center justify-center gap-1 text-xs"
                      >
                        <Plus className="w-3 h-3" /> Configurar
                      </button>
                    </div>
                  </div>
                )}
              </div>

              <div>
                <label className="block text-sm font-medium text-zinc-300 mb-2">Data</label>
                <input
                  type="date"
                  required
                  value={formData.date}
                  onChange={(e) => {
                    setFormData({ ...formData, date: e.target.value })
                    // 👇 Se for débito, a data de compra segue a data do input.
                    // Se for crédito, atualizamos a base de cálculo.
                    if (paymentMethod === 'debit') {
                       setPurchaseDate(e.target.value)
                    } else {
                       // Se o usuário mexer na data enquanto está em crédito, 
                       // assumimos que ele quer mudar a data da COMPRA, e recalculamos o vencimento
                       setPurchaseDate(e.target.value)
                    }
                  }}
                  className="w-full rounded-lg bg-zinc-900 px-4 py-3 text-white focus:outline-none focus:ring-2 focus:ring-purple-500/50 transition-all border border-zinc-800"
                />
              </div>

              {/* Opções Extras */}
              <div>
                 <div className="flex items-center justify-between p-3 bg-zinc-800/50 rounded-lg border border-zinc-800 mb-3">
                  <div className="flex items-center gap-3">
                    <div className="p-2 rounded-lg bg-purple-500/10"><CheckCircle2 className="w-4 h-4 text-purple-400" /></div>
                    <div>
                      <p className="text-sm font-medium text-white">{formData.type === 'expense' ? 'Despesa Fixa' : 'Receita Fixa'}</p>
                    </div>
                  </div>
                  <button type="button" onClick={() => setIsFixed(!isFixed)} className={cn("relative w-11 h-6 rounded-full transition-colors duration-200", isFixed ? "bg-green-500" : "bg-zinc-700")}>
                    <div className={cn("absolute top-1 left-1 w-4 h-4 rounded-full bg-white shadow transition-transform duration-200", isFixed && "translate-x-5")} />
                  </button>
                </div>

                <div className="flex items-center justify-between p-3 bg-zinc-800/50 rounded-lg border border-zinc-800">
                  <div className="flex items-center gap-3">
                    <div className="p-2 rounded-lg bg-blue-500/10"><Clock className="w-4 h-4 text-blue-400" /></div>
                    <div>
                      <p className="text-sm font-medium text-white">Repetir/Parcelar</p>
                    </div>
                  </div>
                  <button type="button" onClick={() => setIsRepeated(!isRepeated)} className={cn("relative w-11 h-6 rounded-full transition-colors duration-200", isRepeated ? "bg-green-500" : "bg-zinc-700")}>
                    <div className={cn("absolute top-1 left-1 w-4 h-4 rounded-full bg-white shadow transition-transform duration-200", isRepeated && "translate-x-5")} />
                  </button>
                </div>

                {isRepeated && (
                  <div className="mt-3 grid grid-cols-2 gap-3 p-3 bg-zinc-800/30 rounded-lg border border-zinc-800">
                      <div>
                        <label className="block text-xs font-medium text-zinc-400 mb-1">Qtd</label>
                        <input type="number" min="2" max="120" value={repeatCount} onChange={(e) => setRepeatCount(e.target.value)} className="w-full rounded-lg bg-zinc-900 px-3 py-2 text-white text-sm border border-zinc-800" />
                      </div>
                      <div>
                        <label className="block text-xs font-medium text-zinc-400 mb-1">Intervalo</label>
                        <select value={repeatInterval} onChange={(e) => setRepeatInterval(e.target.value as any)} className="w-full rounded-lg bg-zinc-900 px-3 py-2 text-white text-sm border border-zinc-800">
                          <option value="months">Meses</option>
                          <option value="weeks">Semanas</option>
                          <option value="years">Anos</option>
                        </select>
                      </div>
                  </div>
                )}
              </div>

               <div className="mt-4">
                <label className="block text-sm font-medium text-zinc-300 mb-2">Status</label>
                <select 
                  value={formData.status} 
                  onChange={(e) => setFormData({ ...formData, status: e.target.value as any })} 
                  className="w-full rounded-lg bg-zinc-900 px-4 py-3 text-white border border-zinc-800 outline-none focus:ring-2 focus:ring-purple-500/50"
                >
                  {formData.type === 'expense' ? (
                    <>
                      <option value="paid" className="bg-zinc-900">Pago</option>
                      {/* Se for crédito, visualmente mostramos A Pagar */}
                      <option value="pending" className="bg-zinc-900">A Pagar</option> 
                    </>
                  ) : (
                    <>
                      <option value="paid" className="bg-zinc-900">Recebido</option>
                      <option value="pending" className="bg-zinc-900">A Receber</option>
                    </>
                  )}
                </select>
              </div>

              <div className="flex gap-3 pt-4">
                <button type="button" onClick={() => setShowModal(false)} className="flex-1 px-4 py-3 border border-zinc-700 rounded-lg text-zinc-300 hover:bg-zinc-800 transition-all">Cancelar</button>
                <button 
                type="submit" 
                disabled={isSubmitting}
                className={cn(
                  "w-full py-3 font-bold rounded-lg mt-2 transition-all",
                  isSubmitting 
                    ? "bg-zinc-700 text-zinc-400 cursor-not-allowed" // Visual cinza quando estiver salvando
                    : "bg-green-600 hover:bg-green-500 text-white"   // Visual verde normal
                )}
              >
                {isSubmitting ? 'Salvando...' : 'Salvar Transação'}
              </button>
              </div>
            </form>
          </div>
        </div>
      )}

      {/* --- MODAL 2: SAÚDE FINANCEIRA --- */}
      {showHealthModal && (
        <div 
          className="fixed inset-0 bg-black/80 backdrop-blur-sm flex items-center justify-center z-50 p-4"
          onClick={() => setShowHealthModal(false)}
        >
          <div 
            className="bg-zinc-900 rounded-2xl border border-zinc-800/50 shadow-2xl shadow-black/50 max-w-lg w-full p-6 md:p-8 max-h-[90vh] overflow-y-auto relative"
            onClick={(e) => e.stopPropagation()}
          >
            <button
              type="button"
              onClick={() => setShowHealthModal(false)}
              className="absolute top-4 right-4 p-2 text-zinc-400 hover:text-white hover:bg-zinc-800 rounded-lg transition-all"
            >
              <X className="w-5 h-5" />
            </button>

            <div className="flex items-center gap-3 mb-6">
              <div className="p-3 rounded-xl bg-gradient-to-br from-violet-500/20 to-purple-500/20 border border-violet-500/30">
                <Activity className="w-6 h-6 text-violet-400" />
              </div>
              <div>
                <h3 className="text-2xl font-bold text-white">Saúde Financeira</h3>
                <p className="text-sm text-zinc-400">Análise do período selecionado</p>
              </div>
            </div>

            {/* Barra de Score */}
            <div className="mb-6">
              <div className="flex items-center justify-between mb-2">
                <span className="text-sm font-medium text-zinc-300">Pontuação</span>
                <span className={cn("text-sm font-bold", statusDetails.color)}>
                {statusDetails.label}
                </span>
              </div>
              <div className="relative w-full h-8 bg-zinc-800/50 rounded-full overflow-hidden border border-zinc-800">
                <div
                  className={cn("h-full rounded-full transition-all duration-700 ease-out bg-gradient-to-r", getHealthColor(healthScore))}
                  style={{ width: `${healthScore}%` }}
                />
                <div className="absolute inset-0 flex items-center justify-center">
                  <span className="text-sm font-bold text-white drop-shadow-lg">{healthScore}/100</span>
                </div>
              </div>
            </div>

            {/* Checklist Gamificado */}
            <div className="mb-6 space-y-3">
              <div className={cn("flex items-center gap-3 p-3 rounded-lg border transition-all", isPositiveBalance ? "bg-green-500/10 border-green-500/30" : "bg-zinc-800/30 border-zinc-800")}>
                <div className={cn("w-6 h-6 rounded-full flex items-center justify-center", isPositiveBalance ? "bg-green-500" : "bg-zinc-700")}>
                  {isPositiveBalance ? <Check className="w-4 h-4 text-white" /> : <X className="w-4 h-4 text-zinc-400" />}
                </div>
                <div className="flex-1">
                  <p className={cn("text-sm font-medium", isPositiveBalance ? "text-green-400" : "text-zinc-400")}>Fechar no Azul</p>
                  <p className="text-xs text-zinc-500">Saldo: {formatCurrency(monthBalance)}</p>
                </div>
                <span className={cn("text-xs font-bold", isPositiveBalance ? "text-green-400" : "text-zinc-600")}>+30pts</span>
              </div>

              <div className={cn("flex items-center gap-3 p-3 rounded-lg border transition-all", isFixedOk ? "bg-green-500/10 border-green-500/30" : "bg-zinc-800/30 border-zinc-800")}>
                <div className={cn("w-6 h-6 rounded-full flex items-center justify-center", isFixedOk ? "bg-green-500" : "bg-zinc-700")}>
                  {isFixedOk ? <Check className="w-4 h-4 text-white" /> : <X className="w-4 h-4 text-zinc-400" />}
                </div>
                <div className="flex-1">
                  <p className={cn("text-sm font-medium", isFixedOk ? "text-green-400" : "text-zinc-400")}>Gastos Fixos {'<'} 50%</p>
                  <p className="text-xs text-zinc-500">Atual: {fixedPercentage.toFixed(1)}%</p>
                </div>
                <span className={cn("text-xs font-bold", isFixedOk ? "text-green-400" : "text-zinc-600")}>+40pts</span>
              </div>

              <div className={cn("flex items-center gap-3 p-3 rounded-lg border transition-all", isSavingsOk ? "bg-green-500/10 border-green-500/30" : "bg-zinc-800/30 border-zinc-800")}>
                <div className={cn("w-6 h-6 rounded-full flex items-center justify-center", isSavingsOk ? "bg-green-500" : "bg-zinc-700")}>
                  {isSavingsOk ? <Check className="w-4 h-4 text-white" /> : <X className="w-4 h-4 text-zinc-400" />}
                </div>
                <div className="flex-1">
                  <p className={cn("text-sm font-medium", isSavingsOk ? "text-green-400" : "text-zinc-400")}>Poupar {'>'} 20%</p>
                  <p className="text-xs text-zinc-500">Atual: {savingsPercentage.toFixed(1)}%</p>
                </div>
                <span className={cn("text-xs font-bold", isSavingsOk ? "text-green-400" : "text-zinc-600")}>+30pts</span>
              </div>
            </div>

            {/* Box Educativo Expansível */}
            <div className="rounded-lg bg-zinc-800/50 border border-zinc-700 overflow-hidden transition-all duration-300">
              <button 
                type="button"
                onClick={() => setShowHealthInfo(!showHealthInfo)}
                className="w-full flex items-center gap-3 p-4 text-left hover:bg-zinc-800 transition-colors"
              >
                <div className="p-2 rounded-lg bg-blue-500/10 mt-0.5">
                  <Info className="w-4 h-4 text-blue-400" />
                </div>
                <div className="flex-1">
                  <h5 className="text-sm font-semibold text-white mb-1 flex items-center justify-between">
                    Metodologia 50-30-20
                    <ChevronDown className={cn("w-4 h-4 text-zinc-400 transition-transform duration-300", showHealthInfo && "rotate-180")} />
                  </h5>
                  <p className="text-xs text-zinc-400">
                    Clique para entender como sua nota é calculada.
                  </p>
                </div>
              </button>

              {/* Área do Texto Completo (Só aparece se clicar) */}
              {showHealthInfo && (
                <div className="px-4 pb-4 pt-0 text-xs text-zinc-300 space-y-3 border-t border-zinc-700/50 mt-2">
                  <div className="pt-3">
                    <p className="mb-2"><strong className="text-white">Como funciona:</strong> O ideal é dividir sua renda em:</p>
                    <ul className="space-y-1 pl-1">
                      <li>🏠 <strong className="text-white">50% Essenciais:</strong> Aluguel, contas, fixos.</li>
                      <li>🎉 <strong className="text-white">30% Estilo de Vida:</strong> Lazer, iFood, compras.</li>
                      <li>💰 <strong className="text-white">20% Futuro:</strong> Investimentos e reservas.</li>
                    </ul>
                  </div>
                  
                  <div>
                    <p className="mb-1"><strong className="text-white">O que significa seu Status?</strong></p>
                    <ul className="grid grid-cols-1 gap-1">
                      <li>🤩 <span className="text-green-400">Excelente:</span> Sobra dinheiro e fixos baixos.</li>
                      <li>🙂 <span className="text-yellow-400">Equilibrado:</span> Paga as contas, mas sobra pouco.</li>
                      <li>🥵 <span className="text-orange-400">Sufocado:</span> Fixos acima de 70%. Perigo!</li>
                      <li>🚨 <span className="text-red-400">Crítico:</span> Gastou mais do que recebeu.</li>
                    </ul>
                  </div>
                  <p className="text-zinc-500 italic border-t border-zinc-700/50 pt-2 mt-2">
                    "Saúde financeira não é sobre quanto ganha, mas como equilibra os gastos."
                  </p>
                </div>
              )}
            </div>

            <button
              type="button"
              onClick={() => setShowHealthModal(false)}
              className="w-full mt-6 px-4 py-3 bg-gradient-to-r from-violet-500 to-purple-500 text-white font-semibold rounded-lg hover:from-violet-400 hover:to-purple-400 transition-all shadow-lg shadow-purple-500/30"
            >
              Entendido
            </button>
          </div>
        </div> // fecha div modal saúde
      )}

      {/* MODAL 3 - Gerenciar Cartões */}
      {showCardSettings && (
        <div className="fixed inset-0 bg-black/80 backdrop-blur-sm flex items-center justify-center z-[9999] p-4">
          <div className="bg-zinc-950 rounded-2xl border border-zinc-800 w-full max-w-md p-6">
            <div className="flex justify-between items-center mb-6">
              <h3 className="text-xl font-bold text-white">Meus Cartões</h3>
              <button onClick={() => { setShowCardSettings(false); setEditingCard(null); }} className="text-zinc-400 hover:text-white">
                <X className="w-5 h-5"/>
              </button>
            </div>

            {/* Lista de Cartões Existentes */}
            <div className="space-y-3 mb-6 max-h-48 overflow-y-auto">
              {cards.map(card => (
                <div key={card.id} className="flex items-center justify-between p-3 bg-zinc-900 rounded-lg border border-zinc-800">
                  <div className="flex items-center gap-3">
                    <div className="w-3 h-3 rounded-full" style={{ backgroundColor: card.color }}></div>
                    <div>
                      <p className="font-bold text-white text-sm">{card.name}</p>
                      <p className="text-xs text-zinc-400">Fecha dia {card.closing_day} • Vence dia {card.due_day}</p>
                    </div>
                  </div>
                  <div className="flex gap-1">
                    {/* Botão Editar Cartão */}
                    <button 
                      onClick={() => {
                        setEditingCard(card)
                        setSelectedCardColor(card.color)
                      }} 
                      className="p-2 text-zinc-500 hover:text-blue-400 hover:bg-blue-500/10 rounded transition-colors"
                    >
                      <Pencil className="w-4 h-4"/>
                    </button>
                    {/* Botão Excluir Cartão */}
                    <button onClick={() => deleteCard(card.id)} className="p-2 text-zinc-500 hover:text-red-400 hover:bg-red-500/10 rounded transition-colors">
                      <Trash2 className="w-4 h-4"/>
                    </button>
                  </div>
                </div>
              ))}
              {cards.length === 0 && <p className="text-center text-zinc-500 text-sm">Nenhum cartão cadastrado.</p>}
            </div>

            {/* Formulário de Adicionar / Editar */}
            <form 
              key={editingCard ? editingCard.id : 'new'} // Força o formulário a resetar quando troca de modo
              onSubmit={async (e: any) => { 
                e.preventDefault();
                const formData = {
                  name: e.target.name.value, 
                  closing_day: +e.target.closing.value, 
                  due_day: +e.target.due.value, 
                  color: selectedCardColor 
                }

                if (editingCard) {
                  // Modo Edição
                  await updateCard(editingCard.id, formData) // Você precisa adicionar updateCard no useCards.ts (Passo 1)
                  setEditingCard(null) // Sai do modo edição
                } else {
                  // Modo Criação
                  await addCard(formData); 
                }
                
                e.target.reset() 
                setSelectedCardColor('#8b5cf6') // Reseta cor
              }} 
              className="pt-4 border-t border-zinc-800 grid gap-3"
            >
              <div className="flex justify-between items-center">
                <h4 className="text-sm font-medium text-white">
                  {editingCard ? `Editando: ${editingCard.name}` : 'Adicionar Novo'}
                </h4>
                {editingCard && (
                  <button type="button" onClick={() => { setEditingCard(null); setSelectedCardColor('#8b5cf6'); }} className="text-xs text-zinc-500 hover:text-white underline">
                    Cancelar Edição
                  </button>
                )}
              </div>

              <input 
                name="name" 
                placeholder="Nome (ex: Nubank)" 
                required 
                defaultValue={editingCard?.name}
                className="bg-zinc-900 border border-zinc-800 p-3 rounded-lg w-full outline-none focus:border-purple-500 transition-colors text-white text-sm" 
              />
              
              <div className="flex gap-2">
                <input 
                  name="closing" 
                  type="number" 
                  min="1" max="31" 
                  placeholder="Dia Fechamento" 
                  required 
                  defaultValue={editingCard?.closing_day}
                  className="bg-zinc-900 border border-zinc-800 p-3 rounded-lg flex-1 outline-none focus:border-purple-500 transition-colors text-white text-sm" 
                />
                <input 
                  name="due" 
                  type="number" 
                  min="1" max="31" 
                  placeholder="Dia Vencimento" 
                  required 
                  defaultValue={editingCard?.due_day}
                  className="bg-zinc-900 border border-zinc-800 p-3 rounded-lg flex-1 outline-none focus:border-purple-500 transition-colors text-white text-sm" 
                />
              </div>

              {/* Seletor de Cores */}
              <div>
                <p className="text-xs text-zinc-500 mb-2">Cor da Tag</p>
                <div className="flex gap-3 justify-between">
                  {['#ef4444', '#f97316', '#eab308', '#22c55e', '#06b6d4', '#3b82f6', '#8b5cf6', '#ec4899'].map(color => (
                    <button
                      key={color}
                      type="button"
                      onClick={() => setSelectedCardColor(color)}
                      className={cn("w-6 h-6 rounded-full transition-transform hover:scale-110", selectedCardColor === color ? "ring-2 ring-white scale-110" : "")}
                      style={{ backgroundColor: color }}
                    />
                  ))}
                </div>
              </div>

              <button className={cn("w-full p-3 rounded-lg font-bold text-white mt-2 transition-colors", editingCard ? "bg-blue-600 hover:bg-blue-500" : "bg-purple-600 hover:bg-purple-500")}>
                {editingCard ? 'Salvar Alterações' : 'Cadastrar Cartão'}
              </button>
            </form>
          </div>
        </div>
      )}
    </div> // fecha div min-h-screen (Principal do App)
  )
}
