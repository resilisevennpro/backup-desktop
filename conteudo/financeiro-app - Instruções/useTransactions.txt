'use client'

import { useState, useEffect, useCallback } from 'react'
import { supabase } from '@/lib/supabase'

export interface Transaction {
  id: string
  description: string
  amount: number
  type: 'income' | 'expense'
  category: string
  status: 'paid' | 'pending'
  date: string
  is_fixed?: boolean
  installment_number?: number
  installment_total?: number
  payment_method?: 'debit' | 'credit'
  card_id?: string
  user_id?: string
  group_id?: string
}

export function useTransactions() {
  const [transactions, setTransactions] = useState<Transaction[]>([])
  const [loading, setLoading] = useState(true)

  const fetchTransactions = useCallback(async () => {
    try {
      const { data: { user } } = await supabase.auth.getUser()
      if (!user) {
        setTransactions([])
        return
      }

      const { data, error } = await supabase
        .from('transactions')
        .select('*')
        .eq('user_id', user.id)
        .order('date', { ascending: false })
      
      if (error) throw error
      setTransactions(data || [])
    } catch (error) {
      console.error('Erro ao buscar transações:', error)
    } finally {
      setLoading(false)
    }
  }, [])

  const addTransaction = async (transaction: Omit<Transaction, 'id' | 'user_id' | 'created_at'>) => {
    const { data: { user } } = await supabase.auth.getUser()
    if (!user) return

    const { error } = await supabase
      .from('transactions')
      .insert([{ ...transaction, user_id: user.id }])

    if (error) throw error
    await fetchTransactions()
  }

  const deleteTransaction = async (id: string, groupId?: string) => {
    let shouldDeleteAll = false
    
    if (groupId) {
      const confirmAll = window.confirm("Esta é uma transação recorrente.\n\n[OK] = Apagar TODAS as parcelas\n[Cancelar] = Apagar APENAS ESTA")
      shouldDeleteAll = confirmAll
    } else {
      if (!window.confirm("Tem certeza que deseja excluir esta transação?")) {
        return
      }
    }

    let query = supabase.from('transactions').delete()

    if (shouldDeleteAll && groupId) {
      query = query.eq('group_id', groupId)
    } else {
      query = query.eq('id', id)
    }

    const { error } = await query
    if (error) throw error
    
    await fetchTransactions()
  }

  // 🔥 LÓGICA CORRIGIDA AQUI
  const updateTransaction = async (id: string, updates: Partial<Transaction>, groupId?: string) => {
    let shouldUpdateAll = false

    if (groupId) {
       const confirmAll = window.confirm("Esta é uma transação recorrente.\n\n[OK] = Aplicar alterações em TODAS (futuras e passadas)\n[Cancelar] = Aplicar APENAS NESTA")
       shouldUpdateAll = confirmAll
    }

    if (shouldUpdateAll && groupId) {
      // 🛑 PROTEÇÃO: Removemos 'date' E 'status' da atualização em lote.
      // Assim, se você mudar o nome ou valor, muda em todas.
      // Mas se mudar para 'Pago', só muda na atual (feita no bloco abaixo).
      const { date, status, ...updatesForGroup } = updates
      
      // 1. Atualiza dados gerais (Nome, Valor, Categoria) em todas
      const { error: groupError } = await supabase
        .from('transactions')
        .update(updatesForGroup)
        .eq('group_id', groupId)
      
      if (groupError) throw groupError

      // 2. Aplica TUDO (incluindo data e status) APENAS na transação que você clicou
      // Isso garante que se você marcou "Pago" nesta, ela fica paga, mas as outras não mudam.
      const { error: singleError } = await supabase
        .from('transactions')
        .update(updates) // Aqui vão todos os campos, inclusive status
        .eq('id', id)

      if (singleError) throw singleError

    } else {
      // Atualização Simples (Apenas esta)
      const { error } = await supabase
        .from('transactions')
        .update(updates)
        .eq('id', id)

      if (error) throw error
    }
    
    await fetchTransactions()
  }

  useEffect(() => {
    fetchTransactions()
    
    const channel = supabase
      .channel('transactions_changes')
      .on('postgres_changes', { event: '*', schema: 'public', table: 'transactions' }, () => {
        fetchTransactions()
      })
      .subscribe()

    return () => {
      supabase.removeChannel(channel)
    }
  }, [fetchTransactions])

  return { 
    transactions, 
    loading, 
    addTransaction, 
    deleteTransaction, 
    updateTransaction
  }
}
